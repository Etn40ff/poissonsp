\begin{abstract}
  We modify the seeds of Berenstein, Fomin and Zelevinsky attached to reduced words for $(w_0,w_0)$, for any simply connected simple complex algebraic group and any aperiodic Belavin--Drinfeld pair.
  In every type, moves of words act on the resulting seeds by mutations. After a change of coordinates introduced by Gekhtman, Shapiro and Vainshtein, the seeds are, for $SL_N$, seeds of the cluster structures of these authors, and for $Sp_{2n}$ they define regular complete cluster structures with the expected properties. This proves the aperiodic version of their conjecture for $Sp_{2n}$, with its uniqueness part in a refined form.
\end{abstract}

\section{Introduction}
\label{sec: intro}

Let $\mcG$ be a simply connected simple complex algebraic group of rank $r$.
Belavin and Drinfeld \cite{BD82} classified the quasitriangular solutions of the classical Yang--Baxter equation on $\mfg=\mathrm{Lie}(\mcG)$ in terms of combinatorial data, which we encode as \emph{BD maps} $\gamma_\bullet:\Gamma_\bullet\to\Gamma$ between subsets of simple roots.
A pair $\gamma=(\gamma_r,\gamma_c)$ of BD maps, together with some Cartan data $s\in\mcS(\gamma)$, defines a Poisson bracket $\{\cdot,\cdot\}_{\gamma,s}$ on $\mcG$, where $\gamma_r$ and $\gamma_c$ govern the behavior under left and right multiplication, respectively.
For $\gamma=(0,0)$ and $s=0$ this is the standard Poisson--Lie structure, which is \emph{compatible} with the cluster structure of Berenstein, Fomin and Zelevinsky \cite{BFZ05}, i.e.\ log-canonical with respect to every extended cluster.
Gekhtman, Shapiro and Vainshtein conjectured \cite[Conjecture~3.2]{GSV11} that every Poisson--Lie structure of this kind is compatible with a regular complete cluster structure on $\mcG$.
In the setting of pairs \cite{GSV19}, every \emph{aperiodic} pair $\gamma$ (\cref{def: aperiodic}) should admit a cluster structure $\mcC_\gamma$ on $\mcG$ with the following properties.
\begin{enumerate}[(i)]
  \item The number of frozen variables is $2r-|\Gamma_r|-|\Gamma_c|$, and the exchange matrices have full rank.
  \item $\mcC_\gamma$ is regular and its upper cluster algebra is $\bbC[\mcG]$.
  \item The global toric action on $\mcC_\gamma$ is generated by the action of $\mcH_{\gamma_r}\times\mcH_{\gamma_c}$ by left and right multiplication.
  Here $\mcH_{\gamma_\bullet}$ is the identity component of the subgroup of the maximal torus on which the characters $\alpha$ and $\gamma_\bullet(\alpha)$ agree for every $\alpha\in\Gamma_\bullet$.
  \item $\mcC_\gamma$ is compatible with $\{\cdot,\cdot\}_{\gamma,s}$ for every $s\in\mcS(\gamma)$.
  \item The Poisson brackets compatible with $\mcC_\gamma$ are exactly the brackets $c\{\cdot,\cdot\}_{\gamma,s}+\mcV_\Xi$, where $c\in\bbC$, $s\in\mcS(\gamma)$, and $\mcV_\Xi$ is a bivector field built from the infinitesimal action of $\mcH_{\gamma_r}\times\mcH_{\gamma_c}$.
\end{enumerate}
Property (v) extends \cite[Conjecture~3.2(v)]{GSV11}, which concerns Poisson--Lie brackets, to all Poisson brackets. For Poisson--Lie brackets, (v) gives the conclusion of \cite[Conjecture~3.2(v)]{GSV11} when $c\neq0$, while for $c=0$ some toric brackets are also Poisson--Lie (\cref{rmk: v}).
For $\mcG=SL_N$, cluster structures with (i)--(iv) are constructed in \cite[Theorem~3.3]{GSV19} for aperiodic oriented pairs, and cluster structures with (i), (ii) and (iv) in \cite{GSV23} for all aperiodic pairs; the latter also have (iii) and (v) (\cref{thm: A}(c), \cref{thm: uniqueness}). In the latter construction, the initial seed is described by a quiver, and its cluster variables are obtained from standard ones through a rational map $h_\gamma$, defined in \cite{GSV23} for every $\mcG$.
In this paper we give seeds for every $\mcG$ and every double reduced word, and, using the same rational map $h_\gamma$, prove the conjecture for $\mcG=Sp_{2n}$.

\subsection*{Main results}
For a double reduced word $\bfi$, i.e.\ a reduced word for $(w_0,w_0)$ in $W\times W$, let $\Sigma(\bfi)$ be the seed of \cite[Theorem~2.10]{BFZ05}, whose cluster variables are generalized minors indexed by the chambers of $\bfi$.
The seed $\Sigma^\Phi_\gamma(\bfi)$ of \cref{def: model} is obtained from $\Sigma(\bfi)$ by three operations.
We thaw one end chamber for each root in $\Gamma_r\sqcup\Gamma_c$.
We add to the exchange matrix the entries of some virtual letters and dipoles determined by $\gamma$.
Finally, we multiply each minor by a monomial in the minors of the end chambers.

\begin{theorem}
  \label{thm: A}
  Let $\gamma$ be an aperiodic BD pair of $\mcG$.
  \begin{enumerate}[(a)]
    \item The seeds $\Sigma^\Phi_\gamma(\bfi)$ of all double reduced words are mutation equivalent with moves on words corresponding to (sequences of) mutations (\cref{thm: moves,cor: mutation equivalent}).
    \item With the thawed chambers frozen, $\Sigma^\Phi_\gamma(\bfi)$ and $\Sigma(\bfi)$ are quasi-isomorphic in the sense of \cite{Fra16} via the identity of $\bbC(\mcG)$ (\cref{cor: Fraser}).
    \item If $\mcG=SL_N$, the seeds $\Sigma^\Phi_\gamma(\bfi)$ composed with $h_\gamma^{-1}$ are seeds of the cluster structure of \cite{GSV23}, and its global toric action is generated by $\mcH_{\gamma_r}\times\mcH_{\gamma_c}$ (\cref{sec: type A concise}).
  \end{enumerate}
\end{theorem}

Whenever $h_\gamma$ is birational, (a) holds for the \emph{BD word seeds} $\Sigma_\gamma(\bfi)$, obtained by composing the functions of $\Sigma^\Phi_\gamma(\bfi)$ with $h_\gamma^{-1}$. This is the case for $SL_N$ \cite{GSV23} and, as we show, for $Sp_{2n}$.

\begin{theorem}
  \label{thm: main}
  Let $\gamma$ be an aperiodic BD pair of type $C_n$.
  Then $h_\gamma$ is birational, and the BD word seeds $\Sigma_\gamma(\bfi)$ are seeds of a single cluster structure $\mcC_\gamma$ on $\Sp$ (\cref{thm: all words}), which satisfies (i)--(v) (\cref{thm: structure,thm: complete,thm: uniqueness}).
\end{theorem}

As $-w_0$ acts trivially on the simple roots of type $C_n$, a Poisson--Lie pair $\gamma_c=\gamma_r^{-1}\neq0$ is never aperiodic, so the nontrivial Poisson--Lie classes of $Sp_{2n}$ are not covered by \cref{thm: main} (\cref{rmk: aperiodic not PL}).

The group $\Sp$ is the fixed point set of an involution $\vt$ of $\SL$, and $\gamma$ lifts to a $\vt$-invariant aperiodic pair of type $A_{2n-1}$.
The BD word seeds of lifted words are $\vt$-symmetric seeds of the cluster structures of \cite{GSV23}, and their folds are the BD word seeds of $\Sp$ (\cref{thm: fold model}).
We deduce the properties of $\mcC_\gamma$ from those of the ambient cluster structure, using that $\bbC[\Sp]$ is factorial for completeness.

\subsection*{Organization}
\Cref{sec: conventions} fixes conventions and recalls the results of \cite{GSV19,GSV23}.
\Cref{sec: model} defines the BD word seeds for every $\mcG$ and proves \cref{thm: A}(a) and~(b), and \cref{sec: type A concise} proves \cref{thm: A}(c).
\Cref{sec: lift} describes $\Sp$ inside $\SL$, BD pairs of type $C_n$ and their lifts, and the restriction of brackets.
\Cref{sec: folding} treats the folding of symmetric seeds, \cref{sec: symmetric} shows that the BD word seeds of lifted words are symmetric with the BD word seeds of $\Sp$ as folds, and \cref{sec: main} proves \cref{thm: main}.

\section{Conventions and the results of Gekhtman, Shapiro and Vainshtein}
\label{sec: conventions}
For integers $a\le b$ we write $[a,b]:=\{a,\dots,b\}$; for $c\in\bbZ$, $[c]_+:=\max(c,0)$; and for a statement $P$, $[P]$ is $1$ if $P$ holds and $0$ otherwise.

\subsection{Cluster algebras}
\label{sec: cluster conventions}
We use the conventions of \cite{BFZ05}; in particular exchange matrices are \emph{tall}.
A seed is a pair $(\bx,\tB)$, where $\bx=(x_p)_{p\in P}$ is an extended cluster of algebraically independent generators of a field.
Its finite index set $P=P_\ex\sqcup P_\fr$ is partitioned into exchangeable and frozen indices.
The integer matrix $\tB=(b_{pq})_{p\in P,q\in P_\ex}$ has skew-symmetrizable principal part $B=(b_{pq})_{p,q\in P_\ex}$.
We call $B$ \newword{block-indecomposable} if no permutation of $P_\ex$ makes it block diagonal, i.e.\ if the graph on $P_\ex$ with edges $\{p,q\}$, $b_{pq}\neq0$, is connected; as $\mu_k$ preserves block decompositions, this is a mutation invariant.
The mutation in direction $k\in P_\ex$ replaces $x_k$ by $x'_k$ with
\begin{equation}
  \label{eq: exchange}
  x_kx'_k=\prod_{q\in P}x_q^{[b_{qk}]_+}+\prod_{q\in P}x_q^{[-b_{qk}]_+},
\end{equation}
and $\tB$ by $\mu_k(\tB)$, with entries $-b_{pq}$ if $p=k$ or $q=k$, and $b_{pq}+[b_{pk}]_+[b_{kq}]_+-[-b_{pk}]_+[-b_{kq}]_+$ otherwise.
We write $\mu_k(\bx)$ for the new extended cluster and $\mu_k(\bx,\tB)=(\mu_k(\bx),\mu_k(\tB))$ for the new seed.
The papers \cite{GSV10,GSV19,GSV23} use \emph{wide} matrices, which are the negative transposes of ours. Both conventions give the same exchange relations, and $\mu_k(-B)=-\mu_k(B)$.
For an extended cluster $\by$ let $\mcL(\by)$ be the ring of Laurent polynomials in the exchangeable variables of $\by$ with coefficients polynomials in the frozen ones. Frozen variables are never inverted.
The \newword{upper cluster algebra} $\oA:=\oA(\bx)$ is the intersection of the rings $\mcL(\by)$ over the extended clusters $\by$ mutation equivalent to $\bx$; it only depends on the cluster structure defined by $(\bx,\tB)$.
If $\tB$ has full rank, $\oA(\bx)$ equals the \newword{upper bound} $\mcL(\bx)\cap\bigcap_{k\in P_\ex}\mcL(\mu_k(\bx))$ \cite[Theorem~3.11]{GSV18dd}. This result is stated for generalized cluster structures and for a ground ring in which frozen variables need not be inverted. Ordinary cluster structures are the case of trivial strings; see also \cite[Theorem~3.21 and Section~3.4]{GSV10}.
A cluster structure on an irreducible variety $X$ is a mutation class of seeds in $\bbC(X)$.
  It is \newword{regular} if all cluster variables are regular functions on $X$, and \newword{complete} if moreover $\oA=\bbC[X]$.

A Poisson bracket $\{\cdot,\cdot\}$ on the field containing $\bx$ is \newword{log-canonical} with respect to $\bx$ if $\{x_p,x_q\}=\omega_{pq}x_px_q$ with $\omega_{pq}\in\bbC$, and \newword{compatible} with a cluster structure if it is log-canonical with respect to every extended cluster.
Assume that the bracket is log-canonical with respect to $\bx$, with coefficient matrix $\Omega=(\omega_{pq})$, and that $\tB$ has full rank.
Then it is compatible if and only if
\begin{equation}
  \label{eq: compatibility}
  \sum_{r\in P}b_{rp}\,\omega_{rq}=\delta_{pq}\,\kappa_p
  \qquad(p\in P_\ex,\ q\in P)
\end{equation}
for some $\kappa_p\in\bbC$ \cite[Proposition~2.2]{GSV11}, see also \cite[Theorem~4.5]{GSV10}, stated there for the wide matrix $-\tB^T$.
Some $\kappa_p$ may vanish, for example for the zero bracket.
Conversely, \eqref{eq: compatibility} with all $\kappa_p\neq0$ implies both compatibility and full rank of $\tB$ \cite[Proposition~2.5]{GSV18dd}.
Since $\Omega$ is skew-symmetric, so is $\tB^T\Omega\tB$, whose entry at $(p,q)\in P_\ex^2$ is $\kappa_pb_{pq}$ by \eqref{eq: compatibility}; the $\kappa_p$ need not be positive.
An integer matrix $W=(w_{pj})$ with rows indexed by $P$ and columns indexed by $[1,m]$ defines an action of the torus $(\bbC^*)^m$ on the field generated by $\bx$, the \newword{local toric action} $x_p\mapsto x_p\prod_jt_j^{w_{pj}}$ for $(t_1,\dots,t_m)\in(\bbC^*)^m$.
  Its \newword{rank} is the rank of $W$, i.e.\ the dimension of the image of $(\bbC^*)^m$ in the group of automorphisms of the field.
It is a \newword{global toric action} if every cluster variable of every seed mutation equivalent to $(\bx,\tB)$ is homogeneous for it, so that it rescales all cluster variables.
  This holds if and only if $\tB^TW=0$, i.e.\ the two monomials of every exchange relation have the same weight \cite[Lemma~5.3]{GSV10}.
If $\tB$ has full rank, $\ker\tB^T$ has dimension $|P_\fr|$, so the rank of a global toric action is at most $|P_\fr|$ \cite[Section~2.2]{GSV11}.

\subsection{Lie theory}
\label{sec: Lie}
Let $\mcG$ be a simply connected simple complex algebraic group of rank $r$, with Lie algebra $\mfg$.
We fix a maximal torus $\mcH$, with Lie algebra $\mfh$, and a pinning: simple roots $\Gamma=\{\alpha_i\}_{i\in[1,r]}$, simple coroots $\alpha_i^\vee$, and Chevalley generators $E_i,F_i$ with $[E_i,F_i]=\alpha_i^\vee$.
We write $\lambda(\zeta)$ for the pairing of $\lambda\in\mfh^*$ and $\zeta\in\mfh$, $\varpi_i$ for the fundamental weights, and $a_{ij}=\alpha_j(\alpha_i^\vee)$ for the Cartan matrix entries.
  For $C_n$, $a_{n-1,n}=-2$ and $a_{n,n-1}=-1$.

\emph{Subgroups.}
Let $\mfn_\pm$ be the nilpotent subalgebras spanned by the positive, resp.\ negative, root vectors, $\mcN_\pm=\exp\mfn_\pm$ the corresponding maximal unipotent subgroups, and $\mcB_\pm=\mcH\mcN_\pm$ the opposite Borel subgroups.
For $K\subseteq\Gamma$, let $\mcG^K$ be the Levi subgroup generated by $\mcH$ and the one-parameter subgroups $\exp(tE_i)$, $\exp(tF_i)$, $\alpha_i\in K$; we write $\mcN_\pm^K=\mcN_\pm\cap\mcG^K$ and $\mcB_\pm^K=\mcB_\pm\cap\mcG^K$.

\emph{The Weyl group.}
Let $W=N_\mcG(\mcH)/\mcH$ be the Weyl group, generated by the simple reflections $s_i$, with length function $\ell$ and longest element $w_0$.
A \newword{reduced word} for $w\in W$ is a sequence $(j_1,\dots,j_m)$ with $w=s_{j_1}\cdots s_{j_m}$ and $m=\ell(w)$.
For $w\in W$ let $\overline{w}\in\mcG$ be the representative of \cite[Section~1.4]{FZ99}: $\overline{s_i}:=\exp(-E_i)\exp(F_i)\exp(-E_i)$, and $\overline{uv}=\overline{u}\,\overline{v}$ whenever $\ell(uv)=\ell(u)+\ell(v)$.
For $K\subseteq\Gamma$, the Weyl group of $\mcG^K$ is the parabolic subgroup $W_K$ generated by the $s_i$, $\alpha_i\in K$, and $w_0^K$ is its longest element.
The involution $i\mapsto i^*$ of $[1,r]$ is defined by $w_0(\alpha_i)=-\alpha_{i^*}$; then $w_0(\alpha_i^\vee)=-\alpha_{i^*}^\vee$ and $w_0\varpi_i=-\varpi_{i^*}$.
If $(j_1,\dots,j_m)$ is a reduced word for $w\in W$, the roots $s_{j_1}\cdots s_{j_{t-1}}(\alpha_{j_t})$, $t\in[1,m]$, are positive and pairwise distinct, and they are the positive roots $\beta$ with $w^{-1}\beta<0$ \cite[Chapter~VI, Section~1.6, Corollary~2 of Proposition~17]{Bou02}, applied to $w^{-1}$.
  For $w=w_0$ every positive root occurs exactly once.

\emph{The invariant form.}
Fix a non-degenerate invariant symmetric bilinear form $\Kil{\cdot}{\cdot}$ on $\mfg$.
Let $t\in\mfg\otimes\mfg$ be the Casimir element, $t=t_0+t_{-+}+t_{+-}$ with $t_0\in\mfh\otimes\mfh$, $t_{-+}\in\mfn_-\otimes\mfn_+$, $t_{+-}=t_{-+}^{21}$, where $z^{21}$ denotes $z\in\mfg\otimes\mfg$ with its two tensor factors swapped.
We write $\xi\wedge\eta:=\xi\otimes\eta-\eta\otimes\xi$, and $\lambda\otimes1:\mfh\otimes\mfh\to\mfh$ for $\zeta\otimes\zeta'\mapsto\lambda(\zeta)\zeta'$.

\emph{Generalized minors.}
Multiplication $\mcN_-\times\mcH\times\mcN_+\to\mcG$ is an isomorphism onto an open dense subset $\mcN_-\mcH\mcN_+$, the \newword{big cell}.
  For $U$ in it we write $U=U_-U_0U_+$, $U_\pm\in\mcN_\pm$, $U_0\in\mcH$, for its \newword{Gauss factorization}, which is unique.
Following \cite{FZ99}, let $\Delta_{\varpi_i,\varpi_i}$ be the regular function on $\mcG$ given by $\Delta_{\varpi_i,\varpi_i}(U)=U_0^{\varpi_i}$ on the big cell.
For $u,v\in W$ let the \newword{generalized minor} $\Delta_{u\varpi_i,v\varpi_i}$ be the function $U\mapsto\Delta_{\varpi_i,\varpi_i}(\overline{u}^{\,-1}U\overline{v})$.
For $SL_N$ with the standard pinning these are the minors of the matrix $U$ (\cref{sec: type A concise}).
The generalized minor depends only on the weights $u\varpi_i$ and $v\varpi_i$.
Let $U\mapsto U^T$ be the anti-involution of $\mcG$ whose differential swaps $E_i$ and $F_i$ and fixes $\mfh$; for $SL_N$ it is the transpose.
Then $\overline{w}^{\,T}=\overline{w}^{\,-1}$, so $\Delta_{u\varpi_i,v\varpi_i}(U^T)=\Delta_{v\varpi_i,u\varpi_i}(U)$.
A function $f$ on $\mcG$ is \newword{$\mcH$-homogeneous of bidegree} $(\lambda,\mu)$ if $f(y_rUy_c)=y_r^\lambda y_c^\mu f(U)$ for $y_r,y_c\in\mcH$; for instance $\Delta_{u\varpi_i,v\varpi_i}$ has bidegree $(u\varpi_i,v\varpi_i)$.

\subsection{Belavin--Drinfeld data and Poisson brackets}
\label{sec: BD}
A \newword{BD map} is an injective map $\gamma_\bullet:\Gamma_\bullet\subseteq\Gamma\to\Gamma$ which is an isometry for $\Kil{\cdot}{\cdot}$ and is nilpotent.
Nilpotency means that for every $\alpha\in\Gamma_\bullet$ some $\gamma_\bullet^m(\alpha)$ is not in $\Gamma_\bullet$.
Thus this is a BD triple $(\Gamma_\bullet,\gamma_\bullet(\Gamma_\bullet),\gamma_\bullet)$ of \cite{BD82}.
The empty map is denoted $0$.
Single BD maps always appear as the components $\gamma_r,\gamma_c$ of a BD pair (defined below), so we write them with an index $\bullet\in\{r,c\}$ and attach the same index to the objects built from them. An unindexed $\gamma$ always denotes a pair.
We identify $\Gamma$ with $[1,r]$ and write $\gamma_\bullet(i)=j$ for $\gamma_\bullet(\alpha_i)=\alpha_j$.
The assignment $E_i\mapsto E_k$, $F_i\mapsto F_k$ for $\gamma_\bullet(\alpha_i)=\alpha_k$ extends to an isomorphism $\mfg_{\Gamma_\bullet}\to\mfg_{\gamma_\bullet(\Gamma_\bullet)}$.
Here, for $K\subseteq\Gamma$, we write $\mfg_K$ for the semisimple subalgebra generated by the $E_i,F_i$, $\alpha_i\in K$.
It is the derived algebra of $\mathrm{Lie}(\mcG^K)$.
Precomposing with the orthogonal projection onto $\mfg_{\Gamma_\bullet}$ gives an endomorphism $\gamma_\bullet$ of $\mfg$, nilpotent on $\mfn_\pm$ and commuting with the transpose.
Set $C_{\gamma_\bullet}:=\frac12+\sum_{k\ge1}\gamma_\bullet^k$ on $\mfn_+\oplus\mfn_-$, and
\[
  \mfh_{\gamma_\bullet}:=\{\zeta\in\mfh\mid\alpha(\zeta)=\gamma_\bullet(\alpha)(\zeta)\ \forall\alpha\in\Gamma_\bullet\},
  \qquad
  \mcH_{\gamma_\bullet}:=\{y\in\mcH\mid y^\alpha=y^{\gamma_\bullet(\alpha)}\ \forall\alpha\in\Gamma_\bullet\}^{\circ},
\]
where $\circ$ denotes the identity component; thus $\mcH_{\gamma_\bullet}=\exp\mfh_{\gamma_\bullet}$ is a torus and $\dim\mfh_{\gamma_\bullet}=|\Gamma|-|\Gamma_\bullet|$.
We take the identity component because the subgroup defined by the character equations can be disconnected.
For example, let $\mcG=SL_3$ and $\gamma_\bullet=\{1\to2\}$.
For $y=\mathrm{diag}(t_1,t_2,t_3)$ the only equation is $t_1/t_2=t_2/t_3$, that is $t_1t_3=t_2^2$. Together with $t_1t_2t_3=1$, this gives $t_2^3=1$.
So the subgroup is $\{\mathrm{diag}(t,\omega,\omega^2t^{-1})\mid t\in\bbC^*,\ \omega^3=1\}$, which has three components, and $\mcH_{\gamma_\bullet}=\{\mathrm{diag}(t,1,t^{-1})\mid t\in\bbC^*\}$.
In terms of characters, $\alpha_1-\alpha_2=3(\varpi_1-\varpi_2)$ is not primitive in the weight lattice.
  Equivalently, the subgroup contains the center $\{\omega I\mid\omega^3=1\}$ of $SL_3$, on which every root is trivial, and it is the direct product of $\mcH_{\gamma_\bullet}$ and the center.

  Let $\mcS(\gamma_\bullet)\subseteq\mfh\wedge\mfh$ be the set of solutions $s_\bullet$ of
\begin{equation}
  \label{eq: system s}
  2\big((\alpha-\gamma_\bullet(\alpha))\otimes1\big)s_\bullet=\big((\alpha+\gamma_\bullet(\alpha))\otimes1\big)t_0
  \qquad\forall\alpha\in\Gamma_\bullet;
\end{equation}
by \cite{BD82} it is a non-empty affine space over $\mfh_{\gamma_\bullet}\wedge\mfh_{\gamma_\bullet}$.
For $s_\bullet\in\mcS(\gamma_\bullet)$,
\[
  r_{\gamma_\bullet,s_\bullet}:=\tfrac12t+s_\bullet+(1\otimes C_{\gamma_\bullet})t_{-+}-(C_{\gamma_\bullet}\otimes1)t_{+-}
\]
is a classical $R$-matrix \cite{BD82}.

For $\xi\in\mfg$ and a function $f$ on $\mcG$ let
\[
  \partial^\rr_\xi f(U):=\tfrac{d}{dt}f(\exp(t\xi)U)|_{t=0},
  \qquad
  \partial^\cc_\xi f(U):=\tfrac{d}{dt}f(U\exp(t\xi))|_{t=0},
\]
and for $z=\sum_m\xi_m\otimes\eta_m\in\mfg\otimes\mfg$ let $z(f,g)^\rr:=\sum_m\partial^\rr_{\xi_m}f\,\partial^\rr_{\eta_m}g$, and similarly $z(f,g)^\cc:=\sum_m\partial^\cc_{\xi_m}f\,\partial^\cc_{\eta_m}g$.

A \newword{BD pair} is a pair $\gamma=(\gamma_r,\gamma_c)$ of BD maps $\gamma_r:\Gamma_r\to\Gamma$, $\gamma_c:\Gamma_c\to\Gamma$.
We set $\mcS(\gamma):=\mcS(\gamma_r)\times\mcS(\gamma_c)$, $r_\rr:=r_{\gamma_r,s_r}$, $r_\cc:=\big((T\otimes T)r_{\gamma_c,s_c}\big)^{21}$ for $s=(s_r,s_c)$, where $T$ is the transpose, and
\begin{equation}
  \label{eq: bracket}
  \{f,g\}_{\gamma,s}:=r_\cc(f,g)^\cc-r_\rr(f,g)^\rr.
\end{equation}
In the notation of \cite{GSV19,GSV23}, $\gamma_r=\gamma^\rr$ and $\gamma_c=(\gamma^\cc)^{-1}$, and \eqref{eq: bracket} is the bracket of \cite[Section~2.2]{GSV23} for the pair $(r_\rr,r_\cc)$, which is Poisson \cite[Section~2.2]{GSV23}, \cite[Section~2.2]{GSV19}.
We set $\mfh_\gamma:=\mfh_{\gamma_r}\oplus\mfh_{\gamma_c}$ and $\mcH_\gamma:=\mcH_{\gamma_r}\times\mcH_{\gamma_c}$, acting on $\mcG$ by $(y_r,y_c)\cdot U=y_rUy_c$.

\subsection{The map \texorpdfstring{$h_\gamma$}{h}}
\label{sec: h}
Let $\gamma$ be a BD pair for $\mcG$, and put $\sigma_r=w_0^{\Gamma_r}w_0$ and $\sigma_c=w_0^{\Gamma_c}w_0$.
A generic $U\in\mcG$ admits factorizations
\begin{equation*}
  U=V_rM_r,\quad V_r\in\mcN_-^{\Gamma_r},\ M_r\in\mcB_+\overline{\sigma_r}\mcB_+,
  \qquad
  U=M_cV_c,\quad V_c\in\mcN_+^{\Gamma_c},\ M_c\in\mcB_-\overline{\sigma_c}^{\,-1}\mcB_-,
\end{equation*}
which determine $V_r$ and $V_c$.
Write $\lfloor A\rfloor_+$ (resp.\ $\lceil A\rceil_-$) for the factor in $\mcN_+$ (resp.\ $\mcN_-$) of $A\in\mcN_+\mcB_-$ (resp.\ $A\in\mcB_+\mcN_-$), and put
\begin{equation*}
  \overline{V}_r=\lfloor V_r\overline{w_0^{\Gamma_r}}\rfloor_+\in\mcN_+^{\Gamma_r},
  \qquad
  \overline{V}_c=\lceil \overline{w_0^{\Gamma_c}}V_c\rceil_-\in\mcN_-^{\Gamma_c}.
\end{equation*}
Let $\boldsymbol\gamma_r:\mcN_+\to\mcN_+$ and $\boldsymbol\gamma_c:\mcN_-\to\mcN_-$ be the homomorphisms integrating the restrictions of the endomorphisms $\gamma_r$ to $\mfn_+$ and $\gamma_c$ to $\mfn_-$; their images lie in $\mcN_+^{\gamma_r(\Gamma_r)}$ and $\mcN_-^{\gamma_c(\Gamma_c)}$.
The map $h_\gamma:\mcG\dashrightarrow\mcG$ is
\begin{equation}
  \label{eq: h}
  h_\gamma(U)=H_r(U)\,U\,H_c(U),
  \qquad
  H_r(U)=\cdots\boldsymbol\gamma_r^2(\overline{V}_r)\,\boldsymbol\gamma_r(\overline{V}_r),
  \qquad
  H_c(U)=\boldsymbol\gamma_c(\overline{V}_c)\,\boldsymbol\gamma_c^2(\overline{V}_c)\cdots,
\end{equation}
where the products are finite because $\gamma_r$ and $\gamma_c$ are nilpotent.
This is the map of \cite[Section~3.1]{GSV23}, defined there for every simple $\mcG$, with the exception that the factors of $H_c$ appear there in the opposite order (see \cref{rmk: Hc order}).
For $\Sp$ we study the map $h_\gamma$ in \cref{prop: h theta}.

\begin{lemma}
  \label{lem: h transpose}
  For every BD pair $\gamma$ of $\mcG$, $h_{(\gamma_r,\gamma_c)}(U)=h_{(\gamma_c,\gamma_r)}(U^T)^T$.
\end{lemma}
\begin{proof}
  The anti-involution $T$ fixes $\mcH$, swaps $\mcN_+^K$ and $\mcN_-^K$, and intertwines the homomorphisms integrating $\gamma_r$ and $\gamma_c$.
  Also, $\overline{w}^{\,T}=\overline{w}^{\,-1}\in\mcH\overline{w}$ for an involution $w$.
  Hence replacing $\overline{w}$ by $\overline{w}^{\,-1}$, for $w=w_0^{\Gamma_r},w_0^{\Gamma_c}$, only changes the Borel factors in the definitions of $\overline{V}_r,\overline{V}_c$.
  Transposing the factorizations of $U^T$ for $(\gamma_c,\gamma_r)$ therefore gives those of $U$ for $(\gamma_r,\gamma_c)$.
  Thus $H_c(U)$, resp.\ $H_r(U)$, is the transpose of $H_r(U^T)$, resp.\ $H_c(U^T)$, computed for $(\gamma_c,\gamma_r)$.
\end{proof}

\begin{remark}
  \label{rmk: Hc order}
  In \cite[Section~3.1]{GSV23} the factor $H_c$ is written as $\cdots\boldsymbol\gamma_c^2(\overline{V}_c)\boldsymbol\gamma_c(\overline{V}_c)$.
  This is a misprint: with that order $h_\gamma$ is not Poisson already for $SL_4$, $\gamma_r=0$ and $\gamma_c=\{1\to2\to3\}$.
  The order in \eqref{eq: h} is the one compatible with \cref{lem: h transpose} and required by the derivation of \cite[(4.2)]{GSV23} from \cite[Proposition~4.6]{GSV23}, and the arguments of \cite{GSV23} apply to it.
  Indeed, that derivation uses $N_-=(\overline{V}_cH_c)^{-1}$, and so needs $H_c=\boldsymbol\gamma_c(\overline{V}_cH_c)$, which holds for the order in \eqref{eq: h} but not for the reversed one; similarly $H_r=\boldsymbol\gamma_r(H_r\overline{V}_r)$.
\end{remark}

\begin{definition}[{\cite[Section~4.1]{GSV23}}]
  \label{def: aperiodic}
  Take disjoint copies $\{a^\rr\mid a\in\Gamma_r\}$ and $\{b^\cc\mid b\in\Gamma_c\}$ of $\Gamma_r$ and $\Gamma_c$.
  The \newword{alternating map} of a BD pair $\gamma$ is the partial injective map on their union sending $a^\rr$ to $(\gamma_r(a)^*)^\cc$ if $\gamma_r(a)^*\in\Gamma_c$, and $b^\cc$ to $(\gamma_c(b)^*)^\rr$ if $\gamma_c(b)^*\in\Gamma_r$.
  The pair $\gamma$ is \newword{aperiodic} if the alternating map has no periodic points.
\end{definition}

For $SL_N$ the periodic points of the alternating map correspond to the alternating cycles of the BD graph of \cite[Section~4.1]{GSV23}, so \cref{def: aperiodic} agrees with the definition there.

\subsection{The cluster structures on \texorpdfstring{$SL_N$}{SLN}}
\label{sec: GSV}
\begin{theorem}[{\cite[Section~1, Theorem~4.4 and (4.6)]{GSV23}}]
  \label{thm: GSV h}
  If $\gamma$ is an aperiodic BD pair of type $A_{N-1}$, the map $h_\gamma$ is birational.
\end{theorem}

Birationality of $h_\gamma$ is proved in \cite{GSV23} only in type $A$. For $\Sp$ we obtain it in \cref{thm: fold model}(c).

\begin{theorem}[{\cite[Theorem~3.3]{GSV19}, \cite[Theorems~4.3, 4.11, 4.14, 4.17]{GSV23}}]
  \label{thm: GSV}
  Let $\gamma$ be an aperiodic BD pair of type $A_{N-1}$.
  The seed of \cite[Theorem~4.11]{GSV23} defines a cluster structure $\mcC_\gamma$ on $SL_N$ with the properties (i)--(iv) of the introduction:
  \begin{enumerate}[(i)]
    \item the number of frozen variables is $2(N-1)-|\Gamma_r|-|\Gamma_c|$ and the exchange matrices have full rank;
    \item $\mcC_\gamma$ is regular and $\oA=\bbC[SL_N]$;
    \item the global toric action on $\mcC_\gamma$ is generated by the action of $\mcH_\gamma$;
    \item $\mcC_\gamma$ is compatible with $\{\cdot,\cdot\}_{\gamma,s}$ for every $s\in\mcS(\gamma)$; in every extended cluster its coefficient matrix satisfies \eqref{eq: compatibility} with all $\kappa_p\neq0$.
  \end{enumerate}
\end{theorem}

In (ii), \cite[Theorem~4.14]{GSV23} gives regularity of the adjacent cluster variables, and \cite[Theorem~4.17]{GSV23} gives Laurent expressions of the matrix entries in the initial and adjacent clusters; the latter is proved by induction on $|\Gamma_r|+|\Gamma_c|$, the case of size $1$ being \cite[Section~7.3]{GSV19}. Regularity and completeness follow as in \cite[Sections~4.4--4.5]{GSV23}.
  For $\gamma=(0,0)$, completeness of the standard cluster structure on $SL_N$ is \cite{Oya25} (and \cite[Theorem~B.1]{QY25} in all types).
In (iv), \eqref{eq: compatibility} is checked with all $\kappa_p$ equal to some $\kappa\neq0$ at the initial seed \cite[proof of Theorem~4.11]{GSV23}.
  The coefficient matrices in adjacent clusters are related as the matrices $\Lambda$ of \cite[(3.4)]{BZ05}, and the proof of \cite[Proposition~3.4]{BZ05} shows that $\tB^T\Omega$ does not change, so the same $\kappa_p$ work in every extended cluster.
Part (iii) is proved in \cite[Theorem~3.3]{GSV19} for the class of oriented pairs considered there, and is not stated in \cite{GSV23}. We prove it for arbitrary aperiodic pairs after \cref{prop: base case concise}, using only (i), (ii), and the initial cluster described there.

\section{The BD word seeds}
\label{sec: model}
In this section we define the BD word seeds for every $\mcG$ and prove that moves of words act on them by mutations.

\subsection{Words, slots and chambers}
\label{sec: words}
We describe the seeds of \cite[Section~2.2]{BFZ05} in a form in which the two ends of a word play the same role. We need this because the BD word seeds are obtained by adding letters at both ends of a word.
The translation to the labels of \cite{BFZ05} is given in \cref{prop: BFZ}.
Readers can follow along using \cref{fig: Sp6 word seed}, ignoring for the moment all the parts in color.

A \newword{word} is a family $\bfi=(i_j)_{j\in J}$ of \newword{letters} $i_j\in\pm[1,r]$, indexed by a finite non-empty interval $J$ of integers, the \newword{positions}.
  The \newword{level} of $i_j$ is $|i_j|$, and $\varepsilon(i_j)$ is its sign.
The \newword{slots} of $\bfi$ are the integers $s$ such that $s$ or $s+1$ is a position.
  The letter $i_j$ lies between the slots $j-1$ and $j$, and the first and the last slot are the \newword{end slots}.
For each $x\in[1,r]$, the letters at level $x$ cut the slots into intervals, the \newword{chambers} of $\bfi$ at level $x$.
  We write $c_x(s)$ for the one at level $x$ containing the slot $s$, and $C(\bfi)$ for the set of all chambers.
A chamber is \newword{$\bfi$-exchangeable} if it contains no end slot, and an \newword{end chamber} otherwise.
So if $\bfi$ has $n$ letters at level $x$, there are $n+1$ chambers at level $x$; the \newword{initial} one contains the first slot and the \newword{final} one the last slot (they are equal if $n=0$).
If $x=|i_j|$, the letter $i_j$ separates $c_x(j-1)$ from $c_x(j)$.

For distinct $x,y\in[1,r]$, let $M_{xy}$ be the set of positions of the letters at levels $x$ and $y$, and for $j\in M_{xy}$ put $\varphi_{xy}(j):=\varepsilon(i_j)$ if $|i_j|=y$ and $\varphi_{xy}(j):=-\varepsilon(i_j)$ if $|i_j|=x$.
Let $\tB(\bfi)=(b_{k\ell})$ be the matrix with rows labeled by $C(\bfi)$ and columns by the $\bfi$-exchangeable chambers, where
\begin{equation}
  \label{eq: B entries}
  b_{k\ell}=
  \begin{cases}
    \varepsilon(i_j) & \text{if } (k,\ell)=\big(c_x(j-1),c_x(j)\big) \text{ with } x=|i_j|,\\
    -\varepsilon(i_j) & \text{if } (k,\ell)=\big(c_x(j),c_x(j-1)\big) \text{ with } x=|i_j|,\\
    -\frac12a_{xy}\big(\varphi_{xy}(q)-\varphi_{xy}(p)\big) & \text{if } (k,\ell)=\big(c_x(p),c_y(p)\big) \text{ with } p<q \text{ consecutive in } M_{xy},\\
    0 & \text{otherwise.}
  \end{cases}
\end{equation}
The cases are exclusive. The first two concern chambers at equal levels and the third at distinct levels. In the third case, a letter at level $x$ or $y$ lying between two pairs $p<q$ separates their chambers at that level.
So the entries of $\tB(\bfi)$ lie in $\{0,\pm1\}$ at equal levels and in $\{0,\pm a_{xy}\}$ at levels $x\neq y$.
As $\varphi_{yx}=-\varphi_{xy}$, if $d_xa_{xy}=d_ya_{yx}$ with $d_x>0$, then $d_xb_{k\ell}=-d_yb_{\ell k}$ for exchangeable chambers $k,\ell$ at levels $x,y$.
  So the principal part of $\tB(\bfi)$ is skew-symmetrizable, and skew-symmetric if the Cartan matrix is symmetric.
Only the order of the letters in a word matters: translating all positions by the same integer changes neither the chambers, nor their exchangeability, nor $\tB(\bfi)$. We use this freely.

\begin{proposition}
  \label{prop: BFZ}
  Let $J=[1,L]$, and label the chamber $c_{|i_k|}(k)$, $k\in[1,L]$, by $k$, and the initial chamber $c_x(0)$ by $-x$.
  Then $\tB(\bfi)$ is the matrix of \cite[Definition~2.3]{BFZ05}.
\end{proposition}
\begin{proof}
  With these labels, the chamber $k$ is $\bfi$-exchangeable if and only if $k$ is $\bfi$-exchangeable in the sense of \cite[Section~2.2]{BFZ05}.
  For every label $k$ the next label at the same level is $k^+$, the smallest index $j>k$ with $|i_j|=|i_k|$, or $L+1$ if there is none.
  Here $i_{-x}:=-x$ as in \cite{BFZ05}, so that $|i_k|$ is the level of the chamber $k$ for every label $k$.
  By \cite[Remark~2.4]{BFZ05}, for $\ell$ exchangeable, $p=\max(k,\ell)$ and $q=\min(k^+,\ell^+)$, the entry of the matrix of \cite{BFZ05} is
  \begin{equation}
    \label{eq: BFZ entries}
    b_{k\ell}=
    \begin{cases}
      -\operatorname{sgn}(k-\ell)\,\varepsilon(i_p) & \text{if } p=q,\\
      -\operatorname{sgn}(k-\ell)\,\varepsilon(i_p)\,a_{|i_k|,|i_\ell|} & \text{if } p<q \text{ and } \varepsilon(i_p)\varepsilon(i_q)(k-\ell)(k^+-\ell^+)>0,\\
      0 & \text{otherwise,}
    \end{cases}
  \end{equation}
  where $p$ and $q$ lie in $[1,L]$, so that $\varepsilon(i_p)$ and $\varepsilon(i_q)$ are signs of letters.
  If $p=q$, then $k$ and $\ell$ are at the same level, $p$ is the letter separating them, and this is one of the first two cases of \eqref{eq: B entries}.
  Otherwise, if $k$ and $\ell$ are at the same level, or at distinct levels $x,y$ with $a_{xy}=0$, both formulas give $0$.
  Let $k$ and $\ell$ be at distinct levels $x,y$ with $a_{xy}\neq0$.
  If $p>q$, the chambers have no common slot, and both formulas give $0$.
  If $p<q$, then $p$ and $q$ are consecutive positions of letters at the levels $x$ and $y$.
  The slot $p$ lies in both chambers, so $k=c_x(p)$ and $\ell=c_y(p)$.
  Conversely, every such pair $p<q$ arises in this way.
  We check that the two formulas agree in each of the four cases for $p\in\{k,\ell\}$ and $q\in\{k^+,\ell^+\}$.
  For instance, if $p=k$ and $q=\ell^+$, then $(k-\ell)(k^+-\ell^+)>0$, so $b_{k\ell}=-a_{xy}\varepsilon(i_p)$ if $\varepsilon(i_p)\varepsilon(i_q)>0$ and $0$ otherwise, while $\varphi_{xy}(p)=-\varepsilon(i_p)$ and $\varphi_{xy}(q)=\varepsilon(i_q)$.
  If $p=k$ and $q=k^+$, then $b_{k\ell}=-a_{xy}\varepsilon(i_p)$ if $\varepsilon(i_p)\varepsilon(i_q)<0$ and $0$ otherwise, while $\varphi_{xy}(p)=-\varepsilon(i_p)$ and $\varphi_{xy}(q)=-\varepsilon(i_q)$.
  The cases $p=\ell$ follow in the same way with the signs of $\varphi_{xy}$ reversed.
\end{proof}

A \newword{commutation} swaps two consecutive letters $i_j,i_{j+1}$ at different levels which have opposite signs or satisfy $a_{|i_j|,|i_{j+1}|}=0$.
A \newword{sign swap} replaces two consecutive letters $i,-i$ by $-i,i$, or conversely.
A \newword{braid move} replaces $m$ consecutive letters $i,j,i,\dots$ by $j,i,j,\dots$, where $i$ and $j$ have the same sign, $a_{|i|,|j|}\neq0$.
The number $m$, the \newword{length} of the move, is the order of $s_{|i|}s_{|j|}$ in $W$, i.e.\ $m=3,4,6$ for $a_{|i|,|j|}a_{|j|,|i|}=1,2,3$.
These are the \newword{moves}.

A move replaces the letters at the positions in an interval $[p,p']$ by letters with the same set of levels.
If $\bfi'$ is obtained from $\bfi$ by a move, a chamber $c_x(s)$ of $\bfi$ with $s\notin[p,p'-1]$ \newword{corresponds} to the chamber $c_x(s)$ of $\bfi'$.
The other chambers, of $\bfi$ and of $\bfi'$, are \newword{replaced} by the move.
A commutation replaces no chamber. A sign swap, resp.\ a braid move of length $3$, replaces one chamber of $\bfi$ and one of $\bfi'$, with the slot $p$, resp.\ the slots $p,p+1$, and we let these correspond to each other.

\begin{lemma}
  \label{lem: commutation}
  Let $\bfi'$ be obtained from $\bfi$ by a commutation.
  Then the correspondence of chambers is a bijection $C(\bfi)\to C(\bfi')$ which preserves levels and exchangeability, and $\tB(\bfi')$ is obtained from $\tB(\bfi)$ by relabeling rows and columns along it.
\end{lemma}
\begin{proof}
  Let the commutation swap the letters at positions $j,j+1$, with levels $x\neq y$.
  The correspondence is a bijection since no chamber is replaced, and it preserves levels and end chambers, hence exchangeability.

  By \eqref{eq: B entries}, the entries between chambers at given levels depend only on the signed sequence of letters at those levels.
  The commutation leaves this sequence unchanged at each single level and at every pair of levels other than $\{x,y\}$.
  Thus it remains to check the entries between levels $x$ and $y$.

  If $a_{xy}=0$, these entries vanish.
  Otherwise the swapped letters have opposite signs, so $\varphi_{xy}(j)=\varphi_{xy}(j+1)$, and the commutation leaves both $M_{xy}$ and $\varphi_{xy}$ unchanged.
  Every consecutive pair $p<q$ in $M_{xy}$ with $p\neq j$ therefore gives the same entry at corresponding chambers, since the slot $p$ lies outside the move.
  The remaining pair $j<j+1$ contributes zero in both words.
  Hence all entries agree along the correspondence.
\end{proof}

\subsection{Double reduced words}
\label{sec: double words}
A double reduced word for $(u,v)\in W\times W$ is a shuffle of a reduced word for $u$ in the alphabet $-[1,r]$ and one for $v$ in the alphabet $[1,r]$ \cite[Section~2.2]{BFZ05}.
We only use $(u,v)=(w_0,w_0)$, and call a double reduced word $\bfi=(i_1,\dots,i_L)$ for $(w_0,w_0)$ simply a \newword{double reduced word}.
  So $J=[1,L]$, the slots are $[0,L]$, $L=2\ell(w_0)$, and every level carries letters of both signs.
Following \cite[(2.9)--(2.11)]{BFZ05}, for a slot $s$ let $u_{\le s}$ be the product of the $s_{|i_j|}$ over the negative letters $i_j$ with $j\le s$, in increasing order of $j$.
Let $v_{>s}$ be the product of the $s_{|i_j|}$ over the positive letters $i_j$ with $j>s$, in decreasing order of $j$.
The \newword{row weight} and the \newword{column weight} of the chamber $k=c_x(s)$ are $\lambda^\rr_k:=u_{\le s}\varpi_x$ and $\lambda^\cc_k:=v_{>s}\varpi_x$, which do not depend on $s\in k$, as the letters between two slots of $k$ are at levels $y\neq x$ and $s_y\varpi_x=\varpi_x$.
  Its \newword{minor} is $\Delta(k;\bfi):=\Delta_{\lambda^\rr_k,\lambda^\cc_k}$.
By \cite[Theorem~2.10]{BFZ05} and \cref{prop: BFZ}, $\Sigma(\bfi):=\big((\Delta(k;\bfi))_{k\in C(\bfi)},\tB(\bfi)\big)$ is a seed in $\bbC(\mcG)$; in particular the $\Delta(k;\bfi)$ are algebraically independent and generate $\bbC(\mcG)$.

The \newword{root} of a negative letter $i_j$ is $u_{\le j-1}(\alpha_{|i_j|})$, and that of a positive letter $i_j$ is $v_{>j}(\alpha_{|i_j|})$.
The negative letters, in increasing order of position, form a reduced word for $w_0$, and so do the positive letters in decreasing order of position.
  So by \cref{sec: Lie} the roots of the negative letters are the positive roots, each occurring once, and similarly for the positive letters.
In particular, every simple root is the root of exactly one negative and exactly one positive letter.

Moves transform double reduced words into double reduced words, and any two double reduced words are related by moves, by \cite[Chapter~IV, Section~1.5, Proposition~5]{Bou02} applied to $W\times W$ and to the monoid presented by the braid relations.

\begin{theorem}[{\cite[Theorem~3.5]{SSVZ00}, \cite[Propositions~3.7 and~3.25]{SW21}}]
  \label{thm: SSVZ}
  Let $\bfi'$ be obtained from $\bfi$ by a move.
  Along the correspondence of chambers:
  \begin{enumerate}
    \item For a commutation, $\Sigma(\bfi')=\Sigma(\bfi)$.
    \item For a sign swap or a braid move of length $3$, $\Sigma(\bfi')=\mu_k(\Sigma(\bfi))$, where $k$ is the replaced chamber.
    \item For a braid move of length $4$, $\Sigma(\bfi')=\mu_k\mu_\ell\mu_k(\Sigma(\bfi))$, where $k$ and $\ell$ are the replaced chambers at the levels of the long and of the short root (equivalently $\mu_\ell\mu_k\mu_\ell(\Sigma(\bfi))$); the replaced chambers of $\bfi'$ correspond to them by level.
    \item For a braid move of length $6$ at the positions $[p,p+5]$, with $\alpha_x$ short for $x=|i_p|$, and $y=|i_{p+1}|$, $\Sigma(\bfi')$ is obtained from $\Sigma(\bfi)$ by the mutations at
    \[\ell_2,\ell_1,k_2,k_1,\ell_2,k_2,\ell_2,\ell_1,k_1,\ell_2,\]
    in this order, where $k_1=c_x(p)$, $k_2=c_x(p+2)$, $\ell_1=c_y(p+1)$, $\ell_2=c_y(p+3)$, corresponding to $c_x(p+1)$, $c_x(p+3)$, $c_y(p)$, $c_y(p+2)$ in $\bfi'$.
    If $\alpha_{|i_p|}$ is long, the inverse move is of this form, and $\Sigma(\bfi')$ is obtained from $\Sigma(\bfi)$ by the same mutations in the reverse order.
  \end{enumerate}
  The statement on the matrices holds for arbitrary words.
\end{theorem}
For simply-laced $\mcG$, where the moves have length at most $3$, the statement on the matrices is \cite[Theorem~3.5]{SSVZ00}, see \cite[Remark~2.14]{BFZ05}, where the non-simply-laced case is left open.
In general the statement follows from \cite[Proposition~3.7]{SW21}.
The strings of the string diagram of $\bfi$ are our chambers, the closed ones being the $\bfi$-exchangeable chambers.
The Cartan matrix is ours \cite[Appendix~A.1]{SW21}, and the exchange matrix of \cite[Section~3.1]{SW21} is $\epsilon_{\ell k}=b_{k\ell}$. For a braid move of length $6$, the strings $a,b,c,d$ of \cite[Proposition~3.7(2)]{SW21} are $k_1,k_2,\ell_1,\ell_2$. In \cite[Proposition~3.7]{SW21} the braid move of length $4$ mutates first at the short root, which gives the same seed, as the alternating mutations at two chambers $k,\ell$ with $b_{k\ell}b_{\ell k}=-2$ have period $6$.
The statement on the functions is \cite[Proposition~3.25]{SW21}, through the realization of the decorated double Bott--Samelson cell \cite[Section~3.3]{SW21} sending $U\in\mcG^{w_0,w_0}$ to the chains of decorated flags $U^T\overline{u_{\le s}}\,\mcN_+$ and $\mcN_-\overline{v_{>s}}^{\,-1}$, $s$ a slot, after deleting repetitions.
  This is not the map of \cite[Section~1.4]{SW21}.
These chains represent the same configuration under the canonical identifications for different reduced words, by uniqueness of compatible refinements \cite[Lemmas~2.10--2.11]{SW21}.
The coordinate \cite[Definition~3.20]{SW21} of $k=c_x(s)$ is $\Delta_{\varpi_x,\varpi_x}\big(\overline{v_{>s}}^{\,-1}U^T\overline{u_{\le s}}\big)=\Delta(k;\bfi)(U)$, by \cref{sec: Lie}.

\begin{remark}
  As any two double reduced words are related by moves, \cref{thm: SSVZ} shows that the seeds $\Sigma(\bfi)$ of all double reduced words are related by sequences of mutations, up to the correspondence of chambers. The quantum version of this claim is \cite[Theorem~A]{QY25}.
\end{remark}

\subsection{The BD word seeds}
\label{sec: model seeds}
The colored parts of \cref{fig: Sp6 word seed} illustrate the following definition.
Fix an aperiodic BD pair $\gamma$ of $\mcG$ and a double reduced word $\bfi$, i.e.\ a reduced word for $(w_0,w_0)$, and call the elements of $\Gamma_r$ and of $\Gamma_c$ \newword{row roots} and \newword{column roots}.

\begin{definition}
  \label{def: model}
  \begin{enumerate}[(a)]
    \item For a row root $a$, its \newword{inversion letter} is the negative letter with root $\alpha_{\gamma_r(a)}$, and $m^\rr_a$ denotes its position; its \newword{thawed chamber} is $t^\rr_a:=c_{a^*}(L)$.
    For a column root $b$, its inversion letter is the positive letter with root $\alpha_{\gamma_c(b)}$, and $m^\cc_b$ denotes its position; its thawed chamber is $t^\cc_b:=c_{b^*}(0)$.
    \item Order the row roots as $a_1,\dots,a_p$ and the column roots as $b_1,\dots,b_q$ so that $m^\rr_{a_1}>\dots>m^\rr_{a_p}$ and $m^\cc_{b_1}>\dots>m^\cc_{b_q}$.
    The \newword{extended word} $\bfi^\gamma$ is obtained from $\bfi$ by adding the \newword{virtual letters} $b_1^*,\dots,b_q^*$ at the positions $1-q,\dots,0$ and $-a_1^*,\dots,-a_p^*$ at the positions $L+1,\dots,L+p$:
    \[
      \bfi^\gamma=(b_1^*,\dots,b_q^*,\,i_1,\dots,i_L,\,-a_1^*,\dots,-a_p^*).
    \]
    For $s\in[0,L]$ we identify the chamber $c_x(s)$ of $\bfi$ with the chamber $c_x(s)$ of $\bfi^\gamma$.
    \item Let $P:=C(\bfi)$, let $P_\ex$ consist of the $\bfi$-exchangeable chambers and of the thawed chambers, and $P_\fr:=P\setminus P_\ex$.
    Let $\tB(\bfi^\gamma)_P$ be the submatrix of $\tB(\bfi^\gamma)$ with rows $P$ and columns $P_\ex$.
    For a row or column root $a$, with inversion letter $i_m$ at level $x$ and thawed chamber $t$, let $D_a$ be the matrix with rows $P$ and columns $P_\ex$ whose only nonzero entries are
    \[
      d_{c_x(m-1),t}=-d_{t,c_x(m-1)}=-\varepsilon(i_m),
      \qquad
      d_{c_x(m),t}=-d_{t,c_x(m)}=\varepsilon(i_m),
    \]
    an entry being omitted if its column is not in $P_\ex$.
    The \newword{dipole matrix} is $D_\gamma(\bfi):=\sum_aD_a$, and $\tB_\gamma(\bfi):=\tB(\bfi^\gamma)_P+D_\gamma(\bfi)$.
    \item For $k\in P$, a row root $a$ and a column root $b$ let
    \[
      e^\rr_a(k):=\big[\lambda^\rr_k(\alpha^\vee_{\gamma_r(a)})\big]_+,
      \qquad
      e^\cc_b(k):=\big[\lambda^\cc_k(\alpha^\vee_{\gamma_c(b)})\big]_+,
    \]
    and define recursively
    \begin{equation}
      \label{eq: Phi}
      \Phi_\gamma(k;\bfi):=\Delta(k;\bfi)\prod_{a\in\Gamma_r}\Phi_\gamma(t^\rr_a;\bfi)^{e^\rr_a(k)}\prod_{b\in\Gamma_c}\Phi_\gamma(t^\cc_b;\bfi)^{e^\cc_b(k)}.
    \end{equation}
    \item If $h_\gamma$ is birational, the \newword{BD word seed} is $\Sigma_\gamma(\bfi):=\big((x_\gamma(k))_{k\in P},\tB_\gamma(\bfi)\big)$, $x_\gamma(k):=\Phi_\gamma(k;\bfi)\circ h_\gamma^{-1}$.
  \end{enumerate}
\end{definition}

\begin{lemma}
  \label{lem: tails}
  Let $x\in[1,r]$, $a\in\Gamma_r$ and $b\in\Gamma_c$.
  The weights of the final chamber $c_x(L)$ are $(-\varpi_{x^*},\varpi_x)$ and those of the initial chamber $c_x(0)$ are $(\varpi_x,-\varpi_{x^*})$, and
  \[
    e^\rr_a(c_x(L))=e^\cc_b(c_x(0))=0,
    \qquad
    e^\cc_b(c_x(L))=\delta_{\gamma_c(b),x},
    \qquad
    e^\rr_a(c_x(0))=\delta_{\gamma_r(a),x}.
  \]
  So in \eqref{eq: Phi} the function $\Phi_\gamma(c_x(L);\bfi)$ involves exactly the $\Phi_\gamma(t^\cc_b;\bfi)$ with $\gamma_c(b)=x$, and $\Phi_\gamma(c_x(0);\bfi)$ exactly the $\Phi_\gamma(t^\rr_a;\bfi)$ with $\gamma_r(a)=x$, each with exponent $1$.
  As $\gamma$ is aperiodic, the recursion \eqref{eq: Phi} terminates.
  Hence the functions $\Phi_\gamma(k;\bfi)$ of the end chambers $k$ are monomials in the minors of end chambers, and do not depend on $\bfi$.
\end{lemma}
\begin{proof}

  The weights of $c_x(L)$ are computed at the slot $L$, with $u_{\le L}=w_0$ and $v_{>L}=e$, and are $w_0\varpi_x=-\varpi_{x^*}$ and $\varpi_x$.
  Those of $c_x(0)$ are computed at the slot $0$, with $u_{\le0}=e$ and $v_{>0}=w_0$, and are $\varpi_x$ and $-\varpi_{x^*}$.
  Fundamental weights pair non-negatively with simple coroots, and $\varpi_x(\alpha^\vee_y)=\delta_{xy}$. This gives the four formulas.
  As $t^\rr_a=c_{a^*}(L)$, $\Phi_\gamma(t^\rr_a;\bfi)$ involves $\Phi_\gamma(t^\cc_b;\bfi)$ if and only if $\gamma_c(b)=a^*$, i.e.\ the alternating map, which sends $b^\cc$ to $(\gamma_c(b)^*)^\rr$, sends $b^\cc$ to $a^\rr$.
  Similarly $\Phi_\gamma(t^\cc_b;\bfi)$ involves $\Phi_\gamma(t^\rr_a;\bfi)$ if and only if it sends $a^\rr$ to $b^\cc$.
  Thus the recursion from a thawed chamber follows the alternating map backwards and terminates since this map is injective and has no periodic points.
  As the functions of end chambers only involve those of thawed chambers, which are end chambers, they are monomials in minors of end chambers.
  These minors do not depend on $\bfi$, since the weights of end chambers do not.
\end{proof}

\cref{def: model} makes sense: the thawed chambers are distinct, as those of row roots are final and those of column roots initial, and every level carries at least two letters.
They are $\bfi^\gamma$-exchangeable, as the virtual letter at their level separates them from the end slot of $\bfi^\gamma$.
  So $P_\ex$ is a set of columns of $\tB(\bfi^\gamma)$, and $\tB(\bfi^\gamma)_P$ is defined.
The dipole entries are off the diagonal, i.e.\ $t^\rr_a$ is neither of the two chambers separated by the inversion letter of $a$.
Indeed, paired with $\alpha^\vee_{\gamma_r(a)}$, the row weights of these chambers give $1$ and $-1$.
In contrast, the row weight $-\varpi_a$ of $t^\rr_a$ gives $0$, as $\gamma_r(a)\neq a$ by nilpotency.
For a column root the column weights give $-1$, $1$ and $0$.
The exponents $e^\rr_a(k)$, $e^\cc_b(k)$ in \eqref{eq: Phi} are $0$ or $1$ for $SL_N$ (\cref{sec: type A concise}), but exponents equal to $2$ occur in type $C_n$ (\cref{ex: Sp6}).

By \cref{lem: commutation}, only the relative order of virtual letters at adjacent levels matters.
For row roots $a\neq a'$ with $a_{aa'}\neq0$, the virtual letters of $a$ and $a'$ are in the order opposite to that of $m^\rr_a$ and $m^\rr_{a'}$, and similarly for column roots.

We now check that the pair $\Sigma^\Phi_\gamma(\bfi):=\big((\Phi_\gamma(k;\bfi))_{k\in P},\tB_\gamma(\bfi)\big)$ is a seed in $\bbC(\mcG)$, for the two reasons given below. Then $\Sigma_\gamma(\bfi)$ is also a seed if $h_\gamma$ is birational, as it is obtained from $\Sigma^\Phi_\gamma(\bfi)$ by composing all functions with $h_\gamma^{-1}$.
First, the principal part of $D_\gamma(\bfi)$ is skew-symmetric.
Its entries join chambers at the level $x$ of an inversion letter $i_m$ and at the level $a^*$, with $d_x=d_{a^*}$.
Indeed, the root $\alpha_{\gamma_\bullet(a)}$ of $i_m$ is conjugate to $\alpha_x$ and $\gamma_\bullet$ and $*$ preserve root lengths; so the principal part of $\tB_\gamma(\bfi)$ is skew-symmetrizable with the symmetrizer of $\tB(\bfi^\gamma)$ (\cref{sec: words}).
Second, by \cref{lem: tails} the thawed chambers can be ordered so that each $\Phi_\gamma(t;\bfi)$ involves only earlier ones. Putting the other chambers after them, each $\Phi_\gamma(k;\bfi)$ is $\Delta(k;\bfi)$ times a monomial in earlier minors, so the $\Phi_\gamma(k;\bfi)$ are algebraically independent generators of $\bbC(\mcG)$, as the $\Delta(k;\bfi)$ are (\cref{sec: double words}).

The map $h_\gamma$ is birational for $SL_N$ by \cref{thm: GSV h} and for $\Sp$ by \cref{thm: fold model}(c).
It is also birational for $\gamma=(0,0)$ in every type.
In this case $h_\gamma$ is the identity and there are no thawed chambers, so $\Sigma_{(0,0)}(\bfi)$ is the seed $\Sigma(\bfi)$ of \cite[Theorem~2.10]{BFZ05}.

\begin{example}
  \label{ex: Sp6}
  In type $C_n$, $w_0=-1$, so $*$ is the identity, and the only long simple root is $\alpha_n$, so a BD map, being an isometry and nilpotent, only involves $\alpha_1,\dots,\alpha_{n-1}$.
  Thus $Sp_4$ only has $\gamma=(0,0)$, and $Sp_6$ is the smallest case with non-trivial BD pairs.
  Let $\mcG=Sp_6$, with $\alpha_3$ long, $\gamma=(\{1\to2\},0)$, and let $\bfi$ be the word of \cref{fig: Sp6 word seed},
  \[
    \bfi=(-3,-2,-3,-1,-2,-3,-1,-2,-1,\,1,2,1,3,2,1,3,2,3).
  \]
  The negative letter $i_7=-1$ has the root $\alpha_2=\alpha_{\gamma_r(1)}$, so it is the inversion letter.
  The thawed chamber is $t^\rr_1=c_1(18)$, with $\Phi_\gamma(t^\rr_1;\bfi)=\Delta_{-\varpi_1,\varpi_1}$, and $\bfi^\gamma$ is $\bfi$ followed by the virtual letter $-1$.
  The non-zero entries of the column $t^\rr_1$ of $\tB_\gamma(\bfi)$ are
  \[
    b_{c_1(6),t^\rr_1}=b_{c_1(14),t^\rr_1}=1,
    \qquad
    b_{c_1(7),t^\rr_1}=b_{c_2(14),t^\rr_1}=-1,
  \]
  those at $c_1(6)$ and $c_1(7)$ from the dipole and the others from the virtual letter.
  The other entries of $\tB_\gamma(\bfi)$ not in $\tB(\bfi)$ are the dipole entries $b_{t^\rr_1,c_1(6)}=-1$ and $b_{t^\rr_1,c_1(7)}=1$.

\begin{lemma}
  \label{lem: model local}
  \begin{enumerate}[(a)]
    \item For $k\in P$ and $\ell$ $\bfi$-exchangeable, the entries of $\tB(\bfi^\gamma)_P$ and of $\tB(\bfi)$ at $(k,\ell)$ coincide; the entries of $\tB(\bfi^\gamma)_P$ between chambers at distinct levels $x,y$ with $a_{xy}=0$ vanish; and if the Cartan matrix is symmetric, the principal part of $\tB_\gamma(\bfi)$ is skew-symmetric.
    \item Every non-zero entry of $D_\gamma(\bfi)$ involves a thawed chamber; so the submatrices of $\tB_\gamma(\bfi)$ and of $\tB(\bfi)$ with rows and columns the $\bfi$-exchangeable chambers coincide.
  \end{enumerate}
\end{lemma}
\begin{proof}
  (a) The entries in the column of $\ell$ come at equal levels from the two letters bounding $\ell$, and at distinct levels from pairs $p<q$ with $p\in\ell$.
  As $\ell$ contains no end slot of $\bfi$, these letters are letters of $\bfi$, and so is $q$, which does not exceed the letter bounding $\ell$ on the right.
  The other claims follow directly from \eqref{eq: B entries} and the antisymmetry of the dipole entries.
  Part (b) follows directly from \cref{def: model}(c).
\end{proof}

For a word $\bfi$ let $\bfi^T$ be the word obtained by reversing the order of the letters and changing their signs.
Its slots correspond to those of $\bfi$ in reverse order.
For a chamber $k$ of $\bfi$ let $k^T$ be the chamber of $\bfi^T$ at the same level formed by the slots corresponding to those of $k$.
For a double reduced word, indexed by $[1,L]$, $\bfi^T=(-i_L,\dots,-i_1)$: the letter $i_j$ becomes the letter $-i_j$ at the position $L+1-j$, and $c_x(s)^T=c_x(L-s)$.
For a BD pair $\gamma$ let $\gamma^T:=(\gamma_c,\gamma_r)$; by \cref{lem: mirror} below, $\bfi\mapsto\bfi^T$ and $\gamma\mapsto\gamma^T$ give the transpose on functions.

\begin{lemma}
  \label{lem: mirror}
  \begin{enumerate}[(a)]
    \item For every word $\bfi$, $k\mapsto k^T$ is a bijection $C(\bfi)\to C(\bfi^T)$ which preserves levels and exchangeability, and $\tB(\bfi^T)$ is obtained from $\tB(\bfi)$ by relabeling rows and columns along it.
    \item For a double reduced word $\bfi$, under these identifications the inversion letters, thawed chambers and exponents of the row roots of $\gamma$ in $\bfi$ become those of the column roots of $\gamma^T$ in $\bfi^T$, and conversely, and $(\bfi^\gamma)^T=(\bfi^T)^{\gamma^T}$ up to translation.
    The matrix $\tB_{\gamma^T}(\bfi^T)$ is obtained from $\tB_\gamma(\bfi)$ by relabeling rows and columns along $k\mapsto k^T$, and $\Phi_{\gamma^T}(k^T;\bfi^T)=\Phi_\gamma(k;\bfi)\circ T$ for every chamber $k$, $T$ being the transpose.
    \item If $h_\gamma$ is birational, so is $h_{\gamma^T}$, and $x_{\gamma^T}(k^T)=x_\gamma(k)\circ T$.
  \end{enumerate}
\end{lemma}
\begin{proof}
We first compare the exchange matrices under reversal, then follow the weights, inversion letters and recursive functions through the same correspondence.

  (a) Index $\bfi^T$ by $-J$, with the letter $-i_j$ at the position $-j$ and the slot $-1-s$ corresponding to $s$.
  The letter $i_j$, at the level $x$ and separating $c_x(j-1)$ from $c_x(j)$, becomes the letter $-i_j$ at the position $-j$, separating $c_x(j)^T$ from $c_x(j-1)^T$; so the entries at equal levels are preserved.
  Consecutive positions $p<q$ at the levels $x,y$ become $-q<-p$, with $\varphi_{xy}$ changing sign, and $c_x(p)^T=c_x(q-1)^T$ is the chamber of $\bfi^T$ containing the slot $-q$.
  So the entries at distinct levels are preserved.

  (b) The negative letters of $\bfi^T$ at positions $\le L-s$ are the negatives of the positive letters of $\bfi$ at positions $>s$, in the opposite order, so $u_{\le L-s}(\bfi^T)=v_{>s}(\bfi)$, and similarly $v_{>L-s}(\bfi^T)=u_{\le s}(\bfi)$.
  So the weights of $k^T$ are those of $k$ in the opposite order, $\Delta(k^T;\bfi^T)=\Delta(k;\bfi)\circ T$ by \cref{sec: Lie}, and $e^\rr_a(k)$ for $\gamma$ equals $e^\cc_a(k^T)$ for $\gamma^T$, and conversely.
  The inversion letters correspond, as $u_{\le j-1}(\bfi)=v_{>L+1-j}(\bfi^T)$, and so do the thawed chambers, as $k\mapsto k^T$ swaps initial and final chambers.
  The virtual letters of $\bfi^\gamma$ go to those of $(\bfi^T)^{\gamma^T}$, in the required order, which gives $(\bfi^\gamma)^T=(\bfi^T)^{\gamma^T}$ up to translation, compatibly with the identifications of chambers.
  Finally, for a position $m$ of $\bfi$ and $m'=L+1-m$, $c_x(m-1)^T=c_x(m')$ and $c_x(m)^T=c_x(m'-1)$, so $D_{\gamma^T}(\bfi^T)$ is $D_\gamma(\bfi)$ relabeled by $k\mapsto k^T$. With (a) applied to $\bfi^\gamma$, this gives the claim on $\tB$, and \eqref{eq: Phi} gives the claim on $\Phi$.
  (c) follows from (b) and \cref{lem: h transpose}.
\end{proof}

\subsection{Moves}
\label{sec: moves}
We prove that moves of words act on the seeds $\Sigma^\Phi_\gamma(\bfi)$ by mutations, by tracking the virtual letters, dipoles and monomial factors.
In particular, these seeds are mutation equivalent for all double reduced words.

\begin{theorem}
  \label{thm: moves}
  Let $\gamma$ be an aperiodic BD pair of $\mcG$, and let $\bfi'$ be obtained from a double reduced word $\bfi$ by a move.
  Along the correspondence of chambers:
  \begin{enumerate}[(a)]
    \item if the move is a commutation, then $\Sigma^\Phi_\gamma(\bfi')=\Sigma^\Phi_\gamma(\bfi)$;
    \item if it is a sign swap or a braid move of length $3$, then $\Sigma^\Phi_\gamma(\bfi')=\mu_k(\Sigma^\Phi_\gamma(\bfi))$, $k$ being the replaced chamber;
    \item if it is a braid move of length $4$, then $\Sigma^\Phi_\gamma(\bfi')=\mu_k\mu_\ell\mu_k(\Sigma^\Phi_\gamma(\bfi))$, where $k$ and $\ell$ are the replaced chambers at the levels of the long and of the short root, and the replaced chambers of $\bfi'$ correspond to them by level.
  \end{enumerate}
  So if $h_\gamma$ is birational, the same holds for $\Sigma_\gamma(\bfi)$ and $\Sigma_\gamma(\bfi')$.
\end{theorem}
The last claim holds because composing with $h_\gamma^{-1}$ commutes with mutations.

We first compare $\Sigma^\Phi_\gamma(\bfi)$ with $\Sigma(\bfi)$ for a single word (\cref{lem: exponents,cor: Fraser}). We then prove the statements on the matrices directly and use this comparison to obtain those on the functions.

For a double reduced word $\bfi$, with $\lambda^\rr_k,\lambda^\cc_k$ the weights of the chambers $k$ and $b_{k\ell}$ the entries of $\tB(\bfi)$,
\begin{equation}
  \label{eq: homogeneity}
  \sum_k\lambda^\rr_k b_{k\ell}=0,
  \qquad
  \sum_k\lambda^\cc_k b_{k\ell}=0
\end{equation}
for every exchangeable chamber $\ell$.
This is \cite[Lemma~4.22]{GSV10} and it expresses the bihomogeneity of the exchange relations of \cite{BFZ05} for the left and right actions of $\mcH$.
We also use the second identity for arbitrary words, with $u_{\le s}$, $v_{>s}$ and the weights defined by the same formulas.
Indeed, let $\ell$ be at the level $y$, bounded by positions $p<p'$.
At a positive letter $i_j=z$, $(v_{>j-1}-v_{>j})\varpi_z=-v_{>j}\alpha_z$ and $(v_{>j-1}-v_{>j})\alpha_y=-a_{zy}v_{>j}\alpha_z$; at a negative letter both differences vanish.
Summing \eqref{eq: B entries} by parts and telescoping over $p<j<p'$ therefore gives
\[
  2\sum_k\lambda^\cc_k b_{k\ell}
  =\bigl(\varepsilon(i_p)v_{>p}-\varepsilon(i_{p'})v_{>p'-1}\bigr)
    \Bigl(\sum_z a_{zy}\varpi_z-\alpha_y\Bigr)=0.
\]

The following lemma follows immediately from the root-sequence property of reduced words recalled in \cref{sec: Lie}.
Geometrically, a reduced gallery from the dominant to the opposite Weyl chamber crosses each reflecting hyperplane exactly once.

\begin{lemma}
  \label{lem: sign}
  Let $\bfi$ be a double reduced word, $g\in[1,r]$, $i_m$ the positive letter with root $\alpha_g$, and $x$ its level.
  For $k\in C(\bfi)$, $\lambda^\cc_k(\alpha_g^\vee)\ge0$ if $k$ has a slot $\ge m$, and $\lambda^\cc_k(\alpha_g^\vee)\le0$ if $k$ has a slot $\le m-1$; moreover,
  \[
    \lambda^\cc_{c_x(m-1)}(\alpha_g^\vee)=-1,
    \qquad \lambda^\cc_{c_x(m)}(\alpha_g^\vee)=1.
  \]
  In particular, $\lambda^\cc_k(\alpha_g^\vee)=0$ for every chamber $k$ with slots on both sides of $m$.
\end{lemma}
\begin{proof}

  Let $\beta^\vee_s:=v_{>s}^{-1}(\alpha_g^\vee)$; then $\lambda^\cc_{c_z(s)}(\alpha_g^\vee)$ is the coefficient of $\alpha_z^\vee$ in $\beta^\vee_s$.
  A positive letter $i_j$ at the level $z$ gives $\beta^\vee_{j-1}=s_z(\beta^\vee_j)$, which has the sign of $\beta^\vee_j$ unless $\beta^\vee_j=\pm\alpha_z^\vee$, i.e.\ the root $v_{>j}(\alpha_z)$ of $i_j$ is $\pm\alpha_g$.
  Roots of positive letters being positive and pairwise distinct, this only happens for $j=m$, with $\beta^\vee_m=\alpha_x^\vee$.
  Negative letters do not change $\beta^\vee_s$, and $\beta^\vee_L=\alpha_g^\vee$; so $\beta^\vee_s$ is positive for $s\ge m$ and negative for $s\le m-1$, and $\beta^\vee_{m-1}=-\alpha_x^\vee$.
\end{proof}
Applying \cref{lem: sign} to $\bfi^T$, whose column weights at $k^T$ are the row weights at $k$, gives the analogue for a negative letter $i_m$ with root $\alpha_g$.
Thus $\lambda^\rr_k(\alpha_g^\vee)\le0$ if $k$ has a slot $\ge m$, $\lambda^\rr_k(\alpha_g^\vee)\ge0$ if $k$ has a slot $\le m-1$, and $\lambda^\rr_{c_x(m-1)}(\alpha_g^\vee)=1$, $\lambda^\rr_{c_x(m)}(\alpha_g^\vee)=-1$.

For a root $a$ of either kind we write $m_a$, $t_a$ and $e_a$ for $m^\rr_a$, $t^\rr_a$ and $e^\rr_a$, resp.\ $m^\cc_a$, $t^\cc_a$ and $e^\cc_a$.

\begin{lemma}
  \label{lem: exponents}
  Let $\bfi$ be a double reduced word and $a$ a row or column root, with inversion letter $i_m$ at the level $x$.
  If $b_{k\ell}$ are the entries of $\tB(\bfi)$ then, for every $\bfi$-exchangeable chamber $\ell$,
  \[
    \sum_{k\in P}e_a(k)\,b_{k\ell}=
    \begin{cases}
      -\varepsilon(i_m)&\text{if }\ell=c_x(m-1),\\
      \varepsilon(i_m)&\text{if }\ell=c_x(m),\\
      0&\text{otherwise}.
    \end{cases}
  \]
\end{lemma}
\begin{proof}
  By \cref{lem: mirror}(b) we may assume that $a=b$ is a column root, so that $\varepsilon(i_m)=1$; let $g:=\gamma_c(b)$, so that $e_b(k)=[\lambda^\cc_k(\alpha_g^\vee)]_+$, and recall \cref{lem: sign}.
  An entry $b_{k\ell}$ vanishes unless $k$ and $\ell$ are at the same level and separated by one letter, or at distinct levels with a common slot.
  If all the slots of $\ell$ are $\ge m$, apply \eqref{eq: homogeneity} to the word $(i_m,\dots,i_L)$.
  Its slots are $[m-1,L]$, and its chambers are the intersections with $[m-1,L]$ of the chambers of $\bfi$ with a slot $\ge m-1$, with the same column weights and the same entries in the column $\ell$.
  Among them the only chamber with negative pairing is $c_x(m-1)$, whose pairing is $-1$; so $\sum_ke_b(k)b_{k\ell}=b_{c_x(m-1),\ell}$, which is $1$ if $\ell=c_x(m)$ and $0$ otherwise.
  If all the slots of $\ell$ are $\le m-1$, a chamber $k$ with $e_b(k)\neq0$ has no slot in common with $\ell$, so $\sum_ke_b(k)b_{k\ell}$ is $b_{c_x(m),c_x(m-1)}=-1$ if $\ell=c_x(m-1)$ and $0$ otherwise.
  Otherwise $\ell$ has the slots $m-1$ and $m$, and is at a level $y\neq x$.
  Apply \eqref{eq: homogeneity} to the word $\hat{\boldsymbol{\imath}}:=(-y,i_m,\dots,i_L)$, with $-y$ at the position $m-1$, and to its exchangeable chamber $\hat\ell$ formed by the slots of $\ell$ which are $\ge m-1$.
  The letter $-y$ changes no column weight; so, by the proof of \cref{lem: sign}, the chambers of $\hat{\boldsymbol{\imath}}$ with a slot $\le m-1$ have $\lambda^\cc_k(\alpha_g^\vee)=0$, except the initial chamber $\hat k$ at the level $x$, with $-1$.
  The remaining chambers are those of $\bfi$ whose slots are all $\ge m$, with the same entries in the columns $\hat\ell$ and $\ell$.
  So $\sum_ke_b(k)b_{k\ell}$ is the entry of $\tB(\hat{\boldsymbol{\imath}})$ at $(\hat k,\hat\ell)$, given by the consecutive elements $m-1<m$ of $M_{xy}(\hat{\boldsymbol{\imath}})$; it vanishes, as $\varphi_{xy}(m-1)=-1=\varphi_{xy}(m)$.
\end{proof}

For a double reduced word $\bfi$ let $\Sigma^\Phi_{\gamma,\rm fr}(\bfi)$ be the seed obtained from $\Sigma^\Phi_\gamma(\bfi)$ by freezing the thawed chambers. This seed and $\Sigma(\bfi)$ have the $\bfi$-exchangeable chambers as exchangeable vertices.
For a seed $(\bx,\tB)$ and an exchangeable index $\ell$ let $\hat y_\ell:=\prod_kx_k^{b_{k\ell}}$.

\begin{corollary}
  \label{cor: Fraser}
  For every double reduced word $\bfi$, $\Sigma(\bfi)$ and $\Sigma^\Phi_{\gamma,\rm fr}(\bfi)$ have the same principal part and the same $\hat y_\ell$ for every $\bfi$-exchangeable chamber $\ell$.
  So the identity of $\bbC(\mcG)$ is a quasi-isomorphism, in the sense of \cite[Definition~3.5]{Fra16}, between the seed patterns of $\Sigma(\bfi)$ and of $\Sigma^\Phi_{\gamma,\rm fr}(\bfi)$.
\end{corollary}
\begin{proof}

  The principal parts agree by \cref{lem: model local}(b).
  By \cref{lem: model local}(a), the column $\ell$ of $\tB_\gamma(\bfi)$ is the column $\ell$ of $\tB(\bfi)$ plus the dipole entries $d_{t_a\ell}$.
  As $\Phi_\gamma(k;\bfi)=\Delta(k;\bfi)\prod_a\Phi_\gamma(t_a;\bfi)^{e_a(k)}$ for every $k\in P$, the quotient of the two $\hat y_\ell$ is $\prod_a\Phi_\gamma(t_a;\bfi)^{\sum_ke_a(k)b_{k\ell}+d_{t_a\ell}}$, which is $1$ by \cref{lem: exponents} and \cref{def: model}(c).
  For $\gamma=(0,0)$ the two seeds coincide. Otherwise $\mcG$ has rank at least $2$, and so do the seeds, as required in \cite[Proposition~2.3]{Fra16}.
  The last claim then follows from \cite[Propositions~2.3 and~3.2]{Fra16}, applied to the identity in both directions: each $\Delta(k;\bfi)$ is $\Phi_\gamma(k;\bfi)$ times a Laurent monomial in the frozen variables of $\Sigma^\Phi_{\gamma,\rm fr}(\bfi)$, and conversely, by the unitriangularity after \cref{def: model}.
\end{proof}

\begin{proof}[Proof of \cref{thm: moves}(a)]
  Let the commutation swap the letters $i_j,i_{j+1}$, at the levels $x\neq y$.
  If both are negative, then $a_{xy}=0$, the roots $u_{\le j-1}(\alpha_x)$ and $u_{\le j}(\alpha_y)=u_{\le j-1}(\alpha_y)$ of the two letters are swapped, and the other roots do not change.
  Similarly, if both are positive, the roots $v_{>j}(\alpha_x)=v_{>j+1}(\alpha_x)$ and $v_{>j+1}(\alpha_y)$ are swapped.
  If the letters have opposite signs, no root changes.
  So the inversion letters of $\bfi$ are those of $\bfi'$, possibly moved from $j$ to $j+1$ or conversely, and the thawed chambers correspond.
  The order of two inversion letters changes only if they are $i_j,i_{j+1}$, negative say.
  Then their roots are orthogonal, so they are $\alpha_{\gamma_r(a)}$, $\alpha_{\gamma_r(a')}$ with $a_{aa'}=0$, $\gamma_r$ being an isometry, and the virtual letters of $a$ and $a'$ commute.
  So $(\bfi')^\gamma$ is obtained from $\bfi^\gamma$ by commutations, and $\tB((\bfi')^\gamma)_P=\tB(\bfi^\gamma)_P$ by \cref{lem: commutation}.
  The two chambers separated by an inversion letter correspond to the two chambers separated by the same letter in $\bfi'$, so $D_\gamma(\bfi')=D_\gamma(\bfi)$.
  The elements $u_{\le s}$, $v_{>s}$ do not change for slots $s\neq j$, and every chamber contains a slot $s\neq j$.
  So the weights, the minors and the exponents of corresponding chambers coincide.
\end{proof}

\begin{proof}[Proof of the statements on the matrices in \cref{thm: moves}(b)]
We separate the ordinary word matrix from the dipole corrections.
After checking that they do not cancel at the mutated chamber, we compare the corrections and the order of virtual letters.

  Under the identification of \cref{lem: mirror}, sign swaps become sign swaps, braid moves of negative letters become braid moves of positive letters, and row roots become column roots.
  As mutation is involutive, the reverse of a move satisfies the same identity.
  Thus it suffices to treat the sign swap $(-x,x)\to(x,-x)$ and the positive braid move $(x,y,x)\to(y,x,y)$, $a_{xy}=a_{yx}=-1$, both starting at $p$.
  Let $p'$ be the last position of the move and $v:=v_{>p'}$.

  \emph{Local data.}
  \Cref{fig: moves} shows the chambers named below.
  For the sign swap, let $k_-=c_x(p-1)$, $k=c_x(p)$ and $k_+=c_x(p+1)$; in $\bfi'$ the replaced chamber is $k'=c_x(p)$.
  By \eqref{eq: B entries}, the non-zero entries of the column $k$ of $\tB(\bfi)$ are $b_{k_-k}=b_{k_+k}=-1$ and $b_{c_z(p),k}=-a_{zx}$ for the levels $z\neq x$.
  For the braid move, let $k_-=c_x(p-1)$, $k=c_x(p)$, $k_+=c_x(p+2)$, $\ell_-=c_y(p)$ and $\ell_+=c_y(p+1)$; in $\bfi'$ the replaced chamber is $k'=c_y(p)$.
  The non-zero entries of the column $k$ of $\tB(\bfi)$ are $b_{k_-k}=b_{\ell_+k}=1$ and $b_{k_+k}=b_{\ell_-k}=-1$, as for a level $z\notin\{x,y\}$ the letters at $p$ and $p+2$ have the same value of $\varphi_{zx}$.

  \emph{Dipoles at $k$.}
  By \cref{lem: model local}, the column $k$ of $\tB_\gamma(\bfi)$ is the column $k$ of $\tB(\bfi)$ plus the dipole entries $d_{t_ak}$ of the roots $a$ with $m_a\in\{p,p'\}$.
  In the sign swap these are a row root with $m_a=p$ and a column root with $m_b=p+1$.
  In the braid move they are column roots $b$ with $m_b\in\{p,p+2\}$, the roots of the letters at $p+2,p+1,p$ being $v(\alpha_x)$, $v(\alpha_x+\alpha_y)$, $v(\alpha_y)$.
  For a column root $b$, $d_{t_bk}=1$ if $m_b=p'$ and $d_{t_bk}=-1$ if $m_b=p$, by \cref{def: model}(c).
  No dipole entry of a column root $b$ cancels an entry of $\tB(\bfi)$.
  Indeed, an entry of the opposite sign at $(t_b,k)$ would force $t_b=k_-$ in a sign swap.
  In a braid move it would force $t_b=\ell_-$ if $m_b=p+2$, resp.\ $t_b=k_-$ if $m_b=p$.
  These are the only possibilities because the other chambers with a non-zero entry in the column $k$ are not initial.
  With $g:=\gamma_c(b)$, $\lambda^\cc_{t_b}(\alpha_g^\vee)=-\varpi_b(\alpha_g^\vee)=0$ by \cref{lem: tails}, as $\gamma_c$ is nilpotent.
  On the other hand, $\lambda^\cc_{k_-}(\alpha_g^\vee)=-1$ in the first and third cases by \cref{lem: sign}.
  Indeed, $k_-$ is $c_x(m_b-1)$ in the third case, and has the column weight of $c_x(m_b-1)=k$ in the first, the letter $i_p$ being negative.
  In the second case $\lambda^\cc_{\ell_-}(\alpha_g^\vee)=-1$, as $v_{>p}^{-1}(\alpha_g^\vee)=s_y(-\alpha_x^\vee)=-\alpha_x^\vee-\alpha_y^\vee$.
  So $t_b$ is none of these chambers.
  Row roots follow by \cref{lem: mirror}.

  \emph{The matrix $\tB(\bfi^\gamma)_P$.}
The ordinary word mutation gives the extended matrix with the old virtual order.
It remains to account for the dipoles and any change in that order.
  By \cref{thm: SSVZ} for the word $\bfi^\gamma$, $\mu_k(\tB(\bfi^\gamma))$ is the matrix of the word obtained from $\bfi^\gamma$ by the move, i.e.\ of $(\bfi')^\gamma$ computed with the virtual letters of $\bfi^\gamma$, in the same order.
  This is $\tB((\bfi')^\gamma)$ unless the move changes the order of the virtual letters.
  As $k\in P_\ex$, restricting to the rows $P$ and the columns $P_\ex$ commutes with $\mu_k$.

  \emph{Dipoles.}
  If no inversion letter is a letter of the move, no dipole entry involves $k$, the inversion letters, the chambers they separate and the order of the virtual letters do not change, and $\mu_k(\tB_\gamma(\bfi))=\mu_k(\tB(\bfi^\gamma)_P)+D_\gamma(\bfi)=\tB_\gamma(\bfi')$.
  Otherwise, as no dipole entry cancels an entry of $\tB(\bfi)$ in the column $k$, the corrections in the mutation rule split into those of $\tB(\bfi^\gamma)_P$ and those involving a dipole entry at $k$. The latter change the columns $t_a$ of the roots $a$ with $m_a$ in the move as follows, which gives $D_\gamma(\bfi')$.
  In \cref{fig: moves} each inversion letter of the move has the color of the inversion letter it becomes in $\bfi'$.
  For a column root $b$ in the sign swap, the column $t_b$ of the correction has entries $-1$ at $k$ and $1$ at $k_+$.
  It becomes the column with $1$ at $k'$ and $-1$ at $k_-$, the entry at $k_+$ canceling.
  Indeed in $\bfi'$, $m_b=p$, $c_x(p-1)=k_-$ and $c_x(p)=k'$.
  In the braid move, if $m_b=p+2$, the column $t_b$, with $-1$ at $k$ and $1$ at $k_+$, becomes the column with $1$ at $k'$ and $-1$ at $\ell_-$, and in $\bfi'$, $m_b=p$, $c_y(p-1)=\ell_-$ and $c_y(p)=k'$.
  If $m_b=p$, the column $t_b$, with $-1$ at $k_-$ and $1$ at $k$, becomes the column with $-1$ at $k'$ and $1$ at $\ell_+$, and in $\bfi'$, $m_b=p+2$, $c_y(p+1)=k'$ and $c_y(p+2)=\ell_+$.
  If in a sign swap $D_a$ and $D_b$ have entries at $k$ for a row root $a$ and a column root $b$, then $d_{t_ak}=d_{t_bk}=1$.
  Hence the mutation at $k$ does not change the entry at $(t_a,t_b)$.
  In a braid move two matrices $D_b$, $D_{b'}$ have entries at $k$ only if $m_b=p+2$ and $m_{b'}=p$.
  Then $d_{t_bk}=1$ and $d_{t_{b'}k}=-1$, the mutation at $k$ adds $1$ at $(t_b,t_{b'})$, and this is accounted for by the change of the order of the virtual letters, treated next.

  \emph{Virtual letters.}
The remaining correction occurs when two inversion letters change order; reversing their virtual letters gives exactly this correction.

  The order of two inversion letters of the same kind changes only in this last case.
  Then $\alpha_{\gamma_c(b)}=v(\alpha_x)$ and $\alpha_{\gamma_c(b')}=v(\alpha_y)$, so $a_{bb'}=a_{xy}=-1$, and the virtual letters of $b$ and $b'$ are at the levels $z:=b^*$ and $w:=b'^*$, with $a_{zw}=-1$.
  In $\bfi'$ the inversion letters of $b$ and $b'$ are at $p$ and $p+2$, so their order is reversed, and so is the order of the two virtual letters.
  In $M_{zw}(\bfi^\gamma)$ the two virtual letters are positive.
  The only entry depending on their order is the one at $(t_b,t_{b'})$.
  It is given by the pair formed by the second virtual letter and the next element $j$ of $M_{zw}$.
  Indeed, the entries for the slots before it involve chambers of $\bfi^\gamma$ which are not chambers of $\bfi$.
  In $\bfi$, $m_{b'}<m_b$, so the virtual letter of $b$ comes first.
  By \eqref{eq: B entries} the entry is $\frac12(\varphi_{zw}(j)-1)$, and in $\bfi'$, where the virtual letter of $b$ comes second, it is $\frac12(\varphi_{zw}(j)+1)$, larger by $1$.
\end{proof}

\begin{proof}[Proof of the statements on the matrices in \cref{thm: moves}(c)]
We again separate the word matrix from the dipoles, now through three mutations.
There is at most one relevant dipole, so its correction can be followed explicitly.

  By \cref{lem: mirror}, and since $\mu_k\mu_\ell\mu_k$ is an involution, so that the reverse of a move satisfies the same identity, it suffices to treat the positive braid move $(x,y,x,y)\to(y,x,y,x)$ starting at $p$, with $\alpha_x$ long, so that $a_{xy}=-1$ and $a_{yx}=-2$.
  We keep the notation of the proof of (b), with $p'=p+3$ and $v:=v_{>p'}$.

  \emph{Local data.}
  As in \cref{fig: moves}, let $k_-=c_x(p-1)$, $k=c_x(p)$, $k_+=c_x(p+2)$, $\ell_-=c_y(p)$, $\ell=c_y(p+1)$ and $\ell_+=c_y(p+3)$; in $\bfi'$ the replaced chambers are $k'=c_x(p+1)$ and $\ell'=c_y(p)$, corresponding to $k$ and $\ell$ by level.
  By \eqref{eq: B entries}, the non-zero entries of the columns $k$ and $\ell$ of $\tB(\bfi)$ are $b_{k_-k}=1$, $b_{k_+k}=-1$, $b_{\ell_-k}=-2$, $b_{\ell k}=2$, and $b_{\ell_-\ell}=1$, $b_{\ell_+\ell}=-1$, $b_{k\ell}=-1$, $b_{k_+\ell}=1$.
  So along $\mu_k,\mu_\ell,\mu_k$ those of the columns of the mutated chambers in the matrices obtained from $\tB(\bfi)$ are these, then $b_{k\ell}=1$, $b_{\ell_-\ell}=b_{\ell_+\ell}=-1$, then $b_{k_-k}=-1$, $b_{k_+k}=1$, $b_{\ell_+k}=-2$, $b_{\ell k}=2$.

  \emph{Roots in the move.}
We identify the possible inversion letters in the move and exclude cancellation with the ordinary matrix at either replaced chamber.

  The roots of the letters at $p,p+1,p+2,p+3$ are $v(\alpha_x)$, $v(\alpha_x+\alpha_y)$, $v(\alpha_x+2\alpha_y)$ and $v(\alpha_y)$, and only the first and the last can be simple, the others being sums of two of these positive roots.
  So the only roots whose inversion letter is a letter of the move are column roots $b$ with $m_b=p$ or $m_b=p+3$, colored in \cref{fig: moves}.
  There is at most one such root: otherwise $\gamma_c$ would map two simple roots joined by a double bond to the simple roots $v(\alpha_x)$, $v(\alpha_y)$, also joined by a double bond.
  As $\gamma_c$ is an isometry and $\mcG$ has only one such pair, $\gamma_c$ would permute it, contradicting nilpotency.
  In particular the move does not change the order of the virtual letters, and the dipole entries at $k$ and $\ell$ are those of $D_b$.
  If there is no such root, there are none, and the mutations of $\tB_\gamma(\bfi)$ are those of $\tB(\bfi^\gamma)_P$.
  We may therefore assume that $b$ exists.
  As $t_b$ is initial, its entries in the columns $k$ and $\ell$ of $\tB(\bfi)$ vanish unless $t_b=k_-$ or $t_b=\ell_-$.
  Put $g:=\gamma_c(b)$; then $\lambda^\cc_{t_b}(\alpha_g^\vee)=0$ by \cref{lem: tails} and nilpotency, while the level $b^*$ of $t_b$ has the root length of $\alpha_g$, as $\gamma_c$ is an isometry.
  If $m_b=p$, root lengths exclude $t_b=\ell_-$, and $\lambda^\cc_{k_-}(\alpha_g^\vee)=-1$ by \cref{lem: sign} excludes $t_b=k_-$.
  If $m_b=p+3$, root lengths exclude $t_b=k_-$, and $v_{>p}^{-1}(\alpha_g^\vee)=-2\alpha_x^\vee-\alpha_y^\vee$ gives $\lambda^\cc_{\ell_-}(\alpha_g^\vee)=-1$, excluding $t_b=\ell_-$.
  Thus both entries vanish.

  \emph{Mutation calculation.}
The vanishing of the ordinary row allows us to compute the dipole correction separately through the three mutations.

  By \cref{lem: model local}, the matrices obtained from $\tB_\gamma(\bfi)$ and from $\tB(\bfi)$, in the columns $k$ and $\ell$, initially differ only in the row $t_b$.
  Write $\delta=(\delta_k,\delta_\ell)$ for this correction, and $E$ for the correction to the matrix obtained from $\tB(\bfi^\gamma)_P$ due to $D_b$; initially $E=D_b$, and $(E_{t_bk},E_{t_b\ell})=\delta$.
  The tables below record these corrections along the three mutations. Entries of $E$ not listed are zero, apart from the entries of the column $t_b$, which are computed in the same way, and entries in columns outside $P_\ex$ are omitted, as for $D_b$.
  The row $t_b$ of the matrices obtained from $\tB(\bfi)$ remains zero in the columns $k,\ell$ throughout the three mutations.
  Indeed, each mutation is at a column where that row is zero, so the matrix mutation formula leaves the row unchanged.
  Hence no cancellation occurs at the mutated chamber.
  Thus the mutations of $\tB_\gamma(\bfi)$ split into those of $\tB(\bfi^\gamma)_P$ and the corrections $E$.
  The corrections in the two cases are:
  \[
    \begin{array}{c|c|c}
      \multicolumn{3}{c}{m_b=p}\\[2pt]
      \text{stage}&\delta&\text{other entries of }E\\ \hline
      \text{initial}&(-1,0)&E_{t_bk_-}=1\\
      \mu_k&(1,-1)&\text{none}\\
      \mu_\ell&(-1,1)&\text{none}\\
      \mu_k&(1,0)&E_{t_bk_+}=-1
    \end{array}
    \qquad
    \begin{array}{c|c|c}
      \multicolumn{3}{c}{m_b=p+3}\\[2pt]
      \text{stage}&\delta&\text{other entries of }E\\ \hline
      \text{initial}&(0,1)&E_{t_b\ell_+}=-1\\
      \mu_k&(0,1)&E_{t_b\ell_+}=-1\\
      \mu_\ell&(0,-1)&E_{t_b\ell_-}=1\\
      \mu_k&(0,-1)&E_{t_b\ell_-}=1
    \end{array}
  \]

  \emph{Matrices.}
  By \cref{thm: SSVZ} for the word $\bfi^\gamma$, whose virtual letters keep their order, $\mu_k\mu_\ell\mu_k(\tB(\bfi^\gamma)_P)=\tB((\bfi')^\gamma)_P$, and the other dipoles do not meet $k$ or $\ell$ and remain unchanged.
  If $m_b=p$, the final correction in the column $t_b$ is $-1$ at $k'$ and $1$ at $k_+$, and in $\bfi'$ the inversion letter is at $p+3$, separating $k'=c_x(p+2)$ from $k_+=c_x(p+3)$.
  If $m_b=p+3$, it is $1$ at $\ell'$ and $-1$ at $\ell_-$, and the inversion letter is at $p$, separating $\ell_-=c_y(p-1)$ from $\ell'=c_y(p)$.
  Thus in both cases the correction is $D_b(\bfi')$, as required.
\end{proof}

To recover the functions from \cref{cor: Fraser}, we use the following lemma.

\begin{lemma}
  \label{lem: BFZ indecomposable}
  For every double reduced word $\bfi$ the principal part of $\tB(\bfi)$ is block-indecomposable.
  In particular, if $\mcG$ has rank at least $2$, every $\bfi$-exchangeable chamber $k$ has an $\bfi$-exchangeable chamber $\ell$ with $b_{\ell k}\neq0$.
\end{lemma}
\begin{proof}
We first rule out a separation of two levels in a reduced word.
We then use this observation to find an exchangeable connection between every pair of adjacent levels.

  A reduced word for $w_0$ cannot have all its letters at a level $x$ before all its letters at another level $y$.
  Otherwise $w_0=uv$ with $u$ not involving $s_y$ and $v$ not involving $s_x$.
  Then $v$ fixes $\varpi_x$, so $\varpi_x+\varpi_{x^*}=\varpi_x-w_0(\varpi_x)=\varpi_x-u(\varpi_x)$.
  As $s_z(\mu)=\mu-\mu(\alpha_z^\vee)\alpha_z$, this is a combination of the $\alpha_z$ with $s_z$ in $u$, so its coefficient at $\alpha_y$ is $0$.
  But the coefficients of the fundamental weights in the simple roots are strictly positive, the root system being irreducible \cite[Theorem~4.3 (Fin)]{Kac90}.

  At every level there are letters of both signs, so at least one exchangeable chamber, and consecutive exchangeable chambers $k,k'$ at the same level have $b_{kk'}=\pm1$.
  Let $x,y$ be levels with $a_{xy}\neq0$. We will find exchangeable chambers $k,\ell$ at the levels $x$ and $y$ with $b_{k\ell}\neq0$.
  Let $F_x,L_x$ and $F_y,L_y$ be the first and last positions of letters at these levels, and assume $F_x<F_y$.
  Then $F_y<L_x$, as otherwise every letter at the level $x$ comes before every letter at the level $y$.
  For consecutive elements $q<q'$ of $M_{xy}$ with $q\in[F_y,\min(L_x,L_y)-1]$, the chambers $c_x(q)$ and $c_y(q)$ are exchangeable, and $b_{c_x(q)c_y(q)}$ vanishes only if $\varphi_{xy}(q)=\varphi_{xy}(q')$.
  If all these entries vanish, $\varphi_{xy}$ is constant, equal to $c$ say, on $M_{xy}\cap[F_y,\min(L_x,L_y)]$.
  If $L_y<L_x$, then all the letters at the level $y$ have the sign $c$, which is impossible.
  If $L_x<L_y$, then the letters at the level $x$ after $F_y$ have the sign $-c$ and those at the level $y$ before $L_x$ have the sign $c$.
  So in the reduced word for $w_0$ formed by the letters of sign $c$ all the letters at the level $x$ come before those at the level $y$, which is impossible.
\end{proof}

\begin{proof}[Proof of the statements on the functions in \cref{thm: moves}(b) and (c)]
The matrix calculation and quasi-isomorphism leave only possible frozen monomial factors at the replaced chambers.
We eliminate these factors by comparing exchange monomials and using block-indecomposability.

  If $\gamma=(0,0)$, these statements are \cref{thm: SSVZ}; so we may assume that $\mcG$ has rank at least $2$.
  Let $\mu$ be the mutation, or the sequence of mutations, of the statement, and $S:=\mu(\Sigma^\Phi_\gamma(\bfi))$.
  By the statements on the matrices, $S$ has the matrix $\tB_\gamma(\bfi')$.
  Its functions at the chambers which are not replaced are those of $\Sigma^\Phi_\gamma(\bfi')$.
  Indeed, the weights of such a chamber can be computed at a slot outside the move, where $u_{\le s}$ and $v_{>s}$ do not change.
  Also, the $\Phi_\gamma(t_a;\bfi)$ do not depend on the word by \cref{lem: tails}.
  Freezing the thawed chambers commutes with $\mu$; let $S_{\rm fr}$ be $S$ with them frozen.
  The quasi-isomorphism of \cref{cor: Fraser} compares $S_{\rm fr}$ with $\mu(\Sigma(\bfi))=\Sigma(\bfi')$ (\cref{thm: SSVZ}): their $\hat y_\ell$ agree and corresponding functions differ by Laurent monomials in the frozen variables.
  Applying \cref{cor: Fraser} at $\bfi'$ gives the same comparison with $\Sigma^\Phi_{\gamma,\rm fr}(\bfi')$. The frozen variables of all three seeds generate the same group of Laurent monomials by unitriangularity.
  So $S_{\rm fr}$ and $\Sigma^\Phi_{\gamma,\rm fr}(\bfi')$ have the same matrix and the same $\hat y_\ell$.
  Their functions differ at most at the replaced chambers $k$ of $\bfi'$, by Laurent monomials $\prod_tx_t^{n_t(k)}$ in the frozen variables $x_t$.
  Comparing the $\hat y_\ell$, and as the variables of a seed are algebraically independent, $\sum_k n_t(k)b_{k\ell}=0$ for every frozen $t$ and every $\bfi'$-exchangeable $\ell$, the $b_{k\ell}$ being the entries of $\tB(\bfi')$.
  If $k'$ is the only replaced chamber, this gives $n_t(k')=0$ for some $\ell$ with $b_{k'\ell}\neq0$, which exists by \cref{lem: BFZ indecomposable} and skew-symmetrizability.
  For a braid move of length $4$, the choices $\ell=k'$ and $\ell=\ell'$ give $n_t(\ell')b_{\ell'k'}=0$ and $n_t(k')b_{k'\ell'}=0$, with $b_{\ell'k'}\neq0\neq b_{k'\ell'}$.
  So $S=\Sigma^\Phi_\gamma(\bfi')$.
\end{proof}

\begin{remark}
  The statements on the functions in \cref{thm: moves} can also be proved without quasi-isomorphisms.
  Substituting $\Phi_\gamma(k;\bfi)=\Delta(k;\bfi)\prod_a\Phi_\gamma(t_a;\bfi)^{e_a(k)}$ in the exchange relations of $\Sigma^\Phi_\gamma(\bfi)$ at the replaced chambers reduces them to those of $\Sigma(\bfi)$ (\cref{thm: SSVZ}) and to identities among the exponents $e_a$. These identities can be checked by a direct, but more tedious, inspection of the position of the inversion letters relative to the move.
\end{remark}

\begin{corollary}
  \label{cor: mutation equivalent}
  For every $\mcG$ and every aperiodic BD pair $\gamma$, the seeds $\Sigma^\Phi_\gamma(\bfi)$ of all double reduced words $\bfi$ are related by sequences of mutations, up to the correspondence of chambers; so are the seeds $\Sigma_\gamma(\bfi)$ if $h_\gamma$ is birational.
\end{corollary}
\begin{proof}

  Any two double reduced words are related by moves (\cref{sec: double words}).
  Moves of length at most $4$ are treated in \cref{thm: moves}.
  Braid moves of length $6$ only occur in type $G_2$, where the only BD pair is $(0,0)$, as BD maps are isometries and the two simple roots have different lengths.
  Then $\Sigma^\Phi_\gamma(\bfi)$ is the seed of \cite[Theorem~2.10]{BFZ05}, and \cref{thm: SSVZ} applies.
\end{proof}

Suppose that $h_\gamma^{-1}$ exists, i.e.\ that $h_\gamma$ is birational.
Then $f\mapsto f\circ h_\gamma^{-1}$ is a quasi-isomorphism from the seed pattern of $\Sigma(\bfi)$ to that of $\Sigma_\gamma(\bfi)$ with the thawed chambers frozen.
Consider the cluster variables obtained without mutating at the thawed chambers.
Up to Laurent monomials in the frozen variables and in the functions of the thawed chambers, they are those of the cluster structure of \cite{BFZ05}, composed with $h_\gamma^{-1}$.

\begin{lemma}
  \label{lem: indecomposable}
  For every double reduced word $\bfi$, the principal part of $\tB_\gamma(\bfi)$ is block-indecomposable.
\end{lemma}
\begin{proof}
  By \cref{lem: model local}(b), the submatrix of $\tB_\gamma(\bfi)$ on the $\bfi$-exchangeable chambers is the principal part of $\tB(\bfi)$, which is block-indecomposable by \cref{lem: BFZ indecomposable}.
  It remains to join each thawed chamber to an $\bfi$-exchangeable chamber.

  Let $a$ be a row root, $t=t^\rr_a$, $i_j$ the last letter at the level $a^*$, and $k=c_{a^*}(j-1)$, which is $\bfi$-exchangeable as every level carries letters of both signs.
  By \eqref{eq: B entries}, $\tB(\bfi^\gamma)_P$ has the entry $\varepsilon(i_j)$ at $(k,t)$.
  As $k$ is not thawed, a dipole entry at $(k,t)$ can only come from $D_a$, and it cancels $\varepsilon(i_j)$ only if the inversion letter $i_m$ of $a$ is the last negative letter at the level $a^*$.
  Then $w_0=u\,s_{a^*}v$ with $u=u_{\le m-1}$ and $v$ not involving $s_{a^*}$, so $\alpha_{\gamma_r(a)}=u(\alpha_{a^*})=-w_0v^{-1}(\alpha_{a^*})$.
  Here $v^{-1}(\alpha_{a^*})$ is a positive root with coefficient $1$ at $\alpha_{a^*}$, and $-w_0$ permutes the simple roots, hence preserves heights; so $v^{-1}(\alpha_{a^*})=\alpha_{a^*}$ and $\gamma_r(a)=a$, contradicting nilpotency.
  Column roots $b$ are treated symmetrically, with $t^\cc_b=c_{b^*}(0)$, the first letter at the level $b^*$ and the first positive letter there.
\end{proof}

\subsection{Identification of the initial seed}
\label{sec: base case concise}
Put $\mathbf e_c=(-c,\dots,-2,-1)$ and
\[
  \bfi_{\rm GSV}=(\mathbf e_1,\dots,\mathbf e_{N-1},\mathbf e_{N-1}^T,\dots,\mathbf e_1^T),
\]
so that $\bfi_{\rm GSV}^T=\bfi_{\rm GSV}$.
For $(i,j)\in[1,N]^2\setminus\{(1,1)\}$ let $F_{ij}=\Delta_{[i,i+s-1],[j,j+s-1]}$, where $s=N+1-\max(i,j)$.
We number the chambers at level $x$ by $j=0,\dots,2N-2x$ in increasing order of their slots.
Define
\begin{equation}
  \label{eq: psi concise}
  \psi_x(j)=
  \begin{cases}
    (j+1,N-x+1)&0\le j\le N-x,\\
    (N-x+1,2N-2x+1-j)&N-x\le j\le2N-2x.
  \end{cases}
\end{equation}
Thus each level traverses the hook $\max(i,j)=N-x+1$, and $\psi$ is a bijection onto $[1,N]^2\setminus\{(1,1)\}$, with initial and final chambers on the first row and column, respectively.
As $k\mapsto k^T$ maps the chamber numbered $j$ at level $x$ to the one numbered $2N-2x-j$, $\psi(k^T)=\psi(k)^T$.
\Cref{fig: GSV23 word seed} shows $\bfi_{\rm GSV}^\gamma$, $\tB_\gamma(\bfi_{\rm GSV})$ and $\psi$ for the running example of \cite{GSV23}.
The BD triples of \cite{GSV23} are, in our notation,
\[
  \mathbf\Gamma^\rr=(\Gamma_r,\gamma_r(\Gamma_r),\gamma_r),\qquad
  \mathbf\Gamma^\cc=(\gamma_c(\Gamma_c),\Gamma_c,\gamma_c^{-1}).
\]

\begin{proposition}
  \label{prop: base case concise}
  Via $\psi$, $\Sigma_\gamma(\bfi_{\rm GSV})$ is the initial seed of \cite[Theorem~4.11]{GSV23}.
\end{proposition}
\begin{proof}
  As $\bfi_{\rm GSV}^T=\bfi_{\rm GSV}$ and $\psi(k^T)=\psi(k)^T$, \cref{lem: mirror} gives $\Delta(k^T;\bfi_{\rm GSV})=\Delta(k;\bfi_{\rm GSV})\circ T$ and $b_{k^T\ell^T}=b_{k\ell}$ in $\tB(\bfi_{\rm GSV})$.
  For the same reason, the inversion letters, thawed chambers, dipoles and virtual letters of the column roots of $\gamma$ are the transposes of those of the row roots of $\gamma^T$.

  \emph{Minors and tails.}
We compute the chamber minors and their exponents, which reproduce the recursion of \cite{GSV23}.
  The block $\mathbf e_c$ moves the entry in position $c+1$ of the row permutation to position $1$.
  Consequently, for the chamber $k$ numbered $j\le N-x$ at level $x$, the row set is $[j+1,j+x]$ and the column set is $[N-x+1,N]$, so $\Delta(k;\bfi_{\rm GSV})=F_{\psi(k)}$, and this holds for all $k$ as $F_{ij}\circ T=F_{ji}$.
  Applying \eqref{eq: pairing A concise} to the rows and columns of $F_{ij}$ gives, for $\psi(k)=(i,j)$,
  \[
    e^\rr_a(k)=[i-j=\gamma_r(a)-N],\qquad
    e^\cc_b(k)=[i-j=N-\gamma_c(b)],
    \qquad
    \psi(t^\rr_a)=(a+1,1),\quad \psi(t^\cc_b)=(1,b+1).
  \]
  Thus \eqref{eq: Phi} reads $\Phi_\gamma(k;\bfi_{\rm GSV})=F_{ij}\pi_{i-j}$, where $\pi_0=1$ and
  \begin{equation}
    \label{eq: tails recursion concise}
    \pi_{g-N}=\begin{cases}F_{a+1,1}\pi_a&g=\gamma_r(a),\\1&g\notin\gamma_r(\Gamma_r),\end{cases}
    \qquad
    \pi_{N-g}=\begin{cases}F_{1,b+1}\pi_{-b}&g=\gamma_c(b),\\1&g\notin\gamma_c(\Gamma_c).
    \end{cases}
  \end{equation}
  This recursion terminates by \cref{lem: tails}, and by \cite[Theorem~4.4, (4.1), and (4.6)]{GSV23}, with the column convention of \cref{rmk: Hc order}, the cluster variables $f_{ij}$ of the initial seed of \cite[Theorem~4.11]{GSV23} satisfy $f_{ij}\circ h_\gamma=F_{ij}\pi_{i-j}$.
  So $x_\gamma(k)=f_{\psi(k)}$.

  \emph{The exchange matrix.}
  By \eqref{eq: B entries}, a negative letter $i_m$ at level $x$ gives the arrow $c_x(m)\to c_x(m-1)$.
  Under $\psi$ these are the arrows $(i,N-x+1)\to(i-1,N-x+1)$ along the hook, and those given by positive letters are their transposes.

  For arrows between adjacent levels, fix $1\le x\le N-2$.
  By \eqref{eq: B entries}, a consecutive pair $p<q$ in $M_{x,x+1}$ with $\varphi_{x,x+1}$-values $(1,-1)$ gives $c_{x+1}(p)\to c_x(p)$; values $(-1,1)$ give the reverse arrow, and equal values give none.
  By transposition, we need only consider pairs in the negative half and the pair crossing the middle of the word.

  At these two levels, the negative half consists of $-x$ in $\mathbf e_x$, followed by $-(x+1),-x$ in each block $\mathbf e_c$ for $x+1\le c\le N-1$.
  Between $\mathbf e_{c-1}$ and $\mathbf e_c$, the consecutive letters $-x,-(x+1)$ have values $(1,-1)$; applying \eqref{eq: psi concise} gives
  \[
    (c-x,N-x)\longrightarrow(c-x+1,N-x+1).
  \]
  Within $\mathbf e_c$, the consecutive letters $-(x+1),-x$ have values $(-1,1)$ and give
  \[
    (c-x+1,N-x+1)\longrightarrow(c-x+1,N-x).
  \]
  Finally, the last $-x$ in $\mathbf e_{N-1}$ and the first $x$ in $\mathbf e_{N-1}^T$ are consecutive in $M_{x,x+1}$, with values $(1,-1)$, and give
  \[
    (N-x,N-x)\longrightarrow(N-x+1,N-x+1).
  \]
  These arrows and their transposes are exactly the arrows between the hooks $N-x+1$ and $N-x$ in $Q_{\varnothing,\varnothing}$ \cite[Section~4.3 and Figure~5]{GSV23}.

  The row inversion for $g=\gamma_r(a)$ is the last letter of $\mathbf e_g$, at level $1$: the letter $-x$ of $\mathbf e_c$ swaps the values $c-x+1$ and $c+1$, which are consecutive precisely when $x=1$.
  Its adjacent chambers map to $(g,N)$ and $(g+1,N)$.
  So the thawing and dipoles of the row roots, and by transposition those of the column roots, give exactly the GSV modifications
  \begin{equation}
    \label{eq: GSV arrows concise}
    (\gamma_r(a),N)\to(a+1,1)\to(\gamma_r(a)+1,N),\qquad
    (N,\gamma_c(b))\to(1,b+1)\to(N,\gamma_c(b)+1).
  \end{equation}

  By \cref{lem: model local}, it remains to check the entries between end chambers.
  On a connected component $[d,d'-1]\subset\Gamma_r$, adjacent virtual letters occur in decreasing order of their $\gamma_r$-images.
  For $a,a+1$ in this component, the last real letter and the two virtual letters on levels $a^*-1,a^*$ have $\varphi$-values $-1,+1,-1$ when $\gamma_r$ preserves orientation, and $-1,-1,+1$ when it reverses orientation.
  By \eqref{eq: B entries}, these give, respectively, the arrow $(a+2,1)\to(a+1,1)$ and no arrow between the old end chambers.
  At the lower end of the component the values $-1,+1$ give $(d+1,1)\to(d,1)$; at the upper end the values $-1,-1$ give no arrow.
  Thus the added first-column arrows are
  \[
    \begin{cases}
      (a+2,1)\to(a+1,1),\quad d-1\le a\le d'-2,&\gamma_r\text{ preserves orientation},\\
      (d+1,1)\to(d,1),&\gamma_r\text{ reverses orientation}.
    \end{cases}
  \]
  Arrows incident with $(1,1)$ are omitted.
  The column virtual letters give the transposes, and no initial and final chambers share a slot, so the extended word adds no entries between them.
  These are precisely the remaining modifications turning $Q_{\varnothing,\varnothing}$ into the quiver $Q_{\mathbf\Gamma^\rr,\mathbf\Gamma^\cc}$ of \cite[Section~4.3]{GSV23}. This completes the seed identification.
\end{proof}

\begin{corollary}
  \label{thm: model concise}
  For every double reduced word $\bfi$, $\Sigma_\gamma(\bfi)$ is a seed of $\mcC_\gamma$.
  Commutations preserve the seed, and sign swaps and braid moves mutate it at the replaced chamber, with the chamber correspondences of \cref{sec: model}.
\end{corollary}
\begin{proof}
  Every double reduced word is connected to $\bfi_{\rm GSV}$ by these moves.
  The claim follows by applying \cref{thm: moves}, using birationality of $h_\gamma$ and the fact that type $A$ is simply-laced.
\end{proof}

\subsection{The global toric action}
\label{sec: toric A concise}
We now prove \cref{thm: GSV}(iii) for arbitrary aperiodic pairs, using only parts (i), (ii) and the seed just identified.
Equivariance gives homogeneity of the initial cluster, and regularity and completeness force homogeneity of the exchange relations.
Let $y_r\in\mcH_{\gamma_r}$ and $y_c\in\mcH_{\gamma_c}$.
As $\mcH$ normalizes $\mcN_\pm^{\Gamma_\bullet}$ and lies in $\mcB_\pm$, the products $y_rUy_c=\big(y_rV_r(U)y_r^{-1}\big)\big(y_rV_r(U)^{-1}Uy_c\big)=\big(y_rUV_c(U)^{-1}y_c\big)\big(y_c^{-1}V_c(U)y_c\big)$ are again factorizations as in \cref{sec: h}, so
\[
  V_r(y_rUy_c)=y_rV_r(U)y_r^{-1},\qquad V_c(y_rUy_c)=y_c^{-1}V_c(U)y_c,
\]
and the same formulas for $\overline{V}_r,\overline{V}_c$.
The defining character equations of these tori imply that $\boldsymbol\gamma_r$ and $\boldsymbol\gamma_c$ commute with the respective conjugations.
The formulas for $H_r,H_c$ therefore give $h_\gamma(y_rUy_c)=y_rh_\gamma(U)y_c$.
For any double reduced word $\bfi$, $\Phi_\gamma(k;\bfi)$ is a monomial in generalized minors, so birationality and equivariance of $h_\gamma$ imply that $x_\gamma(k)$ is $\mcH_\gamma$-homogeneous.
In a basis of the character lattice of $\mcH_\gamma$, the weights of the initial cluster variables are the rows of a matrix $W$, whose local toric action (\cref{sec: cluster conventions}) is the action of $\mcH_\gamma$.

To prove $\tB^TW=0$, suppose the two monomials in an exchange relation $x_kx'_k=M_++M_-$ have different weights.
Regularity of $x'_k$ and the weight decomposition of $\bbC[SL_N]$ then imply that $M_+/x_k$ is regular.
By completeness, it belongs to the Laurent ring of the adjacent cluster, where
\[
  \frac{M_+}{x_k}=\frac{M_+x'_k}{M_++M_-}.
\]
The exchange column is nonzero by full rank, so $M_++M_-$ is a nonunit binomial in that ring.
Its monomials have disjoint supports, so in that factorial ring it is coprime to every monomial. As $x'_k$ and the exchangeable variables are units there, $M_+x'_k$ is a unit times a monomial in the frozen variables, and so are its divisors.
So the displayed fraction does not belong to the ring, a contradiction.
Thus $\tB^TW=0$, and the action is global.

Finally, as the initial cluster generates $\bbC(SL_N)$, the rank of $W$ is the dimension of the image of $\mcH_\gamma$ in the automorphisms of $\bbC(SL_N)$.
If $(y_r,y_c)$ acts trivially, then $y_rUy_c=U$ for all $U$, so $y_c=y_r^{-1}$ and $y_r$ is central; hence this image has dimension $\dim\mcH_\gamma=2(N-1)-|\Gamma_r|-|\Gamma_c|$.
This is the number of frozen variables, the largest possible rank (\cref{sec: cluster conventions}), so the global toric action is maximal.

\section{The symplectic group and its BD pairs}
\label{sec: lift}
\subsection{The symplectic group as a fixed point set}
\label{sec: Sp}
For $n=1$, $Sp_2=SL_2$ and the only BD pair is $(0,0)$. From now on $n\ge2$, $N=2n$, $\tG=\SL$ with the standard pinning, and $\widetilde{\mfg}=\mathfrak{sl}_{2n}$.
Let $\tau(k):=2n-k$ on $[1,2n-1]$; we also write $\tau$ for the induced involutions of the simple roots, of the weights, and of $S_{2n}$ (conjugation by $w_0$).
Thus $\tau(k)=k^*$ in type $A_{2n-1}$; we retain the notation $\tau$ for this involution to distinguish it from $*$ in type $C_n$, where $k^*=k$.
Let
\[
  J:=\sum_{i=1}^{N}(-1)^{i+1}E_{i,N+1-i},
  \qquad
  \vt(U):=J\,(U^T)^{-1}J^{-1}.
\]
The alternating signs are chosen so that $\vt$ preserves the standard pinning.
Then $J^T=-J=J^{-1}$, $\vt$ is an involutive automorphism of $\tG$, and its fixed point set is $\mcG:=\{U\mid UJU^T=J\}=\Sp$.
Its differential is $\Vt(\xi)=-J\xi^TJ^{-1}$ for $\xi\in\widetilde{\mfg}$.

\begin{lemma}
  \label{lem: pinning}
  $\Vt(E_k)=E_{\tau(k)}$, $\Vt(F_k)=F_{\tau(k)}$ and $\Vt(\alpha_k^\vee)=\alpha_{\tau(k)}^\vee$.
  Hence $\vt$ preserves $\widetilde{\mcH}$, $\widetilde{\mcB}_\pm$, $\widetilde{\mcN}_\pm$, $\vt(\overline{w})=\overline{\tau(w)}$ for $w\in S_{2n}$, $\vt$ commutes with the transpose, and $\Vt$ preserves the trace form.
\end{lemma}
\begin{proof}
  We have $JE_{ij}J^{-1}=(-1)^{i+j}E_{w_0(i)w_0(j)}$, so $\Vt(E_k)=-JE_{k+1,k}J^{-1}=E_{N-k,N-k+1}=E_{\tau(k)}$, and similarly for $F_k$.
  Since $\overline{s_k}=\exp(-E_k)\exp(F_k)\exp(-E_k)$, we have $\vt(\overline{s_k})=\overline{s_{\tau(k)}}$.
  Moreover, $\vt(U)^T=J^{-T}U^{-1}J^T=JU^{-1}J^{-1}=\vt(U^T)$.
  Also $\operatorname{tr}(\Vt(\xi)\Vt(\eta))=\operatorname{tr}(J\xi^T\eta^TJ^{-1})=\operatorname{tr}(\xi\eta)$, so $\Vt$ preserves the trace form.
\end{proof}

As $\mcG$ is the fixed point set of $\vt$, $\mfg=\mathrm{Lie}(\mcG)=\widetilde{\mfg}^\Vt$; let $\mfp$ be the $(-1)$-eigenspace of $\Vt$.
As $\Vt$ preserves the trace form, the decomposition $\widetilde{\mfg}=\mfg\oplus\mfp$ is orthogonal for it.
We equip $\mfg$ with the restricted trace form and write $\pr:=\frac12(1+\Vt)$ for the orthogonal projection onto $\mfg$.
The group $\mcG$ is simply connected of type $C_n$, with maximal torus $\mcH=\widetilde{\mcH}^\vt$, Cartan subalgebra $\mfh=\widetilde{\mfh}^\Vt$, $\mcB_\pm=\widetilde{\mcB}_\pm\cap\mcG$, $\mcN_\pm=\widetilde{\mcN}_\pm\cap\mcG$, and Chevalley generators
\[
  E^C_k:=E_k+E_{\tau(k)},\quad F^C_k:=F_k+F_{\tau(k)}\quad(k<n),
  \qquad
  E^C_n:=E_n,\quad F^C_n:=F_n.
\]
The simple roots are $\alpha^C_k:=\alpha_k|_\mfh=\alpha_{\tau(k)}|_\mfh$, $k\in[1,n]$, the long one being $\alpha^C_n$, and the fundamental weights are $\varpi^C_k=\varpi_k|_\mfh=\varpi_{\tau(k)}|_\mfh$.
The Weyl group of $\mcG$ is $W^\tau=\{w\in S_{2n}\mid ww_0=w_0w\}$, with simple reflections $s^C_k=s_ks_{\tau(k)}$ ($k<n$) and $s^C_n=s_n$, and longest element $w_0$.
  Since $E_k,F_k$ commute with $E_{\tau(k)},F_{\tau(k)}$ for $k<n$, the representative of $s^C_k$ in $\mcG$ is $\overline{s_k}\,\overline{s_{\tau(k)}}$, so the representative of $w\in W^\tau$ in $\mcG$ is its representative $\overline{w}$ in $\tG$.
The transpose of $\tG$ preserves $\mcG$ and restricts to the transpose of $\mcG$.

\begin{lemma}
  \label{lem: minors restrict}
  Let $k\in[1,2n-1]$ and $k'=\min(k,\tau(k))$.
  For $u,v\in W^\tau$, $\Delta_{u\varpi_k,v\varpi_k}|_\mcG=\Delta_{u\varpi^C_{k'},v\varpi^C_{k'}}$, and for $u,v\in S_{2n}$, $\Delta_{u\varpi_k,v\varpi_k}\circ\vt=\Delta_{\tau(u)\varpi_{\tau(k)},\tau(v)\varpi_{\tau(k)}}$.
\end{lemma}
\begin{proof}
  The Gauss factors of an element of $\mcG\cap\widetilde{\mcN}_-\widetilde{\mcH}\widetilde{\mcN}_+$ lie in $\mcG$ by uniqueness and \cref{lem: pinning}, and for $y=\mathrm{diag}(t_1,\dots,t_n,t_n^{-1},\dots,t_1^{-1})\in\mcH$ we have $y^{\varpi_k}=t_1\cdots t_{k'}=y^{\varpi^C_{k'}}$.
  This gives the first claim for $u=v=e$, and the general case follows as $u,v$ have the same representatives in $\mcG$ and $\tG$.
  To prove the second claim, write $U=U_-U_0U_+$ with $U_0=\mathrm{diag}(t_1,\dots,t_N)$.
  Then $\vt(U)=\vt(U_-)\vt(U_0)\vt(U_+)$ is the Gauss decomposition of $\vt(U)$, and $\vt(U_0)=\mathrm{diag}(t_N^{-1},\dots,t_1^{-1})$.
  Since $t_1\cdots t_N=1$, we obtain
  \[
    \vt(U_0)^{\varpi_k}=(t_{N-k+1}\cdots t_N)^{-1}=t_1\cdots t_{N-k}=U_0^{\varpi_{\tau(k)}}.
  \]
  This proves the case $u=v=e$. The general case follows from $\vt(\overline{u})=\overline{\tau(u)}$.
\end{proof}

\subsection{BD maps of type \texorpdfstring{$C_n$}{Cn} and their lifts}
We lift a symplectic Belavin--Drinfeld map by reflecting it across the middle of the type A diagram.
Lifting both maps gives a pair that commutes with the involution and has the same aperiodicity condition.

A BD map of type $C_n$ does not involve $\alpha^C_n$: as an isometry, it could only map the unique long simple root to itself, which nilpotency forbids, and similarly for $\gamma_\bullet^{-1}$.
So it is a BD map of type $A_{n-1}$ on $\{\alpha^C_1,\dots,\alpha^C_{n-1}\}$, identified as before with $[1,n-1]$.

\begin{definition}
  \label{def: lift}
  The \newword{lift} of a BD map $\gamma_\bullet:\Gamma_\bullet\to[1,n-1]$ of type $C_n$ is the map $\tg_\bullet$ on $\Gamma_\bullet\sqcup\tau(\Gamma_\bullet)\subseteq[1,2n-1]$ given by $\tg_\bullet(k)=\gamma_\bullet(k)$ and $\tg_\bullet(\tau(k))=\tau(\gamma_\bullet(k))$.
  The lift of a BD pair $\gamma=(\gamma_r,\gamma_c)$ is $\tg:=(\tg_r,\tg_c)$.
\end{definition}

For example, in $Sp_6$ the choice $\gamma_r=\{1\to2\}$ gives $\tg_r=\{1\to2,5\to4\}$.

\begin{proposition}
  \label{prop: lift}
  Let $\gamma_\bullet$ be a BD map of type $C_n$.
  \begin{enumerate}[(a)]
    \item $\tg_\bullet$ is a BD map of type $A_{2n-1}$ commuting with $\tau$.
    \item As endomorphisms of $\widetilde{\mfg}$, $\tg_\bullet\,\Vt=\Vt\,\tg_\bullet$, and $\tg_\bullet|_\mfg=\gamma_\bullet$; in particular $C_{\tg_\bullet}$ commutes with $\pr$ on $\widetilde{\mfn}_+\oplus\widetilde{\mfn}_-$ and restricts to $C_{\gamma_\bullet}$ on $\mfn_+\oplus\mfn_-$.
    \item $\widetilde{\mfh}_{\tg_\bullet}\cap\mfh=\mfh_{\gamma_\bullet}$ and $\widetilde{\mcH}_{\tg_\bullet}\cap\mcG=\mcH_{\gamma_\bullet}$.
  \end{enumerate}
  A BD pair $\gamma$ of type $C_n$ is aperiodic if and only if its lift $\tg$ is.
\end{proposition}
\begin{proof}
  (a) Both $\Gamma_\bullet$ and $\gamma_\bullet(\Gamma_\bullet)$ lie in $[1,n-1]$, while their $\tau$-images lie in $[n+1,2n-1]$.
  Thus the two halves of the domain are not adjacent, nor are their images, so $\tg_\bullet$ is an isometry.
  It is nilpotent since $\tg_\bullet^m(\tau(k))=\tau(\gamma_\bullet^m(k))$.

  (b) The subalgebras $\widetilde{\mfg}_{\Gamma_\bullet}$ and $\widetilde{\mfg}_{\tau(\Gamma_\bullet)}=\Vt(\widetilde{\mfg}_{\Gamma_\bullet})$ commute, $\Vt\tg_\bullet\Vt$ and $\tg_\bullet$ agree on their Chevalley generators by \cref{lem: pinning}, and the orthogonal projection onto their sum commutes with the isometry $\Vt$.
  So $\tg_\bullet\Vt=\Vt\tg_\bullet$.
  The fixed points of $\Vt$ in $\widetilde{\mfg}_{\Gamma_\bullet}\oplus\widetilde{\mfg}_{\tau(\Gamma_\bullet)}$ form the subalgebra of $\mfg$ generated by the $E^C_k,F^C_k$, $k\in\Gamma_\bullet$, on which $\tg_\bullet$ acts by $E^C_k\mapsto E^C_{\gamma_\bullet(k)}$, $F^C_k\mapsto F^C_{\gamma_\bullet(k)}$.
  The orthogonal projection of $\mfg$ onto it is the restriction of the projection of $\widetilde{\mfg}$ onto $\widetilde{\mfg}_{\Gamma_\bullet}\oplus\widetilde{\mfg}_{\tau(\Gamma_\bullet)}$; hence $\tg_\bullet|_\mfg=\gamma_\bullet$.

  (c) On $\mfh$ the conditions $\alpha(\zeta)=\tg_\bullet(\alpha)(\zeta)$ for $\alpha=\alpha_k$ and $\alpha=\alpha_{\tau(k)}$ both read $\alpha^C_k(\zeta)=\gamma_\bullet(\alpha^C_k)(\zeta)$.
  For the tori, write $y=\mathrm{diag}(t_1,\dots,t_n,t_n^{-1},\dots,t_1^{-1})\in\mcH$ and use the coordinates $q_i=t_i/t_{i+1}$ for $i<n$, $q_n=t_n$, with inverse $t_i=q_i\cdots q_n$.
  The character equations for $\gamma_\bullet$, and those for $\tg_\bullet$ restricted to $\mcH$, both become $q_i=q_{\gamma_\bullet(i)}$ for $i\in\Gamma_\bullet$.
  Their full solution subgroup in $\mcH$ is connected, since these equations merely identify coordinates, and hence equals $\mcH_{\gamma_\bullet}$.
  By connectedness it lies in the identity component $\widetilde{\mcH}_{\tg_\bullet}$ of the ambient solution subgroup. Conversely, every element of $\widetilde{\mcH}_{\tg_\bullet}\cap\mcG$ satisfies these equations.
  This proves the torus equality.

  Finally, $-w_0$ acts as $\tau$ on the simple roots of $\SL$ and trivially on those of $\mcG$.
  Hence the map $\alpha_k,\alpha_{\tau(k)}\mapsto\alpha^C_k$ intertwines the alternating maps of $\tg$ and of $\gamma$, and the domain of the former is the preimage of the domain of the latter.
  A periodic point of the former maps to a periodic point of the latter. Conversely, if $x$ is periodic for the latter, the former maps the finite preimage of the orbit of $x$ injectively, hence bijectively, into itself, so it has periodic points.
\end{proof}

\begin{remark}
  \label{rmk: aperiodic not PL}
  A nontrivial aperiodic BD pair of type $C_n$ does not give a Poisson--Lie bracket on $\Sp$.
  Indeed, for a Poisson--Lie bracket every $\{f,g\}$ vanishes at the identity, whereas at the identity $\{\cdot,\cdot\}_{\gamma,s}$ is given by $r_\cc-r_\rr$, whose non-Cartan part vanishes only if $\gamma_c=\gamma_r^{-1}$.
  Since $-w_0$ acts trivially on the simple roots of type $C_n$, every $\alpha\in\Gamma_r$ then gives the alternating two-cycle $\alpha^\rr\mapsto\gamma_r(\alpha)^\cc\mapsto\alpha^\rr$.
  Thus an aperiodic pair can give a Poisson--Lie bracket only if $\gamma_r=\gamma_c=0$.
\end{remark}

\subsection{Cartan data and restriction of brackets}
Let $\mfk$ be the $(-1)$-eigenspace of $\Vt$ on $\widetilde{\mfh}$, so $\widetilde{\mfh}=\mfh\oplus\mfk$ is an orthogonal decomposition.
For a $\Vt$-invariant $\widetilde{s}\in\widetilde{\mfh}\wedge\widetilde{\mfh}$, put $s:=(\pr\otimes\pr)\widetilde{s}\in\mfh\wedge\mfh$; then $\widetilde{s}-s\in\mfk\wedge\mfk$.

\begin{proposition}
  \label{prop: cartan}
  For a BD map $\gamma_\bullet$ of type $C_n$, the map $\widetilde{s}\mapsto s$ sends the $\Vt$-invariant elements of $\mcS(\tg_\bullet)$ onto $\mcS(\gamma_\bullet)$.
\end{proposition}
\begin{proof}

  For a $\Vt$-invariant $\widetilde{s}$ and $\lambda\in\widetilde{\mfh}^*$, the $\mfh$-component of $2(\lambda\otimes1)\widetilde{s}$ is $2(\lambda|_\mfh\otimes1)s$.
  For $\alpha=\alpha_k$ or $\alpha_{\tau(k)}$, $k\in\Gamma_\bullet$, we have $(\alpha-\tg_\bullet(\alpha))|_\mfh=\alpha^C_k-\gamma_\bullet(\alpha^C_k)$ and $\pr\big(((\alpha+\tg_\bullet(\alpha))\otimes1)t_0\big)=\big((\alpha^C_k+\gamma_\bullet(\alpha^C_k))\otimes1\big)t^\mfg_0$, as $\widetilde{\mfh}=\mfh\oplus\mfk$ is orthogonal, $t^\mfg_0$ being the Cartan part of the Casimir element of $\mfg$ for the restricted form.
  So the $\mfh$-components of the equations \eqref{eq: system s} for $\tg_\bullet$ are the equations for $\gamma_\bullet$, and the $\mfk$-components only involve $\widetilde{s}-s$; hence $s\in\mcS(\gamma_\bullet)$.
  Conversely, $\mcS(\tg_\bullet)$ is a non-empty affine space stable under $\Vt\otimes\Vt$, so it contains the $\Vt$-invariant element $\widetilde{s}^0:=\frac12(\widetilde{s}+(\Vt\otimes\Vt)\widetilde{s})$ for any $\widetilde{s}\in\mcS(\tg_\bullet)$.
  Replacing its $\mfh\wedge\mfh$ component by any $s\in\mcS(\gamma_\bullet)$ gives the invariant solution $\widetilde{s}^0+s-(\pr\otimes\pr)\widetilde{s}^0$, which projects to $s$.
\end{proof}

For a BD pair $\gamma$ of type $C_n$ we always take $\widetilde{s}=(\widetilde{s}_r,\widetilde{s}_c)\in\mcS(\tg)$ with $\Vt$-invariant components, and $s=(s_r,s_c)\in\mcS(\gamma)$ with $s_\bullet:=(\pr\otimes\pr)\widetilde{s}_\bullet$. Every $s\in\mcS(\gamma)$ arises in this way.

\begin{proposition}
  \label{prop: restriction}
  \begin{enumerate}[(a)]
    \item $\vt$ is a Poisson automorphism of $(\tG,\{\cdot,\cdot\}_{\tg,\widetilde{s}})$.
    \item For $\vt$-invariant $f,g\in\bbC[\tG]$, $\{f,g\}_{\tg,\widetilde{s}}|_\mcG=\{f|_\mcG,g|_\mcG\}_{\gamma,s}$.
  \end{enumerate}
\end{proposition}
\begin{proof}
  (a) Since $\partial^\rr_\xi(f\circ\vt)=(\partial^\rr_{\Vt\xi}f)\circ\vt$, and similarly for $\partial^\cc$, it suffices that $\Vt\otimes\Vt$ fixes $r_\rr$ and $r_\cc$.
  This holds because $\Vt$ preserves the trace form, hence $t$, preserves $\widetilde{\mfn}_\pm$, hence $t_{-+}$ and $t_{+-}$, commutes with $C_{\tg_r}$, $C_{\tg_c}$ and with the transpose, and fixes $\widetilde{s}_r,\widetilde{s}_c$.

  (b) Let $U\in\mcG$.
  For $\xi\in\widetilde{\mfg}$ we have $\partial^\rr_\xi f(U)=\partial^\rr_{\Vt\xi}f(U)$ because $f=f\circ\vt$ and $\vt(U)=U$, hence $\partial^\rr_\xi f(U)=\partial^\rr_{\pr\xi}f(U)$.
  For $\xi\in\mfg$ the curve $\exp(t\xi)U$ lies in $\mcG$, hence $\partial^\rr_\xi f(U)=\partial^\rr_\xi(f|_\mcG)(U)$.
  The same holds for $\partial^\cc$.
  For any tensor $z=\sum_m\xi_m\otimes\eta_m\in\widetilde{\mfg}\otimes\widetilde{\mfg}$, we therefore have
  \[
    z(f,g)^\rr(U)=\sum_m\partial^\rr_{\pr\xi_m}(f|_\mcG)(U)\,\partial^\rr_{\pr\eta_m}(g|_\mcG)(U)=((\pr\otimes\pr)z)(f|_\mcG,g|_\mcG)^\rr(U),
  \]
  and the same holds for column derivatives, with $\cc$ in place of $\rr$.
  Thus it suffices to show that $(\pr\otimes\pr)r_{\tg_\bullet,\widetilde{s}_\bullet}=r^\mfg_{\gamma_\bullet,s_\bullet}$, the $R$-matrix of $\mfg$, for each of the two BD maps (the transpose commutes with $\pr$).
  By orthogonality the Casimir element of $\widetilde{\mfg}$ is $t^\mfg+t^\mfp$, so $(\pr\otimes\pr)t=t^\mfg$, and $\pr$ maps the components $t_0,t_{-+},t_{+-}$ to those of $t^\mfg$, as it preserves $\widetilde{\mfh}$ and $\widetilde{\mfn}_\pm$.
  By \cref{prop: lift}(b), $(\pr\otimes\pr)(1\otimes C_{\tg_\bullet})t_{-+}=(1\otimes C_{\gamma_\bullet})t^\mfg_{-+}$, and similarly for $t_{+-}$; finally $(\pr\otimes\pr)\widetilde{s}_\bullet=s_\bullet$.
\end{proof}

\begin{remark}
  \label{rmk: not Poisson subvariety}
  $\mcG$ is not a Poisson subvariety of $(\tG,\{\cdot,\cdot\}_{\tg,\widetilde{s}})$, already for $\gamma=(0,0)$.
  For $n=2$ and $\widetilde{s}=0$, write $U=(x_{ij})$ and
  \[
    q:=(UJU^T-J)_{13}=x_{11}x_{34}-x_{12}x_{33}+x_{13}x_{32}-x_{14}x_{31}.
  \]
  Then $q|_{Sp_4}=0$, whereas $\{x_{11},q\}_{(0,0),0}(I+E_{12}+E_{34})=-\tfrac12$, with $I+E_{12}+E_{34}\in Sp_4$, so the defining ideal of $Sp_4$ is not a Poisson ideal.
  This failure is independent of the chosen compatible pinnings. Indeed, changing the pinning transports $\vt$, its fixed subgroup and the associated Poisson bracket by the same automorphism of $\tG$.
  \Cref{prop: restriction} says instead that the restriction $\bbC[\tG]^\vt\to\bbC[\mcG]$, which is surjective, is a Poisson morphism.
\end{remark}

\subsection{Lifted words}
We lift symplectic words by replacing each short-root letter with its commuting pair in type A.
To apply the ambient word-seed construction, we must check that the lifted word is reduced.

A double reduced word of type $C_n$ is a shuffle $\bfi$ of a reduced word for $w_0\in W^\tau$ in the alphabet $[-n,-1]$ and one in the alphabet $[1,n]$.
Two such words are related by commutations, sign swaps, braid moves of length $3$, $(k,k+1,k)\leftrightarrow(k+1,k,k+1)$ with $k+1<n$, and of length $4$, $(n-1,n,n-1,n)\leftrightarrow(n,n-1,n,n-1)$.

\begin{definition}
  \label{def: lifted word}
  The \newword{lift} $\widetilde{\boldsymbol{\jmath}}$ of an arbitrary word $\bfj$ in $\pm[1,n]$ is obtained by replacing each letter $\pm k$, $k<n$, by the two letters $\pm k,\pm\tau(k)$, and keeping each $\pm n$.
\end{definition}

\begin{lemma}
  \label{lem: lifted word}
  If $\bfi$ is a double reduced word of type $C_n$, its lift $\tbfi$ is a double reduced word for $(w_0,w_0)$ in $\SL$, and replacing every letter $\pm k$ of $\tbfi$ by $\pm\tau(k)$ gives a word related to $\tbfi$ by commutations.
\end{lemma}
\begin{proof}
  Braid moves preserve the number of occurrences of the long-root reflection $s_n^C$.
  Comparing with the reduced expression $(s_1^C\cdots s_n^C)^n$, every reduced expression for $w_0$ in $W^\tau$ therefore has $n$ long-root factors and $n^2-n$ short-root factors.
  Lifting keeps each long-root factor and replaces each short-root factor by two simple reflections.
  Since $s^C_k=s_ks_{\tau(k)}$ in $S_{2n}$, the lifted expression still has product $w_0$, and its length is $n+2(n^2-n)=\binom{2n}{2}$, so it is reduced.
  Applying this to the positive and negative subwords proves the first claim.
  For the second, $\tau$ fixes the letters $\pm n$ and swaps the two letters in each lifted pair; these commute because $k$ and $\tau(k)$ are not adjacent for $k<n$.
\end{proof}

\subsection{The map \texorpdfstring{$h_\gamma$}{h}}
Let $\gamma$ be an aperiodic BD pair of type $C_n$ and $\tg$ its lift.

\begin{proposition}
  \label{prop: h theta}
  $h_{\tg}\circ\vt=\vt\circ h_{\tg}$, and $h_{\tg}$ is defined on a dense open subset of $\mcG$, which it maps into $\mcG$; on $\mcG$ it agrees with the map $h_\gamma$ of \eqref{eq: h} for $\mcG$ and $\gamma$.
\end{proposition}
\begin{proof}
  We use the notation from the definition of $h_{\tg}$ in \cref{sec: h}; in particular, $\sigma_r=w_0^{\widetilde{\Gamma}_r}w_0$ and $\sigma_c=w_0^{\widetilde{\Gamma}_c}w_0$.
  The sets $\widetilde{\Gamma}_r=\Gamma_r\sqcup\tau(\Gamma_r)$ and $\widetilde{\Gamma}_c=\Gamma_c\sqcup\tau(\Gamma_c)$ are $\tau$-stable, so $w_0^{\widetilde{\Gamma}_r}$, $w_0^{\widetilde{\Gamma}_c}$, $\sigma_r$, $\sigma_c$ lie in $W^\tau$ and their representatives are fixed by $\vt$. Moreover, $\vt$ preserves $\widetilde{\mcN}_\pm^{\widetilde{\Gamma}_r}$, $\widetilde{\mcN}_\pm^{\widetilde{\Gamma}_c}$ and the cells $\widetilde{\mcB}_\pm\overline{w}\widetilde{\mcB}_\pm$ for $w\in W^\tau$, and it commutes with $\boldsymbol\gamma_r,\boldsymbol\gamma_c$ by \cref{prop: lift}(b).
  By uniqueness of the factorizations in \eqref{eq: h}, $V_r(\vt(U))=\vt(V_r(U))$, $\overline{V}_r(\vt(U))=\vt(\overline{V}_r(U))$, and similarly for columns; so $h_{\tg}\circ\vt=\vt\circ h_{\tg}$.
  In particular $h_{\tg}$ maps the fixed points of $\vt$ in its domain into $\mcG$.
  It remains to show that the domains of $H_r$ and $H_c$ each meet $\mcG$. Their intersections with $\mcG$ are then nonempty open subsets, and irreducibility of $\mcG$ ensures that their intersection is nonempty and dense, with $h_{\tg}$ defined there by \eqref{eq: h}.
  By \cref{lem: h transpose}, as the transpose preserves $\mcG$, it suffices to treat $H_r$; write $K=\widetilde{\Gamma}_r$, which is $\tau$-stable.

  Put $y_k(t):=\exp(tF_k)$, $y^C_k:=y_k y_{\tau(k)}$ for $k<n$, and $y^C_n:=y_n$.
  Choose a reduced word $(j_1,\dots,j_\ell)$ for $w_0$ in $W^\tau$ whose first $\ell_K$ letters form a reduced word for $w_0^K\in W^\tau$, $\ell_K$ being the length of $w_0^K$ in $W^\tau$.
  As in \cref{lem: lifted word}, its lift is a reduced word $\widetilde{\boldsymbol{\jmath}}$ for $w_0$ in $S_{2n}$ beginning with one for $w_0^K$.
  By \cite[Proposition~2.18]{FZ99} (transposed), the map $(t_m)\mapsto\prod_m y_{\widetilde{\jmath}_m}(t_m)$ is an isomorphism from a torus onto an open subset $\mathcal{U}_-$ of $\widetilde{\mcN}_-\cap\widetilde{\mcB}_+\overline{w_0}\widetilde{\mcB}_+$, hence of $\widetilde{\mcN}_-$.
  For $U=U_-U_0U_+$ in the big cell $\widetilde{\mcN}_-\widetilde{\mcH}\widetilde{\mcN}_+$, let $U_-$ denote its negative Gauss factor.
  On the open set $\mathcal{U}$ of those $U$ with $U_-\in\mathcal{U}_-$, write $U_-=\prod_m y_{\widetilde{\jmath}_m}(t_m)$ and let $V(U)$ be the product of the factors coming from the letters of $\widetilde{\boldsymbol{\jmath}}$ lifted from $j_1,\dots,j_{\ell_K}$.
  The $t_m$ are regular functions on $\mathcal{U}$, as $U\mapsto U_-$ is regular on the big cell, so this defines a morphism $V\colon\mathcal{U}\to\widetilde{\mcN}_-^K$; moreover the product of the remaining factors lies in $\widetilde{\mcB}_+\overline{w_0^Kw_0}\widetilde{\mcB}_+$, since $y_k(t)\in\widetilde{\mcB}_+\overline{s_k}\widetilde{\mcB}_+$ for $t\neq0$ and Bruhat cells multiply along reduced words.
  Similarly $V(U)\in\widetilde{\mcB}_+^K\overline{w_0^K}\widetilde{\mcB}_+^K=\widetilde{\mcN}_+^K\overline{w_0^K}^{\,-1}\widetilde{\mcB}_+^K$, so $V(U)\overline{w_0^K}$ lies in the big cell $\widetilde{\mcN}_+^K\widetilde{\mcB}_-^K$ of $\tG^K$, and $\overline{V}(U):=\lfloor V(U)\overline{w_0^K}\rfloor_+$ defines a morphism $\overline{V}$ on $\mathcal{U}$, as Gauss factors are regular on the big cell.
  Finally $\mathcal{U}$ contains the elements $y^C_{j_1}(t_1)\cdots y^C_{j_\ell}(t_\ell)$ of $\mcG$ with $t_m\neq0$.
  By the defining factorizations, $V=V_r$ and $\overline{V}=\overline{V}_r$, so $H_r$ is a morphism on $\mathcal{U}$.

  On the common domain constructed above, let $U\in\mcG$.
  The factors $V_r(U)$, $M_r=V_r(U)^{-1}U$ and $\overline{V}_r(U)$ lie in $\mcG$, being fixed by $\vt$.
  Moreover,
  \[
    \widetilde{\mcN}_\pm^{\widetilde{\Gamma}_r}\cap\mcG=\mcN_\pm^{\Gamma_r}:
  \]
  both sides are connected unipotent subgroups.
  The left one is the image under the $\vt$-equivariant isomorphism $\exp$ of the $\Vt$-fixed part of $\widetilde{\mfn}_\pm^{\widetilde{\Gamma}_r}$.
  Both have the same Lie algebra $\mfn_\pm^{\Gamma_r}$, by the fixed-point calculation in the proof of \cref{prop: lift}(b).
  Thus $V_r(U)\in\mcN_-^{\Gamma_r}$ and $\overline{V}_r(U)\in\mcN_+^{\Gamma_r}$.
  Also $M_r\in\mcB_+\overline{\sigma_r}\mcB_+$, since
  \[
    \widetilde{\mcB}_+\overline{\sigma_r}\widetilde{\mcB}_+\cap\mcG
    =\mcB_+\overline{\sigma_r}\mcB_+,
  \]
  by the Bruhat decompositions of $\tG$ and $\mcG$ and the inclusions $\mcB_+\overline{w}\mcB_+\subseteq\widetilde{\mcB}_+\overline{w}\widetilde{\mcB}_+$ for $w\in W^\tau$.
  The representative of $w_0^{\widetilde{\Gamma}_r}$ is that of the longest element $w_0^{\Gamma_r}$ of $W_{\Gamma_r}$.
  Uniqueness of the factorizations therefore identifies $V_r(U)$ and $\overline{V}_r(U)$ with the factors defining $h_\gamma$ in \eqref{eq: h}. The same argument applies to columns.
  Finally, $\boldsymbol\gamma_r,\boldsymbol\gamma_c$ restrict to the corresponding homomorphisms for $\mcG$ by \cref{prop: lift}(b), so $h_{\tg}(U)=h_\gamma(U)$.
\end{proof}

\section{Folding seeds}
\label{sec: folding}
In this section $Y$ is an irreducible affine variety with an involutive automorphism $\vt$, and $X\subseteq Y$ is an irreducible closed subvariety fixed pointwise by $\vt$.
  We write $\overline{f}:=f|_X$.
In our application $Y=\SL$ and $X=\Sp$.
We adapt Dupont's folding construction \cite{Dup08} to seeds of functions.
For the definitions concerning matrices, we use the restatement in \cite[Section~5.1]{NS14}, whose notation is close to ours; see also \cite[Section~2.3]{Vie18} for a further account.

\begin{remark}
  \label{rmk: folding generality}
  One can show that the results below extend to skew-symmetrizable principal parts and to automorphisms of finite order.
  For the latter, mutations within an orbit need not commute.
  One must require that the entries between indices of each mutated orbit vanish.
  One must also impose admissibility by requiring that, in each row, the entries in the columns of that orbit agree weakly in sign.
  The Poisson and toric arguments use the same orbit-sum formulas.
  We retain skew-symmetric principal parts and involutions, which are simpler to state and suffice for our application.
\end{remark}

\begin{definition}
  \label{def: symmetric seed}
  Let $\Sigma=(\bx,\tB)$ be a seed in $\bbC(Y)$, with index set $P=P_\ex\sqcup P_\fr$ and skew-symmetric principal part.
  The seed $\Sigma$ is \newword{$\vt$-symmetric} if there is an involution $\tau$ of $P$ such that
  \begin{enumerate}[(S1)]
    \item $\tau$ preserves $P_\ex$ and $P_\fr$, and $b_{\tau(p)\tau(q)}=b_{pq}$ for every entry $b_{pq}$ of $\tB$;
    \item $x_{\tau(p)}=x_p\circ\vt$ for all $p\in P$.
  \end{enumerate}
\end{definition}

The entries of $\bx$ are algebraically independent, hence pairwise distinct, so (S2) determines $\tau$: it is the permutation of $P$ through which $f\mapsto f\circ\vt$ acts on $\bx$.
We call it the \newword{involution} of $\Sigma$, and its orbits the \newword{orbits} of $\Sigma$.
For $p\in P_\ex$, (S1) and skew-symmetry give $b_{p\tau(p)}=b_{\tau(p)p}=-b_{p\tau(p)}$, so $b_{p\tau(p)}=0$.
Thus the mutations within any exchangeable orbit $O$ commute, and the \newword{orbit mutation} $\mu_O:=\prod_{p\in O}\mu_p$ is well defined.

\begin{definition}
  \label{def: admissible}
  Let $\Sigma$ be a $\vt$-symmetric seed with involution $\tau$. We say that $\tau$ is \newword{admissible at} the exchangeable orbit $O$ if, for every orbit $R$, the entries of the $R\times O$ block of $\tB$ agree weakly in sign.
\end{definition}

This is the sign condition of \cite[(5.2)]{NS14} and \cite[Definition~2.3.2(ii)]{Vie18}, imposed only on the columns of $O$ but including the frozen rows.
By (S1), for $q\in R$ and $p\in O$ the $R\times O$ block consists of the entries $b_{qp}$ and $b_{q\tau(p)}$, so $\tau$ is admissible at $O$ if and only if $b_{qp}b_{q\tau(p)}\ge0$ for all $q\in P$; in particular it is admissible at every orbit of size one.
Admissibility of $\tau$ at an exchangeable orbit $R\neq O$ need not be preserved by $\mu_O$ \cite[Remark~2.18]{Dup08}, that is, admissible automorphisms need not be stable \cite[Definition~2.3.4]{Vie18}. We do not need stability: we apply \cref{prop: fold mutation} only to seeds whose involution is checked directly to be admissible at the mutated orbit.

\begin{definition}
  \label{def: fold}
  The \newword{fold} of a $\vt$-symmetric seed $\Sigma$ whose extended cluster consists of regular functions on $Y$ is $\overline{\Sigma}=(\overline{\bx},\overline{B})$, with $\overline{\bx}=(\overline{x}_O)_{O\in P/\tau}$, $\overline{x}_O:=\overline{x}_p$, and $\overline{B}=(\overline{b}_{RO})_{R\in P/\tau,O\in P_\ex/\tau}$, $\overline{b}_{RO}:=\sum_{q\in R}b_{qp}$, for any $p\in O$.
\end{definition}

These definitions do not depend on $p\in O$ by (S1) and (S2), as $x_p\circ\vt$ and $x_p$ agree on $X$.
The fold is a seed in $\bbC(X)$ precisely when the functions $\overline{x}_O$ are algebraically independent and generate $\bbC(X)$. These properties do not follow from symmetry alone.
Since $B$ is skew-symmetric, the principal part of $\overline{B}$ is skew-symmetrizable, $|O|\overline{b}_{RO}=-|R|\overline{b}_{OR}$; it is the folded matrix of \cite[(5.5)]{NS14}.

\begin{proposition}
  \label{prop: fold mutation}
  Let $\Sigma$ be a $\vt$-symmetric seed, belonging to a regular cluster structure on $Y$, whose fold is a seed in $\bbC(X)$, and let $O$ be an exchangeable orbit.
  Then $\mu_O(\Sigma)$ is $\vt$-symmetric.
  If $\tau$ is admissible at $O$, then it is also admissible at $O$ for $\mu_O(\Sigma)$, and the extended cluster and the matrix of the fold of $\mu_O(\Sigma)$ are obtained from those of $\overline{\Sigma}$ by the mutation in direction $O$.
\end{proposition}
\begin{proof}
  Choose $p\in O$. Since $b_{p\tau(p)}=0$, the mutations at the distinct indices in $O$ do not change each other's rows and columns, so $\mu_O(\tB)$ satisfies (S1).
  The new variables at $p$ and $\tau(p)$ are given by exchange relations swapped by $\tau$ and $\vt$, so (S2) holds. The new variables are regular since the cluster structure is, so the fold of $\mu_O(\Sigma)$ is defined.
  Admissibility at $O$ is preserved since the entries $b_{qp}$, $p\in O$, only change sign.
  If $\tau$ is admissible at $O$ and $p\in O$, then $\sum_{q\in R}[\pm b_{qp}]_+=[\pm\overline{b}_{RO}]_+$ for every orbit $R$, so the restriction to $X$ of the exchange relation at $p$ is the exchange relation of $\overline{\Sigma}$ in direction $O$.
  For the matrix statement, let $R,S\neq O$, with $S$ exchangeable, and choose $s\in S$.
  Admissibility at $O$ and (S1) give
  \[
    \sum_{r\in R,\,p\in O}[b_{rp}]_+[b_{ps}]_+
      =[\overline{b}_{RO}]_+[\overline{b}_{OS}]_+,
  \]
  and the same identity with both entries negated.
  Summing the ordinary matrix-mutation formula therefore gives the mutation formula for $\overline{B}$. Entries with row or column orbit $O$ simply change sign.
  This uses only admissibility at $O$, rather than at every orbit; compare \cite[proof of Theorem~2.24]{Dup08}.
\end{proof}

\begin{proposition}
  \label{prop: fold compatible}
  Let $\{\cdot,\cdot\}$ be a Poisson bracket on $\bbC(Y)$ for which $\vt$ is a Poisson automorphism, and $\{\cdot,\cdot\}_X$ a Poisson bracket on $\bbC(X)$ with $\{f,g\}|_X=\{\overline{f},\overline{g}\}_X$ for all $\vt$-invariant $f,g\in\bbC[Y]$.
  Let $\Sigma$ be a $\vt$-symmetric seed whose extended cluster consists of regular functions on $Y$, such that $\overline{\bx}$ is algebraically independent and generates $\bbC(X)$, and suppose that $\{\cdot,\cdot\}$ is log-canonical with respect to $\bx$, with coefficient matrix $\Omega$ satisfying \eqref{eq: compatibility} with all $\kappa_p\neq0$.
  Then $\{\cdot,\cdot\}_X$ is log-canonical with respect to $\overline{\bx}$, with coefficients $\overline{\omega}_{OR}=\frac1{|R|}\sum_{q\in R}\omega_{pq}$, $p\in O$, and these satisfy \eqref{eq: compatibility} with $\overline{\kappa}_O=\kappa_p/|O|$, $p\in O$.
  In particular $\overline{B}$ has full rank, and $\{\cdot,\cdot\}_X$ is compatible with the cluster structure defined by $\overline{\Sigma}$.
\end{proposition}
\begin{proof}

  As $\vt$ is Poisson, $\omega_{\tau(p)\tau(q)}=\omega_{pq}$ and $\kappa_{\tau(p)}=\kappa_p$, so $\overline{\omega}_{OR}$ does not depend on $p\in O$.
  The functions $f_O:=\prod_{p\in O}x_p$ are regular (\cref{def: fold}) and $\vt$-invariant, with $\overline{f}_O=\overline{x}_O^{|O|}$ and $\{f_O,f_R\}=\big(\sum_{p\in O,q\in R}\omega_{pq}\big)f_Of_R$; restricting to $X$ gives $\{\overline{x}_O,\overline{x}_R\}_X=\overline{\omega}_{OR}\overline{x}_O\overline{x}_R$.
  For $p\in O$ exchangeable and any orbit $S$,
  \[
    \sum_R\overline{b}_{RO}\overline{\omega}_{RS}=\frac1{|S|}\sum_{q\in S}\sum_{r\in P}b_{rp}\omega_{rq}=\frac1{|S|}\sum_{q\in S}\kappa_p\delta_{pq},
  \]
  which is $\kappa_p/|O|$ if $S=O$ and $0$ otherwise.
  As all $\overline{\kappa}_O$ are nonzero, full rank and compatibility follow from \cite[Proposition~2.5]{GSV18dd}, see \cref{sec: cluster conventions}.
\end{proof}

\begin{proposition}
  \label{prop: fold toric}
  Let $\Sigma$ be a $\vt$-symmetric seed whose fold is a seed, and $\mathsf T$ a torus acting on $Y$ such that every $x_p$ is $\mathsf T$-homogeneous of weight $w_p$, with $\tB^TW=0$ for $W=(w_p)_p$.
  Let $\mathsf T_X\subseteq\mathsf T$ be a subtorus preserving $X$ and commuting with $\vt$.
  Then $\overline{x}_O$ is $\mathsf T_X$-homogeneous of weight $\overline{w}_O:=w_p|_{\mathsf T_X}$, $p\in O$, and $\overline{B}^T\overline{W}=0$ for $\overline{W}=(\overline{w}_O)_O$; if the $w_p$ span the character space of $\mathsf T$ over $\bbC$, the $w_p|_{\mathsf T_X}$ span that of $\mathsf T_X$.
\end{proposition}
\begin{proof}
  Homogeneity is immediate. We have $w_{\tau(p)}|_{\mathsf T_X}=w_p|_{\mathsf T_X}$ as $x_{\tau(p)}=x_p\circ\vt$; then, for $p\in O$, $\sum_R\overline{b}_{RO}\overline{w}_R=\big(\sum_{q\in P}b_{qp}w_q\big)|_{\mathsf T_X}=0$, and restriction of characters is surjective.
\end{proof}

\section{Symmetric BD word seeds and their folds}
\label{sec: symmetric}
From now on $\gamma$ is an aperiodic BD pair of type $C_n$, and $\tg$ is its lift.
Let $\bfi$ be a double reduced word of type $C_n$.
The lift is order preserving: the letters lifted from $i_j$ precede those lifted from $i_{j'}$ if $j<j'$.
  The letters at the level $x$ of $\tbfi$ are those lifted from the letters $\pm\min(x,\tau(x))$ of $\bfi$, with the same signs.
For a slot $s$ of $\bfi$ let $\hat s$ be the slot of $\tbfi$ just after the letters lifted from the letters at the positions $\le s$.
Every chamber of $\tbfi$ contains such a slot, and the chamber $c_x(\hat s)$ of $\tbfi$ only depends on the chamber $c_{x'}(s)$ of $\bfi$, $x'=\min(x,\tau(x))$. We say that it \newword{lies over} $c_{x'}(s)$.
We also write $\tau$ for the induced involution of $C(\tbfi)$, given by $\tau(c_x(\hat s))=c_{\tau(x)}(\hat s)$.
  Its fixed points are the chambers at the level $n$, and we call the chambers at the levels $<n$ (resp.\ $>n$) \newword{lower} (resp.\ \newword{upper}).
So the $\tau$-orbits of $C(\tbfi)$ are the sets of chambers lying over the chambers of $\bfi$, and exchangeable chambers and end chambers lie over exchangeable chambers and end chambers.
The same holds for the lift of an arbitrary word in $\pm[1,n]$.
For $k\in C(\bfi)$ we write $O_k$ for the orbit of the chambers of $\tbfi$ lying over $k$, so that $k\mapsto O_k$ is a bijection $C(\bfi)\to C(\tbfi)/\tau$.
We write $\widetilde P:=C(\tbfi)$, with $\widetilde P_\ex$ and $\widetilde P_\fr$ as in \cref{def: model} for $\tbfi$ and $\tg$.

\subsection{Symmetry}
For a BD pair $\gamma'$ of type $A_{2n-1}$ let $\tau(\gamma'):=(\tau\gamma'_r\tau,\tau\gamma'_c\tau)$, and for a word $\bfj$ let $\tau(\bfj)$ be obtained by replacing every letter $\pm k$ by $\pm\tau(k)$.
  We identify $C(\bfj)$ with $C(\tau(\bfj))$ by mapping $c_x(s)$ to $c_{\tau(x)}(s)$.

\begin{lemma}
  \label{lem: naturality}
  Let $\gamma'$ be an aperiodic BD pair of type $A_{2n-1}$ and $\bfj$ a double reduced word for $\SL$.
  With this identification, $\tB_{\tau(\gamma')}(\tau(\bfj))=\tB_{\gamma'}(\bfj)$ and $\Phi_{\tau(\gamma')}(k;\tau(\bfj))=\Phi_{\gamma'}(k;\bfj)\circ\vt$ for every $k$.
\end{lemma}
\begin{proof}
  The elements $u_{\le s}$, $v_{>s}$ for $\tau(\bfj)$ are the images under $\tau$ of those for $\bfj$.
  Hence the weights of a chamber for $\tau(\bfj)$ are the images under $\tau=-w_0$ of its weights for $\bfj$, and $\Delta(k;\tau(\bfj))=\Delta(k;\bfj)\circ\vt$ by \cref{lem: minors restrict}.
  As $\tau(u)(\alpha_{\tau(x)})=\tau(u\alpha_x)$, the inversion letter of the root $\tau(a)$ of $\tau(\gamma')$ in $\tau(\bfj)$ is at the position of that of $a$ in $\bfj$.
  The thawed chambers, which are at the levels $\tau(a)^*=\tau(a^*)$, the extended words and the dipole matrices correspond.
  As $\tau$ preserves the pairing, the exponents correspond.
  Finally, \eqref{eq: B entries} only depends on the levels through the Cartan entries, and $a_{\tau(x)\tau(y)}=a_{xy}$.
\end{proof}

\begin{theorem}
  \label{thm: symmetric}
  The BD word seed $\Sigma_{\tg}(\tbfi)$ is a $\vt$-symmetric seed of $\mcC_{\tg}$ with involution $\tau$, and its extended cluster consists of regular functions on $\SL$.
\end{theorem}
\begin{proof}
  By \cref{thm: model concise,thm: GSV}, $\Sigma_{\tg}(\tbfi)$ is a seed of the regular cluster structure $\mcC_{\tg}$.
  By \cref{lem: lifted word}, $\tau(\tbfi)$ is obtained from $\tbfi$ by commutations swapping the two letters of each lifted pair, and $\tau(\tg)=\tg$ by \cref{prop: lift}(a).
  By \cref{thm: moves}(a), these commutations preserve the exchange matrix and the functions $\Phi_{\tg}$ attached to corresponding chambers.
  The correspondence maps the chamber of $\tau(\tbfi)$ at the level $x$ containing a slot $\hat s$ to the chamber of $\tbfi$ at the level $x$ containing $\hat s$, as the slots $\hat s$ lie outside the commuted pairs.
  Composing with \cref{lem: naturality}, which maps $c_x(\hat s)$ of $\tbfi$ to $c_{\tau(x)}(\hat s)$ of $\tau(\tbfi)$, we get $b_{\tau(\widetilde{k})\tau(\widetilde{\ell})}=b_{\widetilde{k}\widetilde{\ell}}$ and $\Phi_{\tg}(\tau(\widetilde{k});\tbfi)=\Phi_{\tg}(\widetilde{k};\tbfi)\circ\vt$. The involution $\tau$ preserves $\widetilde P_\ex$ and $\widetilde P_\fr$, as it maps thawed chambers to thawed chambers.
  So (S1) holds, and (S2) follows from \cref{prop: h theta}: $x_{\tg}(\tau(\widetilde{k}))=\Phi_{\tg}(\widetilde{k};\tbfi)\circ\vt\circ h_{\tg}^{-1}=x_{\tg}(\widetilde{k})\circ\vt$.
\end{proof}

\subsection{The fold}
We compute the fold of $\Sigma_{\tg}(\tbfi)$ in two steps: \cref{lem: index minors,lem: fold BFZ} treat the chamber minors and the matrices of ordinary words, and \cref{thm: fold model} adds the dipoles and the exponents.

\begin{lemma}
  \label{lem: index minors}
  If $\widetilde{k}\in C(\tbfi)$ lies over $k\in C(\bfi)$, then $\Delta(\widetilde{k};\tbfi)|_\mcG=\Delta(k;\bfi)$.
  In particular the functions $\Delta(\widetilde{k};\tbfi)|_\mcG$, one for each orbit, are algebraically independent and generate $\bbC(\mcG)$.
\end{lemma}
\begin{proof}
  Let $\widetilde{k}=c_x(\hat s)$ and $k=c_{x'}(s)$.
  The elements $u_{\le\hat s}$, $v_{>\hat s}$ for $\tbfi$ are the elements $u_{\le s}$, $v_{>s}$ of $W^\tau$ attached to $\bfi$, so $\Delta(\widetilde{k};\tbfi)=\Delta_{u_{\le s}\varpi_x,v_{>s}\varpi_x}$.
  The first claim now follows from \cref{lem: minors restrict}, and the second from \cite[Theorem~2.10]{BFZ05}.
\end{proof}

\begin{lemma}
  \label{lem: fold BFZ}
  Let $\bfj$ be a word in $\pm[1,n]$ and $\widetilde{\boldsymbol{\jmath}}$ its lift.
  For every $\bfj$-exchangeable $\ell$, every $k\in C(\bfj)$ and every chamber $\widetilde{\ell}$ over $\ell$, the entry of $\tB(\bfj)$ at $(k,\ell)$ is $\sum_{\widetilde{k}}b_{\widetilde{k}\widetilde{\ell}}$, the sum running over the chambers $\widetilde{k}$ over $k$ and $b_{\widetilde{k}\widetilde{\ell}}$ being the entries of $\tB(\widetilde{\boldsymbol{\jmath}})$.
\end{lemma}
\begin{proof}
  By \eqref{eq: B entries}, an entry depends only on the signed sequences of letters at its row and column levels and on the Cartan entry between them.
  We show that the lift preserves these sequences, and that counting the lifted levels that contribute recovers the Cartan entry.

  Let $k$ and $\ell$ be at the levels $x$ and $y$, and let $\widetilde{y}\in\{y,\tau(y)\}$ be the level of the chosen chamber $\widetilde{\ell}$.

  If $x=y$, only the chamber over $k$ at the level $\widetilde{y}$ can contribute: any other chamber over $k$ is at a non-adjacent level.
  The lift preserves the order and signs of the letters at each level, so the first two cases of \eqref{eq: B entries} give the required equality.
  If $x\neq y$ and $a_{xy}=0$, no level over $x$ is adjacent to $\widetilde{y}$, and both sides vanish.

  Suppose now that $a_{xy}\neq0$ and $x\neq y$.
  Only the chambers $\widetilde{k}$ at levels $\widetilde{x}\in\{x,\tau(x)\}$ adjacent to $\widetilde{y}$ can contribute.
  Restricting $\bfj$ to the levels $x,y$ and its lift to $\widetilde{x},\widetilde{y}$ gives the same signed sequence after relabeling, with $k,\ell$ corresponding to $\widetilde{k},\widetilde{\ell}$.
  The third case of \eqref{eq: B entries} therefore gives the same entry, except that the factor $-a_{xy}$ is replaced by $1$.
  There are exactly $-a_{xy}$ such levels $\widetilde{x}$: two, $n-1$ and $n+1$, if $(x,y)=(n-1,n)$, and one otherwise.
  The claim follows by summing these contributions.
\end{proof}

\begin{theorem}
  \label{thm: fold model}
  Index the fold of $\Sigma_{\tg}(\tbfi)$ by $C(\bfi)$ via $O_k\mapsto k$.
  \begin{enumerate}[(a)]
    \item The exchange matrix of the fold of $\Sigma_{\tg}(\tbfi)$ is $\tB_\gamma(\bfi)$.
    \item For $\widetilde{k}$ over $k$, $\Phi_{\tg}(\widetilde{k};\tbfi)|_\mcG=\Phi_\gamma(k;\bfi)$.
    \item The map $h_\gamma$ is birational, and the extended cluster of the fold of $\Sigma_{\tg}(\tbfi)$ is $(x_\gamma(k))_k$.
  \end{enumerate}
  So the fold of $\Sigma_{\tg}(\tbfi)$ is the BD word seed $\Sigma_\gamma(\bfi)$ of \cref{def: model} for $\mcG$ and $\gamma$.
\end{theorem}
\begin{proof}
  \emph{Inversion letters, thawed chambers and virtual letters.}
We begin by showing that inversion letters and thawed chambers lift in pairs, so the extended words agree up to commutations.
  Let $a$ be a row root of $\gamma$, $g=\gamma_r(a)$, and $i_m$ its inversion letter, at the position $m=m^\rr_a$ and level $x$, and put $u=u_{\le m-1}\in W^\tau$, so that $u(\alpha^C_x)=\alpha^C_g$.
  As $g<n$ and $u$ preserves root lengths, $x<n$, so $i_m$ lifts to the two letters $-x,-\tau(x)$.
  Since $s^C_k=s_ks_{\tau(k)}$ for $k<n$ and $s^C_n=s_n$, the negative letters of $\tbfi$ preceding these two letters give the product $u\in S_{2n}$.
  Hence the roots of the two letters are $u\alpha_x$ and $us_x\alpha_{\tau(x)}=u\alpha_{\tau(x)}$, as $x$ and $\tau(x)$ are not adjacent.

  As $u$ preserves $\mfh$, the positive root $u\alpha_x$ restricts to $u\alpha^C_x=\alpha^C_g$.
  It is a sum of simple roots, each restricting to a simple root of type $C_n$, so it is $\alpha_g$ or $\alpha_{\tau(g)}$, and $u\alpha_{\tau(x)}=\tau(u\alpha_x)$ is the other one, as $u\in W^\tau$.
  Since $\tg_r(a)=g$ and $\tg_r(\tau(a))=\tau(g)$, the two letters are the inversion letters of $a$ and $\tau(a)$ for $\tg$, and they separate chambers over $c_x(m-1)$ and $c_x(m)$.
  The thawed chambers of $a$ and $\tau(a)$ are the final chambers of $\tbfi$ at the levels $\tau(a)$ and $a$, i.e.\ the orbit over the thawed chamber $t^\rr_a$ of $a$.
  The same holds for column roots.

  As the roots of $\tg$ are the roots $a$ of $\gamma$ and their images $\tau(a)$, the inversion letters of $\tg$ in $\tbfi$ come in consecutive pairs, in the order of the inversion letters of $\gamma$ in $\bfi$.
  By \cref{def: model}(b), the lift of $\bfi^\gamma$ and $\tbfi^{\tg}$ thus differ only in the order of the two virtual letters of each pair, at the non-adjacent levels $a,\tau(a)$, and so are related by commutations.

  \emph{(a)}
  By \cref{lem: fold BFZ} for $\bfj=\bfi^\gamma$ and \cref{lem: commutation}, the orbit sums of $\tB(\tbfi^{\tg})_{\widetilde P}$ are the entries of $\tB(\bfi^\gamma)_P$.
  The nonzero entries of $D_a$ and $D_{\tau(a)}$ for $\tbfi$ are swapped by $\tau$, and their rows and columns lie over those of the nonzero entries of $D_a$ for $\bfi$, by the description of inversion letters and thawed chambers above.
  None of their rows and columns is at the level $n$.
  Indeed, their levels are $a,\tau(a)$ and $x,\tau(x)$, where $a<n$ and, as shown above, $x<n$.
  Consequently, for a nonzero entry of $D_\gamma(\bfi)$ in the row $k$ and the column $\ell$, and a chamber $\widetilde{\ell}$ over $\ell$, exactly one entry of $D_{\tg}(\tbfi)$ in a row over $k$ and the column $\widetilde{\ell}$ is nonzero.
  It has the same sign, and all nonzero entries of $D_{\tg}(\tbfi)$ arise in this way.
  So the orbit sums of $D_{\tg}(\tbfi)$ are the entries of $D_\gamma(\bfi)$.

  \emph{(b)}
  Let $\widetilde{k}=c_x(\hat s)$ lie over $k=c_{x'}(s)$, and let $u=u_{\le s}\in W^\tau$, so that the row weight of $\widetilde{k}$ is $u\varpi_x$ by the proof of \cref{lem: index minors}, and let $a$ be a row root of $\gamma$, $g=\gamma_r(a)$.
  Then $e^\rr_a(\widetilde{k})+e^\rr_{\tau(a)}(\widetilde{k})=[(u\varpi_x)(\alpha_g^\vee)]_++[(u\varpi_x)(\alpha_{\tau(g)}^\vee)]_+$, and $(u\varpi_x)(\alpha^\vee_{\tau(g)})=(u\varpi_{\tau(x)})(\alpha^\vee_g)$, as $u$ commutes with $w_0$.
  The sets $S=u([1,x])$ and $S'=u([1,\tau(x)])$ are nested, so the two pairings, which are $[g\in S]-[g+1\in S]$ and $[g\in S']-[g+1\in S']$ by \eqref{eq: pairing A concise}, do not have opposite signs, and the sum is
  \[
    \big[(u\varpi_x)(\alpha_g^\vee+\alpha_{\tau(g)}^\vee)\big]_+=\big[(u\varpi^C_{x'})(\alpha^{C\vee}_g)\big]_+=e^\rr_a(k),
  \]
  where $x'=\min(x,\tau(x))$ and $\alpha^{C\vee}_g=[E^C_g,F^C_g]=\alpha^\vee_g+\alpha^\vee_{\tau(g)}$.
  The same calculation applies to columns.
  By \cref{thm: symmetric}, $\Phi_{\tg}(\widetilde{t};\tbfi)|_\mcG$ is the same function for the two thawed chambers $\widetilde{t}$ of an orbit.
  By \cref{lem: index minors} and induction along the recursion \eqref{eq: Phi}, which terminates by \cref{lem: tails},
  \[
    \Phi_{\tg}(\widetilde{k};\tbfi)|_\mcG=\Delta(k;\bfi)\prod_a\Phi_\gamma(t^\rr_a;\bfi)^{e^\rr_a(k)}\prod_b\Phi_\gamma(t^\cc_b;\bfi)^{e^\cc_b(k)}=\Phi_\gamma(k;\bfi).
  \]

  \emph{(c)}
  For $\widetilde{k}\in O_k$, let $\overline{x}_k:=x_{\tg}(\widetilde{k})|_\mcG$; it is regular, and $\overline{x}_k\circ h_\gamma=\Phi_{\tg}(\widetilde{k};\tbfi)|_\mcG=\Phi_\gamma(k;\bfi)$ by (b) and \cref{prop: h theta}.
  The $\Phi_\gamma(k;\bfi)$ are $\dim\mcG$ algebraically independent generators of $\bbC(\mcG)$, as noted after \cref{def: model}.
  So $h_\gamma$ is dominant, and $h_\gamma^*$ is an injective endomorphism of $\bbC(\mcG)$ whose image contains the $\Phi_\gamma(k;\bfi)$. Hence $h_\gamma^*$ is an automorphism, $h_\gamma$ is birational, and $\overline{x}_k=\Phi_\gamma(k;\bfi)\circ h_\gamma^{-1}=x_\gamma(k)$.
  Together with (a) and \cref{thm: symmetric}, this shows that the fold is the seed $\Sigma_\gamma(\bfi)$.
\end{proof}

\begin{example}[\cref{ex: Sp6} continued]
  The lift of $\gamma=(\{1\to2\},0)$ is $\tg=(\{1\to2,5\to4\},0)$.
  By \cref{lem: minors restrict}, $\Delta_{-\varpi_1,\varpi_1}$ and $\Delta_{2\varpi_2-\varpi_3,-\varpi_3}$ are the restrictions of the matrix minors $\Delta_{6,1}$ and $\Delta_{124,456}$.
  The thawed chambers of the roots $1$ and $5$ of $\tg$ in $\tbfi$ are the final chambers at the levels $5$ and $1$, with minors $\Delta_{23456,12345}$ and $\Delta_{6,1}$, which agree on $Sp_6$ by \cref{lem: minors restrict}.
  The chamber of $\tbfi$ over $c_3(1)$ has the minor $\Delta_{124,456}$, and its row weight pairs to $1$ with both $\alpha_2^\vee$ and $\alpha_4^\vee$.
  Hence its function is $\Delta_{124,456}\Delta_{6,1}\Delta_{23456,12345}$, which restricts to $\Phi_\gamma(c_3(1);\bfi)$.
  The exponent $2$ is the sum of the exponents of the two thawed chambers of an orbit, as in the proof of (b).
  Each of the six chambers over $c_1(6)$, $c_2(0)$, $c_2(2)$ has a single one of the two factors.
\end{example}

\subsection{The initial word}
We choose a word for which every dipole stays in one half of the lifted diagram.
The absence of entries between the two halves will make the involution of the lifted seed admissible at every exchangeable orbit.

Let $K:=[1,n-1]$, so that $W_K\subseteq W^\tau$ is of type $A_{n-1}$; in $S_{2n}$, $w_0^K$ is the longest element of the parabolic subgroup generated by the $s_k$, $k\neq n$.
Since $w_0$ is central in $W^\tau$, $w_0w_0^K$ is an involution; in $S_{2n}$ it is the permutation $\sigma:c\mapsto c\pm n$ swapping the two halves of $[1,2n]$ and preserving the order inside each of them.

\begin{definition}
  \label{def: split}
  Let $\mathbf w:=(-n)(-n+1,-n)\cdots(-1,-2,\dots,-n)$ and $\mathbf l:=(-1)(-2,-1)\cdots(-n+1,\dots,-2,-1)$, whose levels form reduced words for $w_0w_0^K$ and $w_0^K$.
  Let $\bfi_0:=(\mathbf w,\mathbf l,\mathbf l^T,\mathbf w^T)$, so that $\bfi_0^T=\bfi_0$.
\end{definition}

The negative part of $\bfi_0$ is a reduced word for $w_0w_0^K\,w_0^K=w_0$, and so is the positive part by symmetry.
  So $\bfi_0$ is a double reduced word of type $C_n$.

\begin{lemma}
  \label{lem: split halves}
  In $\tbfi_0$, the row and column inversion letters of a root $a<n$ of $\tg$ lie at levels $>n$, and those of a root $a>n$ at levels $<n$.
  Consequently, the entries of $\tB_{\tg}(\tbfi_0)$ between a lower chamber and an upper one vanish, and the involution of $\Sigma_{\tg}(\tbfi_0)$ is admissible at every exchangeable orbit.
\end{lemma}
\begin{proof}
  In decreasing order of position, the positive letters of $\tbfi_0$ first come from the lift of $\mathbf w^T$, and the product of their reflections is $\sigma$.
  Their roots are the positive roots $\beta$ with $\sigma\beta<0$, so they swap values $c<d$ with $\sigma(c)>\sigma(d)$, i.e.\ $c\le n<d$, and none of them has a simple root $\alpha_g$, $g\neq n$.
  So the column inversion letter of a root $a$ of $\tg$, at a position $m$ and level $x$, lies in the lift of $\mathbf l^T$.
  There $v_{>m}=\sigma v'$ with $v'$ in the parabolic subgroup generated by the $s_k$, $k\neq n$.
  Then $\sigma v'(\alpha_x)=\alpha_g$, $g=\tg_c(a)$, gives $v'(\alpha_x)=\sigma(\alpha_g)$, which is $\alpha_{g+n}$ if $g<n$ and $\alpha_{g-n}$ if $g>n$, so $x>n$ if $a<n$ and $x<n$ if $a>n$.
  For rows we argue in the same way with the negative letters in increasing order of position, which by symmetry have the levels of the positive letters in decreasing order.
  The thawed chamber of a root $a<n$ lies at the level $\tau(a)>n$, in the same half as its inversion letter.
  Hence its dipole entries are in rows and columns of upper chambers, and symmetrically for $a>n$.
  All other nonzero entries have row and column chambers whose levels differ by at most one (\cref{lem: model local}).
  Admissibility follows by considering the two cases: for a non-fixed chamber $\widetilde{k}$ and an exchangeable orbit $\{\widetilde{\ell},\tau(\widetilde{\ell})\}$ of size two, at most one of $b_{\widetilde{k}\widetilde{\ell}}$ and $b_{\widetilde{k}\tau(\widetilde{\ell})}$ is nonzero, while for a fixed chamber $\widetilde{k}$, $b_{\widetilde{k}\tau(\widetilde{\ell})}=b_{\widetilde{k}\widetilde{\ell}}$.
\end{proof}

\section{Proof of \texorpdfstring{\cref{thm: main}}{the main theorem}}
\label{sec: main}
\subsection{The cluster structure}
\begin{definition}
  \label{def: C gamma}
  The cluster structure $\mcC_\gamma$ on $\mcG=\Sp$ is the one defined by the BD word seed $\Sigma_\gamma(\bfi_0)$ (\cref{def: split,def: model,thm: fold model}).
  We write $\oA_\gamma$ for its upper cluster algebra.
\end{definition}

\begin{proposition}
  \label{prop: adjacent}
  For every exchangeable chamber $k$ of $\Sigma_\gamma(\bfi_0)$, the fold of the $\vt$-symmetric seed $\mu_{O_k}\big(\Sigma_{\tg}(\tbfi_0)\big)$ of $\mcC_{\tg}$ is $\mu_k\big(\Sigma_\gamma(\bfi_0)\big)$.
  In particular the cluster variables of the seeds adjacent to $\Sigma_\gamma(\bfi_0)$ are restrictions to $\mcG$ of cluster variables of $\mcC_{\tg}$, hence regular functions on $\mcG$.
\end{proposition}
\begin{proof}
  The claim follows from \cref{prop: fold mutation}, by \cref{thm: symmetric,thm: fold model,lem: split halves,thm: GSV}.
\end{proof}

\begin{theorem}
  \label{thm: structure}
  The cluster structure $\mcC_\gamma$ has the properties (i), (iii) and (iv) of the introduction:
  \begin{enumerate}[(i)]
    \item $\mcC_\gamma$ has $2n-|\Gamma_r|-|\Gamma_c|$ frozen variables, and its exchange matrices have full rank.
    \setcounter{enumi}{2}
    \item The global toric action on $\mcC_\gamma$ is generated by the action of $\mcH_\gamma$.
    \item $\mcC_\gamma$ is compatible with $\{\cdot,\cdot\}_{\gamma,s}$ for every $s\in\mcS(\gamma)$.
  \end{enumerate}
\end{theorem}
\begin{proof}
  (i) The frozen chambers of $\bfi_0$ are the $2n$ end chambers other than the $|\Gamma_r|+|\Gamma_c|$ thawed chambers.

  (iv) Let $s\in\mcS(\gamma)$ and $\widetilde{s}$ as in \cref{sec: lift}.
  By \cref{thm: GSV}(iv), $\{\cdot,\cdot\}_{\tg,\widetilde{s}}$ is log-canonical with respect to the extended cluster of the seed $\Sigma_{\tg}(\tbfi_0)$ of $\mcC_{\tg}$, with coefficient matrix satisfying \eqref{eq: compatibility} with all $\kappa_p\neq0$.
  By \cref{prop: restriction,thm: symmetric,thm: fold model}, \cref{prop: fold compatible} applies, and gives compatibility and full rank.

  (iii) By \cref{thm: GSV}(iii), the cluster variables of $\Sigma_{\tg}(\tbfi_0)$ are homogeneous for the action of $\widetilde{\mcH}_{\tg}$.
  The weights span its character space by the argument of \cref{sec: toric A concise}, as the cluster variables generate $\bbC(\SL)$.
  Also $\tB^TW=0$ for their weight matrix $W$.
  The subtorus $\mcH_\gamma=\widetilde{\mcH}_{\tg}\cap(\mcH\times\mcH)$ (\cref{prop: lift}(c), applied to $\gamma_r$ and $\gamma_c$) preserves $\mcG$ and commutes with $\vt$. By \cref{prop: fold toric}, the weights of the extended cluster of $\Sigma_\gamma(\bfi_0)$ for $\mcH_\gamma$ span a space of dimension $\dim\mfh_\gamma=2n-|\Gamma_r|-|\Gamma_c|$, and $\overline{B}^T\overline{W}=0$ for their weight matrix $\overline{W}$, where $\overline{B}=\tB_\gamma(\bfi_0)$.
  As $\overline{B}$ has full rank, this is a global toric action of maximal rank.
\end{proof}

\subsection{Regularity and completeness}
We obtain Laurent expressions for regular functions by restriction from type A, and use factoriality to exclude denominators in the upper cluster algebra; this proves regularity and completeness.

\begin{lemma}
  \label{lem: primes}
  Let $R$ be a factorial domain with $R^\times=\bbC^*$, and $(\bx,\tB)$ a seed in its fraction field with $\tB$ of full rank, such that $\bx\subseteq R\subseteq\mcL(\bx)\cap\bigcap_{k\in P_\ex}\mcL(\mu_k(\bx))$.
  Then the exchangeable elements of $\bx$ are pairwise non-associate primes of $R$, and $x'_k/x_k\notin R$ for every $k\in P_\ex$.
\end{lemma}
\begin{proof}
  As $\bx\subseteq R\subseteq\mcL(\bx)$, we have $R[x_k^{-1}\mid k\in P_\ex]=\mcL(\bx)$.
  The units of $\mcL(\bx)$ are the elements $c\prod_kx_k^{a_k}$, $c\in\bbC^*$, by algebraic independence, and those of $R[x_k^{-1}]$ are the elements $c\prod_jq_j^{b_j}$, where $q_1,\dots,q_s$ represent the primes dividing $\prod_kx_k$.
  Comparing these descriptions, we obtain $s=|P_\ex|$ and mutually inverse integer matrices $F=(F_{kj})$ and $G=(G_{jk})$ such that $x_k=c_k\prod_jq_j^{F_{kj}}$ and $q_j=c'_j\prod_kx_k^{G_{jk}}$; moreover $F_{kj}\ge0$.
  Write $x_kx'_k=M_++M_-$, with distinct monomials $M_\pm$ not involving $x_k$ ($\tB$ has no zero column).
  The exchange monomials have disjoint supports, so $M_++M_-$ is coprime to every monomial in the adjacent Laurent ring and, being a sum of distinct monomials, is not a unit there.
  If $G_{jk}<0$, then $q_j=(x'_k)^{-G_{jk}}(M_++M_-)^{G_{jk}}M'$ with $M'$ a Laurent monomial in $\mu_k(\bx)\setminus\{x'_k\}$; this cannot belong to $\mcL(\mu_k(\bx))$, a contradiction.
  So $F$ and $G$ have non-negative entries.
  Every column $k$ of $F$ has an entry $F_{ik}>0$, and then $0=(FG)_{ij}\ge F_{ik}G_{kj}\ge0$ for $j\neq i$, so the only non-zero entry of the row $k$ of $G$ is in the column $i$.
  Hence the invertible matrix $G$ is monomial, and so is $F=G^{-1}$; their non-zero entries are positive integers inverse to each other, hence equal to $1$.
  So $F$ is a permutation matrix, that is, the $x_k$ are pairwise non-associate primes.

  If $x'_k=x_kg$ with $g\in R$, then $g=(x'_k)^2/(M_++M_-)\in\mcL(\mu_k(\bx))$, which is a contradiction for the same reason.
\end{proof}

Let $\bx$ be the extended cluster of $\Sigma_\gamma(\bfi_0)$, with index set $P=C(\bfi_0)$.

\begin{lemma}
  \label{lem: Laurent}
  $\bbC[\mcG]\subseteq\mcL(\bx)\cap\bigcap_{k\in P_\ex}\mcL(\mu_k(\bx))$.
\end{lemma}
\begin{proof}
  Let $f\in\bbC[\mcG]$; as $\mcG$ is a closed subvariety of the affine variety $\SL$, restriction $\bbC[\SL]\to\bbC[\mcG]$ is surjective, so $f=F|_\mcG$ with $F\in\bbC[\SL]$.
  By \cref{thm: GSV}(ii), $F$ is a Laurent polynomial in the extended clusters $\by$ of $\Sigma_{\tg}(\tbfi_0)$ and of $\mu_{O_k}(\Sigma_{\tg}(\tbfi_0))$ for every $k\in P_\ex$, polynomial in the frozen variables.
  By \cref{prop: adjacent}, restriction to $\mcG$ maps the variables of $\by$ to those of $\bx$, resp.\ $\mu_k(\bx)$, exchangeable to exchangeable and frozen to frozen; so $f$ is a Laurent polynomial of the same kind.
\end{proof}


\begin{theorem}
  \label{thm: complete}
  The cluster structure $\mcC_\gamma$ has the property (ii) of the introduction: $\oA_\gamma=\bbC[\mcG]$; in particular $\mcC_\gamma$ is regular and complete.
\end{theorem}
\begin{proof}
  By \cref{lem: Laurent}, $\bbC[\mcG]$ is contained in the upper bound of $\Sigma_\gamma(\bfi_0)$, which equals $\oA_\gamma$ by full rank (\cref{thm: structure}).
  Conversely, let $f\in\oA_\gamma$.
  The ring $R=\bbC[\mcG]$ is factorial \cite[Proposition~1 and Corollary]{Pop74}, \cite[Theorem~3]{KP83}, and its units are the non-zero constants: for a unit $f$ and $U\in\mcG$, $t\mapsto f(U\exp(tE_i))$ and $t\mapsto f(U\exp(tF_i))$ are nowhere vanishing polynomials, hence constant, and these one-parameter subgroups generate $\mcG$. So, by \cref{prop: adjacent,lem: Laurent}, \cref{lem: primes} applies to $\bx$ and $R$.
  Write $f=g/M$ in lowest terms, with $g\in R$ and $M$ a product of exchangeable variables of $\bx$, which is possible as $f\in\mcL(\bx)$.
  Suppose that $x_k$ divides $M$.
  As $f\in\mcL(\mu_k(\bx))$, $f=g'/M'$ with $g'\in R$ and $M'$ a monomial in $x'_k$ and the exchangeable $x_\ell$, $\ell\neq k$.
  Then $gM'=g'M$, and since $x_k$ divides $M$ but not $g$, the prime $x_k$ divides $x'_k$ or some $x_\ell$, which \cref{lem: primes} excludes.
  So $f\in R$. Every cluster variable lies in $\oA_\gamma$ by the Laurent phenomenon in geometric type, polynomial in the frozen variables \cite[Proposition~11.2]{FZ03}, hence is regular.
\end{proof}

\subsection{Arbitrary words}
\begin{theorem}
  \label{thm: all words}
  The first claim of \cref{thm: main} holds: for every double reduced word $\bfi$ of type $C_n$, $\Sigma_\gamma(\bfi)$ is a seed of $\mcC_\gamma$, and its cluster variables are restrictions of cluster variables of $\mcC_{\tg}$.
  Commutations do not change $\Sigma_\gamma(\bfi)$, sign swaps and braid moves of length $3$ act on it by the mutation at the replaced chamber, and braid moves of length $4$ by $\mu_k\mu_\ell\mu_k$, where $k$ and $\ell$ are the replaced chambers at the levels $n$ and $n-1$.
\end{theorem}
\begin{proof}
  The action of the moves is \cref{thm: moves}, as $h_\gamma$ is birational (\cref{thm: fold model}(c)) and $\alpha_n$ is the long simple root.
  So $\Sigma_\gamma(\bfi)$ is a seed of $\mcC_\gamma$, as every double reduced word is connected to $\bfi_0$ by moves.
  By \cref{thm: fold model}, $\Sigma_\gamma(\bfi)$ is the fold of $\Sigma_{\tg}(\tbfi)$, which is a seed of $\mcC_{\tg}$ by \cref{thm: model concise}.
\end{proof}

\subsection{Compatible Poisson brackets}
\label{sec: uniqueness}
We classify compatible brackets for $SL_N$ and $\Sp$.
By block-indecomposability, the coefficients $\kappa_p$ in \eqref{eq: compatibility} of a compatible bracket are a common multiple of those of $\{\cdot,\cdot\}_{\gamma,s}$, and the toric action of $\mcH_\gamma$ accounts for the remaining difference.

For $\mcG=SL_N$, let $\gamma$ be an aperiodic BD pair of type $A_{N-1}$, $\mcC_\gamma$ the cluster structure of \cref{thm: GSV} and $\Sigma=(\bx,\tB)$ its seed $\Sigma_\gamma(\bfi_{\rm GSV})$.
For $\mcG=\Sp$, let $\gamma$ be an aperiodic BD pair of type $C_n$, $\mcC_\gamma$ the cluster structure of \cref{def: C gamma} and $\Sigma=(\bx,\tB)$ its seed $\Sigma_\gamma(\bfi_0)$.

For $\xi=(\xi_r,\xi_c)\in\mfh_\gamma$ let $\partial_\xi:=\partial^\rr_{\xi_r}+\partial^\cc_{\xi_c}$, the infinitesimal action of $\mcH_\gamma$.
For $\Xi=\sum_m\xi_m\otimes\eta_m\in\bigwedge^2\mfh_\gamma$ let $\mcV_\Xi(f,g):=\sum_m\partial_{\xi_m}f\,\partial_{\eta_m}g$, and write $\Xi=\Xi_{\rm rr}+\Xi_{\rm rc}+\Xi_{\rm cc}$ with $\Xi_{\rm rr}\in\bigwedge^2\mfh_{\gamma_r}$, $\Xi_{\rm rc}\in\mfh_{\gamma_r}\otimes\mfh_{\gamma_c}$, $\Xi_{\rm cc}\in\bigwedge^2\mfh_{\gamma_c}$, where $\mfh_{\gamma_r}\otimes\mfh_{\gamma_c}$ is embedded by $a\otimes b\mapsto(a,0)\wedge(0,b)$.

\begin{theorem}
  \label{thm: uniqueness}
  Property (v) holds for $\mcC_\gamma$, for $\Sp$ and for $SL_N$: a Poisson bracket on $\bbC(\mcG)$ is compatible with $\mcC_\gamma$ if and only if it is of the form
  \[
    c\{\cdot,\cdot\}_{\gamma,s}+\mcV_\Xi,
    \qquad
    c\in\bbC,\quad s\in\mcS(\gamma),\quad \Xi\in\textstyle\bigwedge^2\mfh_\gamma.
  \]
  If $c\neq0$, the parts $\Xi_{\rm rr}$, $\Xi_{\rm cc}$ can be absorbed by changing $s$, and the bracket is a multiple of a bracket $\{\cdot,\cdot\}_{\gamma,s'}$ if and only if $\Xi_{\rm rc}=0$.
\end{theorem}
\begin{proof}
  We use (i), (iii) and (iv) of \cref{thm: GSV}, resp.\ of \cref{thm: structure}.
  Namely, $\tB$ has full rank, and $\{\cdot,\cdot\}_{\gamma,s}$ is compatible, with coefficient matrix $\Omega$ in $\bx$ satisfying \eqref{eq: compatibility} with all $\kappa_p\neq0$ (for $\Sp$ by \cref{prop: fold compatible}).
  Also, the left and right action of $\mcH_\gamma$ is a global toric action whose weight matrix $W$ in $\bx$ has rank $\dim\mfh_\gamma=|P_\fr|$.

  \emph{The brackets are Poisson and compatible.}
  For $y_r\in\mcH_{\gamma_r}$, $\mathrm{Ad}_{y_r}$ fixes $\mfh$ and commutes with $\gamma_r$ on $\mfn_\pm$, as $y_r^{\gamma_r(\alpha)}=y_r^\alpha$, so it fixes $r_\rr$. Hence left multiplication by $y_r$ is a Poisson automorphism of $\{\cdot,\cdot\}_{\gamma,s}$, and similarly for right multiplication by $\mcH_{\gamma_c}$.
  So the vector fields $\partial_\xi$, $\xi\in\mfh_\gamma$, preserve $\{\cdot,\cdot\}_{\gamma,s}$, and they pairwise commute, as $\mcH_\gamma$ is abelian.
  Hence $\mcV_\Xi$ is a Poisson bracket whose Schouten bracket with $\{\cdot,\cdot\}_{\gamma,s}$ vanishes, and $c\{\cdot,\cdot\}_{\gamma,s}+\mcV_\Xi$ is Poisson.
  It is log-canonical in every cluster, as $\{\cdot,\cdot\}_{\gamma,s}$ is compatible and all cluster variables are $\mcH_\gamma$-homogeneous.

  \emph{Every compatible bracket is of this form.}
  Let $\{\cdot,\cdot\}'$ be compatible, with coefficient matrix $\Omega'$ in $\bx$.
  For $p$ exchangeable and $q\neq p$, $\{\log(M_++M_-),\log x_q\}$ is constant, where $x_px'_p=M_++M_-$; since $M_\pm$ are distinct monomials, $\{\log M_+,\log x_q\}=\{\log M_-,\log x_q\}$, i.e.\ $\sum_{q'}b_{q'p}\omega'_{q'q}=0$.
  So $\Omega'$ satisfies \eqref{eq: compatibility} for some $\kappa'_p$.
  The matrix $\tB^T\Omega'\tB$ is skew-symmetric with $(p,q)$ entry $\kappa'_pb_{pq}$, so $\kappa'_pb_{pq}=-\kappa'_qb_{qp}$, and similarly for the $\kappa_p$. As the $\kappa_p$ are nonzero, $\kappa'_p/\kappa_p=\kappa'_q/\kappa_q$ whenever $b_{pq}\neq0$. By \cref{lem: indecomposable}, the graph on $P_\ex$ with these edges is connected, so this ratio is a constant $c$, i.e.\ $\kappa'_p=c\kappa_p$ for all $p$.
  Then $\Upsilon:=\Omega'-c\Omega$ satisfies $\tB^T\Upsilon=0$.
  By full rank, $\ker\tB^T$ has dimension $|P_\fr|=\dim\mfh_\gamma$, and it is spanned by the columns of $W$, as $\tB^TW=0$.
  As $\Upsilon$ is skew-symmetric with columns in $\ker\tB^T$, $\Upsilon=W\Xi W^T$ for a unique skew-symmetric $\Xi$, which is the coefficient matrix of $\mcV_\Xi$; so $\{\cdot,\cdot\}'=c\{\cdot,\cdot\}_{\gamma,s}+\mcV_\Xi$.

  \emph{The last claims.}
  Replacing $s_r$ by $s_r+\delta$, $\delta\in\mfh_{\gamma_r}\wedge\mfh_{\gamma_r}$, changes $\{f,g\}_{\gamma,s}$ by $-(\lambda\otimes\lambda')(\delta)fg$ for $f,g$ of bidegrees $(\lambda,\mu)$, $(\lambda',\mu')$, by \eqref{eq: bracket}, i.e.\ it adds $\mcV_\Xi$ with $\Xi\in\bigwedge^2\mfh_{\gamma_r}$ determined by $-\delta$.
  As $\mcS(\gamma_r)=s_r+\mfh_{\gamma_r}\wedge\mfh_{\gamma_r}$ (\cref{sec: BD}), this absorbs $\Xi_{\rm rr}$, and similarly $\Xi_{\rm cc}$.
  Conversely, if $c(\{\cdot,\cdot\}_{\gamma,s}+\mcV_{\Xi_{\rm rc}})=c'\{\cdot,\cdot\}_{\gamma,s'}$, comparing the $\kappa_p$ gives $c=c'$, and then $\mcV_{\Xi_{\rm rc}}=\mcV_{\Xi'}$ with $\Xi'\in\bigwedge^2\mfh_{\gamma_r}\oplus\bigwedge^2\mfh_{\gamma_c}$; by uniqueness of $\Xi$, $\Xi_{\rm rc}=0$.
\end{proof}

\begin{remark}
  \label{rmk: v}
  By \cref{thm: uniqueness}, a compatible bracket with $c\neq0$ is a multiple of a bracket in (iv) of the introduction exactly when $\Xi_{\rm rc}=0$.
  Mixed terms exist for every pair, as $\dim\mfh_{\gamma_\bullet}=|\Gamma|-|\Gamma_\bullet|\ge1$; they correspond to the term $V_{12}$ in \cite[Section~4]{GSV11}.
  For instance, for $\gamma=(0,0)$ on $SL_2$ and $\Xi=\alpha_1^\vee\otimes\alpha_1^\vee\in\mfh_{\gamma_r}\otimes\mfh_{\gamma_c}$, the bracket $\mcV_\Xi$ gives $\{x_{11},x_{12}\}=-2x_{11}x_{12}$ and $\{x_{11},x_{22}\}=0$, so it is not a multiple of the standard bracket.

  The original \cite[Conjecture~3.2(v)]{GSV11} concerns Poisson--Lie brackets; in our conventions $\{\cdot,\cdot\}_{\gamma,s}$ is a Sklyanin bracket when $\gamma_c=\gamma_r^{-1}$ and $s_c=-s_r$.
  For $c\neq0$, a bracket of \cref{thm: uniqueness} can be Poisson--Lie only if $\gamma_c=\gamma_r^{-1}$, and then it is Poisson--Lie exactly when it is a multiple of a Sklyanin bracket of the BD class of $\gamma_r$ \cite[Lemma~4.3(ii)]{GSV11}; in particular, multiplicativity excludes mixed terms.
  For $c=0$, some toric brackets are Poisson--Lie as well: if $\gamma_c=\gamma_r^{-1}$ and $0\neq\lambda\in\bigwedge^2\mfh_{\gamma_r}$, then $\lambda(f,g)^\rr-\lambda(f,g)^\cc$ is the difference of two Sklyanin brackets of the BD class of $\gamma_r$, hence multiplicative, and it is Poisson and compatible with $\mcC_\gamma$ by \cref{thm: uniqueness}, but it is not a multiple of such a bracket, whose $R$-matrix has non-zero root components.
  Such $\lambda$ exist when $\dim\mfh_{\gamma_r}\ge2$, for instance for $\gamma=(0,0)$ on $SL_3$ or $Sp_4$.
  Thus, for $c\neq0$, the compatible Poisson--Lie brackets have the form specified in \cite[Conjecture~3.2(v)]{GSV11}, and for $c=0$ there are also these toric brackets.
\end{remark}
